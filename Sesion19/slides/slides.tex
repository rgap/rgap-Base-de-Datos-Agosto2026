Índice & Una lista ordenada que dice dónde está cada fila. & CREATE INDEX idx\_cliente\_nombre ON cliente (nombre); \\

\texttt{COALESCE} / \texttt{NULLIF} & Un repuesto para el \texttt{NULL}; convertir un valor en \texttt{NULL}. & COALESCE(telefono, 'sin teléfono') \\

\texttt{CASE WHEN} & Decide un valor según una condición. & CASE WHEN sueldo >= 3000 THEN 'alto' ELSE 'bajo' END \\
Agregación & Muchas filas, un solo valor. & COUNT(*), SUM, AVG, MIN, MAX \\
\texttt{GROUP BY} / \texttt{HAVING} & Un resultado por grupo; filtrar los grupos. & GROUP BY area HAVING COUNT(*) >= 2 \\

\begin{frame}{Revisión: el orden de las columnas importa}
\small Con el índice \texttt{(categoria, precio)}, ¿qué consultas salen rápido?
\vspace{0.1cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{>{\ttfamily\footnotesize\raggedright\arraybackslash}p{7.0cm}>{\centering\arraybackslash}p{1.5cm}>{\raggedright\arraybackslash}p{4.5cm}}
\toprule
\normalfont\bfseries La consulta tiene\ldots & \bfseries ¿Lo usa? & \bfseries Por qué \\
\midrule
WHERE categoria = 'Muebles' & Sí & usa la \textbf{primera} columna \\
WHERE categoria = 'Muebles' AND precio < 700 & Sí & usa las \textbf{dos} columnas \\
WHERE categoria = 'Muebles' ORDER BY precio & Sí & ya vienen \textbf{ordenados} por precio \\
WHERE precio < 700 & \mal{No} & se \textbf{saltó} la primera columna \\
\bottomrule
\end{tabular}
\end{center}
\vspace{-0.1cm}
\concepto{La regla}{\small El índice se usa \textbf{de izquierda a derecha}. Pon primero la columna que se compara con \texttt{=}, y después la del rango o la del \texttt{ORDER BY}.}
\vspace{0.05cm}
\analogia{En la guía buscas rápido a los «Pérez» o a «Pérez, Ana». Pero no a todas las «Ana»: están repartidas por toda la guía.}
\end{frame}

\column{.50\textwidth}
\small Cada filtro que elige el cliente se vuelve \textbf{una parte} de la consulta:
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} nombre, precio\\
\kw{FROM} producto\\
\kw{WHERE} \nuevo{categoria = 'Hogar'}\\
\ind \kw{AND} \nuevo{precio \kw{BETWEEN} 2000 \kw{AND} 2500}\\
\kw{ORDER BY} \nuevo{precio}\\
\kw{LIMIT} 10;
\end{sqlcode}
\vspace{-0.15cm}
\small Esta consulta se repite \textbf{cada vez} que alguien filtra. El índice que la acelera:
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{CREATE INDEX} idx\_producto\_categoria\_precio\\
\ind \kw{ON} producto (\nuevo{categoria}, \nuevo{precio});
\end{sqlcode}
\end{columns}
\vspace{0.1cm}
\small\color{utpGris} Primero \texttt{categoria}, que va con \texttt{=}; después \texttt{precio}, que es un rango y además ordena.
\end{frame}

\begin{frame}{\ins{Práctica 1} ¿Índice? ¿Sobre qué columnas?}
\small La tabla \tbl{pedido}\texttt{(id, cliente\_id, fecha, estado, total)} de una tienda en línea tiene \textbf{dos millones} de filas.
\vspace{0.15cm}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{>{\raggedright\arraybackslash}p{9.2cm}>{\raggedright\arraybackslash}p{3.8cm}}
\toprule
\bfseries Situación & \bfseries ¿Índice? ¿Sobre qué? \\
\midrule
1. La página «Mis pedidos»: \texttt{WHERE cliente\_id = ? ORDER BY fecha} & \bl[3.6cm] \\[4pt]
2. El almacén ve los pendientes: \texttt{WHERE estado = 'pendiente' AND fecha >= ?} & \bl[3.6cm] \\[4pt]
3. Abrir un pedido por su número: \texttt{WHERE id = ?} & \bl[3.6cm] \\[4pt]
4. Un reporte que se corre \textbf{una vez al año}: \texttt{WHERE total > 1000} & \bl[3.6cm] \\[4pt]
5. La tabla \tbl{metodo\_pago} tiene solo 3 filas: efectivo, tarjeta, Yape. & \bl[3.6cm] \\[4pt]
6. Ya existe \texttt{(cliente\_id, fecha)}. ¿Sirve para \texttt{WHERE fecha = ?} & \bl[3.6cm] \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.15cm}
\small \textbf{Para escribir:} el \texttt{CREATE INDEX} de las situaciones 1 y 2. \textbf{Pista:} el número de pedido es la clave primaria.
\end{frame}

\begin{frame}{Revisión: todo junto, lo que puede llevar un \texttt{SELECT}}
\begin{center}
\begin{tikzpicture}[y=0.54cm]
  \renewcommand{\piezax}{7.4}
  \pieza{7}{\kw{SELECT} c.nombre, \kw{COUNT}(*) \kw{AS} pedidos}{qué columnas o cálculos se \textbf{muestran}}
  \pieza{6}{\kw{FROM} cliente c}{la tabla \textbf{principal}}
  \pieza{5}{\kw{JOIN} pedido p \kw{ON} c.id = p.cliente\_id}{qué otra tabla se \textbf{une}, y por dónde}
  \pieza{4}{\kw{WHERE} p.fecha >= '2026-01-01'}{qué \textbf{filas} entran}
  \pieza{3}{\kw{GROUP BY} c.nombre}{cómo se \textbf{agrupan}}
  \pieza{2}{\kw{HAVING} \kw{COUNT}(*) >= 2}{qué \textbf{grupos} se quedan}
  \pieza{1}{\kw{ORDER BY} pedidos \kw{DESC}}{en qué \textbf{orden} salen}
  \pieza{0}{\kw{LIMIT} 5;}{cuántas filas, \textbf{como máximo}}
\end{tikzpicture}
\end{center}
\vspace{-0.05cm}
\begin{itemize}\setlength\itemsep{3pt}
  \item \texttt{SELECT} indica qué mostrar. \texttt{FROM} se usa al consultar tablas. Lo demás se agrega si hace falta.
  \item Si se usan, se escriben \textbf{siempre en este orden}.
  \item \texttt{SELECT DISTINCT} quita las filas \textbf{repetidas} del resultado.
\end{itemize}
\vspace{0.05cm}
\small\color{utpGris} Se lee: «los 5 clientes con más pedidos este año, entre los que tienen 2 o más».
\end{frame}
