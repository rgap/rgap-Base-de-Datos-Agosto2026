\begin{frame}{Revisión: una pregunta que necesita subconsulta}
\renewcommand{\arraystretch}{0.82}
\preg{¿Qué pagos están \textbf{por encima del promedio}?}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\centering\etiqueta{Así no se puede}
\begin{sqlcode}[\col]
\kw{SELECT} id, monto \kw{FROM} pago\\
\kw{WHERE} monto > \mal{\kw{AVG}(monto)};\\[4pt]
\err{ERROR 1111 (HY000):}\\
\err{Invalid use of group function}
\end{sqlcode}
\small \texttt{WHERE} revisa \textbf{una fila a la vez}, pero el promedio necesita \textbf{todas} las filas.
\column{.5\textwidth}
\centering\etiqueta{Con subconsulta}
\begin{sqlcode}[\col]
\kw{SELECT} id, monto, metodo \kw{FROM} pago\\
\kw{WHERE} monto > \dentro{(\kw{SELECT AVG}(monto) \kw{FROM} pago)}\\
\kw{ORDER BY} id;
\end{sqlcode}
\vspace{-0.1cm}
\res{rrl}{\cab{id} & \cab{monto} & \cab{metodo}}{2 & 55.00 & tarjeta \\ 4 & 55.00 & efectivo \\ 8 & 55.00 & transferencia \\}
\end{columns}
\concepto{¿Por qué una subconsulta?}{\small La de adentro calcula \textbf{primero} el promedio (\texttt{37.500000}) y después cada fila se compara con ese número.}
\end{frame}

\begin{frame}{Quiero cada pago y, al lado, el promedio}
\small Para comparar, lo primero que se nos ocurre es pedir las columnas y el promedio juntos:
\begin{sqlcode}[.8\textwidth]
\kw{SELECT} \mal{id}, \mal{monto}, \kw{AVG}(monto) \kw{AS} promedio\\
\kw{FROM} pago;\\[5pt]
\scriptsize\err{-- MySQL: ERROR 1140: In aggregated query without GROUP BY, expression \#1}\\
\scriptsize\err{-- \ \ of SELECT list contains nonaggregated column 'peluqueria.pago.id'\ldots}\\
\scriptsize\err{-- SQL Server: Msg 8120: Column 'pago.id' is invalid in the select list because}\\
\scriptsize\err{-- \ \ it is not contained in either an aggregate function or the GROUP BY clause.}
\end{sqlcode}
\vspace{-0.2cm}
\begin{center}
\begin{tikzpicture}
  \node (a) {\forma{|r|r|r|r|r|r|r|r|}{1 & 2 & 3 & 4 & 5 & 6 & 7 & 8\\ \hline}};
  \node[below=1pt of a,font=\scriptsize] {\texttt{id}: 8 valores};
  \node[right=2.2cm of a] (b) {\forma{|r|}{37.500000\\ \hline}};
  \node[below=1pt of b,font=\scriptsize] {\texttt{AVG}: 1 valor};
  \node[font=\Large\bfseries,text=utpRojo] at ($(a)!.5!(b)$) {?};
\end{tikzpicture}
\end{center}
\vspace{0.1cm}
\concepto{Por qué falla}{\small \texttt{AVG} deja \textbf{una sola fila}, pero \texttt{id} y \texttt{monto} tienen \textbf{ocho} valores: el gestor no sabe cuál poner. Necesito el promedio \textbf{sin juntar} las filas.}
\end{frame}

\begin{frame}{\texttt{OVER}: el promedio en cada fila}
\renewcommand{\arraystretch}{0.82}
\begin{sqlcode}[.75\textwidth]
\kw{SELECT} id, monto, \dentro{\kw{AVG}(monto) \kw{OVER} ()} \kw{AS} promedio\\
\kw{FROM} pago\\
\kw{ORDER BY} id;
\end{sqlcode}
\vspace{-0.15cm}
{\centering\small\textit{Se lee:} «Muestra el \texttt{id} y el \texttt{monto} de cada pago y, al lado, el \textbf{promedio} de \texttt{monto} calculado sobre \textbf{todas las filas}, sin juntarlas».\par}
\vspace{0.05cm}
\begin{columns}[c,onlytextwidth]
\column{.48\textwidth}
\centering
\res{rrr}{\cab{id} & \cab{monto} & \cab{promedio}}{1 & 35.00 & 37.500000 \\ \fila 2 & 55.00 & 37.500000 \\ 3 & 15.00 & 37.500000 \\ \fila 4 & 55.00 & 37.500000 \\ 5 & 35.00 & 37.500000 \\ 6 & 15.00 & 37.500000 \\ 7 & 35.00 & 37.500000 \\ \fila 8 & 55.00 & 37.500000 \\}
\column{.5\textwidth}
\raggedright\small
\texttt{OVER ()} dice \textbf{sobre qué filas} se calcula: con los paréntesis vacíos, sobre \textbf{las 8}.\\[6pt]
El resultado, 37.5, se escribe en \textbf{cada fila}. Así se ve que los pagos 2, 4 y 8 (cortes de pelo) están \textbf{por encima del promedio}.
\end{columns}
\vspace{0.05cm}
\concepto{Función de ventana}{\small Calcula sobre un grupo de filas, \textbf{la ventana}, y deja cada fila en su lugar. Se reconoce por \texttt{OVER}.}
\end{frame}

\begin{frame}{\texttt{PARTITION BY}: un cálculo por grupo}
\renewcommand{\arraystretch}{0.82}
\small ¿Cuánto trabajo tiene cada empleado? Cada cita y, al lado, cuántas citas tiene \textbf{su} empleado.\par
\vspace{0.3cm}
\begin{columns}[c,onlytextwidth]
\column{.55\textwidth}
\begin{sqlcode}[\col]
\kw{SELECT} id, empleado\_id,\\
\ind\kw{COUNT}(*) \kw{OVER} (\dentro{\kw{PARTITION BY} empleado\_id})\\
\ind\ind\kw{AS} citas\_del\_empleado\\
\kw{FROM} cita\\
\kw{ORDER BY} empleado\_id, id;
\end{sqlcode}
\vspace{0.1cm}
\centering\colorbox{yellow!45}{\strut\footnotesize empleado 1: 6 citas}\quad\colorbox{utpGrisClaro}{\strut\footnotesize empleado 2: 4 citas}
\column{.42\textwidth}
\res{rrr}{\cab{id} & \cab{empleado\_id} & \cab{citas\_del\_empleado}}{%
    \fila 1 & 1 & 6 \\ \fila 3 & 1 & 6 \\ \fila 4 & 1 & 6 \\ \fila 6 & 1 & 6 \\ \fila 8 & 1 & 6 \\ \fila 10 & 1 & 6 \\
    \filagris 2 & 2 & 4 \\ \filagris 5 & 2 & 4 \\ \filagris 7 & 2 & 4 \\ \filagris 9 & 2 & 4 \\}
\end{columns}
\vspace{0.3cm}
\concepto{\texttt{PARTITION BY}}{\small Separa las filas en grupos, como \texttt{GROUP BY}, pero \textbf{sin juntarlas}. \texttt{COUNT} cuenta dentro de cada grupo y escribe el resultado en todas sus filas.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{ORDER BY} dentro de \texttt{OVER}: el total acumulado}
\renewcommand{\arraystretch}{0.9}
\small ¿Cuánto llevamos cobrado \textbf{hoy} (4 de noviembre) después de cada pago? La fecha está en \tbl{cita}.\par
\begin{sqlcode}[.66\textwidth]
\kw{SELECT} c.fecha\_hora, p.monto,\\
\ind\kw{SUM}(p.monto) \kw{OVER} (\dentro{\kw{ORDER BY} c.fecha\_hora}) \kw{AS} acumulado\\
\kw{FROM} pago p \kw{JOIN} cita c \kw{ON} c.id = p.cita\_id\\
\kw{WHERE} \kw{CAST}(c.fecha\_hora \kw{AS DATE}) = '2026-11-04'\\
\kw{ORDER BY} c.fecha\_hora;
\end{sqlcode}
\vspace{-0.2cm}
\begin{center}
\tabla{resultado}{lrr}{\cab{fecha\_hora} & \cab{monto} & \cab{acumulado}}{2026-11-04 09:00:00 & 15.00 & 15.00 \\ 2026-11-04 10:00:00 & 35.00 & 50.00 \\ 2026-11-04 11:00:00 & 55.00 & 105.00 \\}
\qquad
\tabla{cómo se calcula}{l}{\cab{lo anterior + este pago}}{15 \\ 15 + 35 \\ 50 + 55 \\}
\par\vspace{2pt}{\scriptsize\color{utpGris}SQL Server muestra la fecha como \texttt{2026-11-04 09:00:00.0000000}.}
\end{center}
\vspace{-0.15cm}
\concepto{Idea clave}{\small Con \texttt{ORDER BY}, la ventana de cada fila va \textbf{desde el primer pago del día hasta ella}. Al final del día, la caja tiene \textbf{105} soles.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Numerar: \texttt{ROW\_NUMBER}}
\renewcommand{\arraystretch}{0.74}
\preg{Ordena los pagos del más alto al más bajo y \textbf{numéralos}.}
\begin{sqlcode}[.8\textwidth]
\kw{SELECT} id, monto,\\
\ind\dentro{\kw{ROW\_NUMBER}()} \kw{OVER} (\kw{ORDER BY} monto \kw{DESC}, id) \kw{AS} puesto\\
\kw{FROM} pago \kw{ORDER BY} puesto;
\end{sqlcode}
\vspace{-0.15cm}
\begin{columns}[c,onlytextwidth]
\column{.4\textwidth}
\res{rrr}{\cab{id} & \cab{monto} & \cab{puesto}}{\fila 2 & 55.00 & 1 \\ \fila 4 & 55.00 & 2 \\ \fila 8 & 55.00 & 3 \\ 1 & 35.00 & 4 \\ 5 & 35.00 & 5 \\ 7 & 35.00 & 6 \\ 3 & 15.00 & 7 \\ 6 & 15.00 & 8 \\}
\column{.57\textwidth}
\raggedright\small Los pagos 2, 4 y 8 \textbf{empatan} en 55, pero cada uno recibe \textbf{su propio número}: 1, 2 y 3.\\[6pt] El \texttt{id} al final del \texttt{ORDER BY} decide quién va primero en un empate. Sin él, MySQL y SQL Server los numeran distinto.
\end{columns}
\vspace{0.05cm}
\concepto{\texttt{ROW\_NUMBER}}{\small Numera 1, 2, 3, 4, 5\ldots{} \textbf{sin repetir}, aunque haya empates.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Rankear con empates: \texttt{RANK}}
\renewcommand{\arraystretch}{0.74}
\preg{Del más alto al más bajo, ¿en qué \textbf{puesto} queda cada pago?}
\begin{sqlcode}[.8\textwidth]
\kw{SELECT} id, monto,\\
\ind\dentro{\kw{RANK}()} \kw{OVER} (\kw{ORDER BY} monto \kw{DESC}) \kw{AS} puesto\\
\kw{FROM} pago \kw{ORDER BY} puesto, id;
\end{sqlcode}
\vspace{-0.15cm}
\begin{columns}[c,onlytextwidth]
\column{.4\textwidth}
\res{rrr}{\cab{id} & \cab{monto} & \cab{puesto}}{2 & 55.00 & 1 \\ 4 & 55.00 & 1 \\ 8 & 55.00 & 1 \\ 1 & 35.00 & \mal{4} \\ 5 & 35.00 & \mal{4} \\ 7 & 35.00 & \mal{4} \\ 3 & 15.00 & 7 \\ 6 & 15.00 & 7 \\}
\column{.57\textwidth}
\raggedright\small Los tres de 55 comparten el puesto \textbf{1}.\\[6pt] El siguiente es el \textbf{4}, no el 2: hay tres pagos delante. Los puestos 2 y 3 \textbf{se saltan}.
\end{columns}
\vspace{0.05cm}
\concepto{\texttt{RANK}}{\small Los empatados \textbf{comparten puesto} y después \textbf{salta}: 1, 1, 1, 4, 4, 4, 7, 7.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Rankear sin saltos: \texttt{DENSE\_RANK}}
\renewcommand{\arraystretch}{0.74}
\preg{Del más alto al más bajo, ¿en qué \textbf{puesto} queda cada pago?}
\begin{sqlcode}[.8\textwidth]
\kw{SELECT} id, monto,\\
\ind\dentro{\kw{DENSE\_RANK}()} \kw{OVER} (\kw{ORDER BY} monto \kw{DESC}) \kw{AS} puesto\\
\kw{FROM} pago \kw{ORDER BY} puesto, id;
\end{sqlcode}
\vspace{-0.15cm}
\begin{columns}[c,onlytextwidth]
\column{.4\textwidth}
\res{rrr}{\cab{id} & \cab{monto} & \cab{puesto}}{2 & 55.00 & 1 \\ 4 & 55.00 & 1 \\ 8 & 55.00 & 1 \\ 1 & 35.00 & \mal{2} \\ 5 & 35.00 & \mal{2} \\ 7 & 35.00 & \mal{2} \\ 3 & 15.00 & \mal{3} \\ 6 & 15.00 & \mal{3} \\}
\column{.57\textwidth}
\raggedright\small Es la misma consulta, cambiando \texttt{RANK} por \texttt{DENSE\_RANK}.\\[6pt] Los tres de 55 siguen en el puesto \textbf{1}, pero los de 35 ahora quedan en el \textbf{2}, no en el 4, y los de 15 en el \textbf{3}: \textbf{no se salta} ningún número.
\end{columns}
\vspace{0.05cm}
\concepto{\texttt{DENSE\_RANK}}{\small Los empatados \textbf{comparten puesto} y el siguiente \textbf{no salta}: 1, 1, 1, 2, 2, 2, 3, 3.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{El pago anterior: \texttt{LAG}}
\renewcommand{\arraystretch}{0.74}
\preg{Junto a cada pago, ¿de cuánto fue el pago \textbf{anterior}?}
\begin{sqlcode}[.8\textwidth]
\kw{SELECT} id, monto,\\
\ind\dentro{\kw{LAG}(monto)} \kw{OVER} (\kw{ORDER BY} id) \kw{AS} anterior\\
\kw{FROM} pago \kw{ORDER BY} id;
\end{sqlcode}
\vspace{-0.15cm}
\begin{columns}[c,onlytextwidth]
\column{.4\textwidth}
\res{rrr}{\cab{id} & \cab{monto} & \cab{anterior}}{1 & \cellcolor{yellow!45}35.00 & \cellcolor{utpRojo!18}NULL \\ 2 & 55.00 & \cellcolor{yellow!45}35.00 \\ 3 & 15.00 & 55.00 \\ 4 & 55.00 & 15.00 \\ 5 & 35.00 & 55.00 \\ 6 & 15.00 & 35.00 \\ 7 & 35.00 & 15.00 \\ 8 & 55.00 & 35.00 \\}
\column{.57\textwidth}
\raggedright\small \texttt{LAG} trae el monto de la fila de \textbf{arriba}: el 35 del pago 1 aparece al lado del pago 2.\\[6pt] El pago 1 no tiene anterior: sale \texttt{NULL}.\\[6pt] Sirve para comparar: el pago 3 (15) fue menor que el 2 (55).
\end{columns}
\vspace{0.05cm}
\concepto{\texttt{LAG}}{\small Trae un valor de la fila \textbf{anterior}, según el orden que se pone en \texttt{OVER}.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{El pago siguiente: \texttt{LEAD}}
\renewcommand{\arraystretch}{0.74}
\preg{Junto a cada pago, ¿de cuánto fue el pago \textbf{siguiente}?}
\begin{sqlcode}[.8\textwidth]
\kw{SELECT} id, monto,\\
\ind\dentro{\kw{LEAD}(monto)} \kw{OVER} (\kw{ORDER BY} id) \kw{AS} siguiente\\
\kw{FROM} pago \kw{ORDER BY} id;
\end{sqlcode}
\vspace{-0.15cm}
\begin{columns}[c,onlytextwidth]
\column{.4\textwidth}
\res{rrr}{\cab{id} & \cab{monto} & \cab{siguiente}}{1 & 35.00 & \cellcolor{yellow!45}55.00 \\ 2 & \cellcolor{yellow!45}55.00 & 15.00 \\ 3 & 15.00 & 55.00 \\ 4 & 55.00 & 35.00 \\ 5 & 35.00 & 15.00 \\ 6 & 15.00 & 35.00 \\ 7 & 35.00 & 55.00 \\ 8 & 55.00 & \cellcolor{utpRojo!18}NULL \\}
\column{.57\textwidth}
\raggedright\small Es la misma consulta, cambiando \texttt{LAG} por \texttt{LEAD}.\\[6pt] \texttt{LEAD} trae el monto de la fila de \textbf{abajo}: el 55 del pago 2 aparece al lado del pago 1.\\[6pt] El pago 8 es el último: no tiene siguiente y sale \texttt{NULL}.
\end{columns}
\vspace{0.05cm}
\concepto{\texttt{LEAD}}{\small Trae un valor de la fila \textbf{siguiente}. Es lo contrario de \texttt{LAG}.}
\end{frame}
