\seccion{Usos comunes de funciones}
\begin{frame}{Funciones, uso 1: aplicar un porcentaje}
\small La tienda en línea necesita saber cuánto paga el cliente con un descuento en porcentaje.
\begin{columns}[c,onlytextwidth]
\column{.47\textwidth}
\begin{sqlchico}[\col]
\kw{CREATE FUNCTION} aplicar\_porcentaje(\\
\ind p\_precio \ \ \ \ \kw{DECIMAL}(10,2),\\
\ind p\_porcentaje \kw{DECIMAL}(5,2))\\
\kw{RETURNS DECIMAL}(10,2)\\
\kw{DETERMINISTIC}\\
\kw{RETURN} \dentro{p\_precio - p\_precio * p\_porcentaje / 100};
\end{sqlchico}
\column{.5\textwidth}
\begin{sqlmini}[\col]
\kw{SELECT} codigo, valor\_descuento,\\
\ind\dentro{aplicar\_porcentaje(120, valor\_descuento)}\\
\ind\ind\kw{AS} a\_pagar\\
\kw{FROM} cupon\\
\kw{WHERE} tipo\_descuento = 'porcentaje'\\
\kw{ORDER BY} id;
\end{sqlmini}
\res{lrr}{\cab{codigo} & \cab{valor\_descuento} & \cab{a\_pagar}}{BIENVENIDA10 & 10.00 & 108.00 \\ VERANO15 & 15.00 & 102.00 \\ NAVIDAD25 & 25.00 & 90.00 \\}
\end{columns}
\vspace{0.2cm}
\concepto{Se usa para}{\small Fórmulas que muchos necesitan: el precio con descuento, el total con IGV, la comisión de un vendedor. Si la fórmula cambia, se cambia \textbf{en un solo lugar}.}
\end{frame}

\begin{frame}{Funciones, uso 2: ¿el cupón está vigente?}
\small La app móvil solo debe mostrar los cupones que se pueden usar en una fecha.
\begin{columns}[c,onlytextwidth]
\column{.47\textwidth}
\begin{sqlmini}[\col]
\kw{CREATE FUNCTION} es\_vigente(\\
\ind p\_estado \kw{VARCHAR}(10),\\
\ind p\_desde \ \kw{DATETIME},\\
\ind p\_hasta \ \kw{DATETIME},\\
\ind p\_fecha \ \kw{DATETIME})\\
\kw{RETURNS BOOLEAN}\\
\kw{DETERMINISTIC}\\
\kw{RETURN} \dentro{p\_estado = 'activo'}\\
\ind\ind\dentro{\kw{AND} p\_desde <= p\_fecha}\\
\ind\ind\dentro{\kw{AND} (p\_hasta \kw{IS NULL OR} p\_fecha < p\_hasta)};
\end{sqlmini}
{\centering\scriptsize Activo, ya empezó y no ha terminado. La condición vale \textbf{1} (sí) o \textbf{0} (no).\par}
\column{.5\textwidth}
\begin{sqlmini}[\col]
\kw{SELECT} codigo,\\
\ind\dentro{es\_vigente(estado, vigente\_desde,}\\
\ind\ind\dentro{vigente\_hasta, '2026-11-10')} \kw{AS} vigente\\
\kw{FROM} cupon \kw{ORDER BY} id;
\end{sqlmini}
\res{lr}{\cab{codigo} & \cab{vigente}}{BIENVENIDA10 & 1 \\ NOVIEMBRE20 & 1 \\ VERANO15 & 0 \\ NAVIDAD25 & 0 \\ PRUEBA50 & 0 \\}
\end{columns}
\vspace{0.15cm}
\concepto{Se usa para}{\small Reglas que responden sí o no: ¿el cupón está vigente?, ¿el cliente es frecuente?, ¿hay stock? Sirve en el \texttt{SELECT} y en el \texttt{WHERE}.}
\end{frame}

\hspace*{6.2mm}\parbox[t]{.93\textwidth}{\footnotesize\gris{¿\textit{Error 1304: FUNCTION aplicar\_porcentaje already exists}? Ya la habías creado: el script empieza con \texttt{DROP FUNCTION IF EXISTS}. En SQL Server, \texttt{CREATE OR ALTER} la reemplaza.}}
\end{frame}

\begin{frame}{¿Qué hace \texttt{DELIMITER}?}
\small Workbench \textbf{corta} lo que escribes en cada \texttt{;}. Con \texttt{BEGIN}\ldots\texttt{END}, adentro hay \textbf{varios} \texttt{;}:
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{Sin DELIMITER}
\begin{sqlmicro}[\col]
\kw{CREATE FUNCTION} aplicar\_porcentaje(\ldots)\\
\kw{RETURNS DECIMAL}(10,2)\\
\kw{DETERMINISTIC}\\
\kw{BEGIN}\\
\ind\kw{DECLARE} v\_desc \kw{DECIMAL}(10,2)\mal{;}\ \com{← corta aquí}\\
\ind\kw{SET} v\_desc = p\_precio * p\_porcentaje / 100;\\
\ind\kw{RETURN} p\_precio - v\_desc;\\
\kw{END};
\end{sqlmicro}
\vspace{-2pt}
{\footnotesize Llega un \texttt{CREATE} \textbf{a medias}:}\\[1pt]
{\ttfamily\scriptsize\color{utpRojo}Error Code: 1064. You have an error in your SQL syntax\ldots}
\column{.48\textwidth}
\centering\etiqueta{Con DELIMITER}
\begin{sqlmicro}[\col]
\dentro{\kw{DELIMITER} \$\$}\ \com{← ahora se termina en \$\$}\\
\kw{CREATE FUNCTION} aplicar\_porcentaje(\ldots)\\
\kw{RETURNS DECIMAL}(10,2)\\
\kw{DETERMINISTIC}\\
\kw{BEGIN}\\
\ind\kw{DECLARE} v\_desc \kw{DECIMAL}(10,2);\\
\ind\kw{SET} v\_desc = p\_precio * p\_porcentaje / 100;\\
\ind\kw{RETURN} p\_precio - v\_desc;\\
\kw{END}\dentro{\$\$}\ \com{← fin del CREATE}\\
\dentro{\kw{DELIMITER} ;}\ \com{← vuelve el ;}
\end{sqlmicro}
\end{columns}
\vspace{0.2cm}
\concepto{DELIMITER}{\small Cambia el signo que marca el \textbf{fin de una instrucción}. No es SQL: es una orden para Workbench. Los usos 1 y 2 no lo necesitan: tienen \textbf{un solo} \texttt{;}. En SQL Server no existe: allí se usa \texttt{GO}.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{¿Qué significa \texttt{DETERMINISTIC}?}
\small \textbf{Determinista}: con los \textbf{mismos datos de entrada}, la función devuelve \textbf{siempre lo mismo}.
\vspace{0.2cm}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}>{\raggedright}p{6.6cm}>{\raggedright}p{2.9cm}>{\raggedright\arraybackslash}p{3.1cm}@{}}
\toprule
\bfseries Llamada & \bfseries Devuelve & \bfseries ¿Determinista? \\
\midrule
\texttt{aplicar\_porcentaje(120, 10)} & 108.00, siempre & \si\ \texttt{DETERMINISTIC} \\
\texttt{\scriptsize es\_vigente('activo', '2026-11-01',\newline\hspace*{1.5em}'2026-12-01', '2026-11-10')} & 1, siempre & \si\ \texttt{DETERMINISTIC} \\
una función igual, pero con \texttt{NOW()} en vez de la fecha que recibe & 1 hoy, 0 en diciembre & \no\ \texttt{NOT DETERMINISTIC} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.1cm}
\small Por eso \texttt{es\_vigente} recibe la fecha como parámetro: así, la misma pregunta siempre tiene la misma respuesta.
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Una función \texttt{NOT DETERMINISTIC}: \texttt{es\_vigente\_hoy}}
\small Igual que \texttt{es\_vigente}, pero sin recibir la fecha: usa \texttt{NOW()}, la fecha y hora de \textbf{este momento}.
\begin{columns}[c,onlytextwidth]
\column{.5\textwidth}
\begin{sqlmini}[\col]
\kw{CREATE FUNCTION} es\_vigente\_hoy(\\
\ind p\_estado \kw{VARCHAR}(10),\\
\ind p\_desde \ \kw{DATETIME},\\
\ind p\_hasta \ \kw{DATETIME})\\
\kw{RETURNS BOOLEAN}\\
\dentro{\kw{NOT DETERMINISTIC}}\\
\kw{NO SQL}\ \com{-- no lee ninguna tabla}\\
\kw{RETURN} p\_estado = 'activo'\\
\ind\ind\kw{AND} p\_desde <= \dentro{\kw{NOW}()}\\
\ind\ind\kw{AND} (p\_hasta \kw{IS NULL OR} \dentro{\kw{NOW}()} < p\_hasta);
\end{sqlmini}
\column{.46\textwidth}
\begin{sqlmicro}[\col]
\kw{SELECT} codigo,\\
\ind es\_vigente\_hoy(estado, vigente\_desde,\\
\ind\ind vigente\_hasta) \kw{AS} vigente\_hoy\\
\kw{FROM} cupon \kw{ORDER BY} id;
\end{sqlmicro}
\vspace{-6pt}
\res{lr}{\cab{codigo} & \cab{vigente\_hoy}}{BIENVENIDA10 & 1 \\ \fila NOVIEMBRE20 & 1 \\ VERANO15 & 0 \\ NAVIDAD25 & 0 \\ PRUEBA50 & 0 \\}
\par\vspace{3pt}
{\centering\normalfont\scriptsize\gris{Hoy, 18 de noviembre. El 2 de diciembre, \texttt{NOVIEMBRE20} da \textbf{0}.}\par}
\end{columns}
\vspace{0.1cm}
\concepto{NOT DETERMINISTIC}{\small Con los \textbf{mismos datos} puede devolver \textbf{otra cosa}, según cuándo se ejecute: \texttt{NOW()}, \texttt{RAND()}\ldots}
\end{frame}

\begin{frame}{\ins{Práctica 2} \texttt{aplicar\_porcentaje} con \texttt{BEGIN}\ldots\texttt{END}}
\npieza{1}\ \small Abre una pestaña nueva sobre la base \texttt{cupones}. \npieza{2}\ Copia y pega el código. \npieza{3}\ Ejecuta todo.
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\centering\etiqueta{MySQL Workbench: rayo \rayo}
\begin{sqlmicro}[\col]
\kw{USE} cupones;\\
\kw{DROP FUNCTION IF EXISTS} aplicar\_porcentaje;\\[2pt]
\dentro{\kw{DELIMITER} \$\$}\\
\kw{CREATE FUNCTION} aplicar\_porcentaje(\\
\ind p\_precio \ \ \ \ \kw{DECIMAL}(10,2),\\
\ind p\_porcentaje \kw{DECIMAL}(5,2))\\
\kw{RETURNS DECIMAL}(10,2)\\
\kw{DETERMINISTIC}\\
\kw{BEGIN}\\
\ind\kw{DECLARE} v\_desc \kw{DECIMAL}(10,2);\\
\ind\kw{SET} v\_desc = p\_precio * p\_porcentaje / 100;\\
\ind\kw{RETURN} p\_precio - v\_desc;\\
\kw{END}\dentro{\$\$}\\
\dentro{\kw{DELIMITER} ;}\\[2pt]
\kw{SELECT} aplicar\_porcentaje(120, 10) \kw{AS} a\_pagar;
\end{sqlmicro}
\column{.49\textwidth}
\centering\etiqueta{SQL Server (SSMS): Execute}
\begin{sqlmicro}[\col]
\kw{USE} cupones;\\
\dentro{\kw{GO}}\\[2pt]
\kw{CREATE OR ALTER FUNCTION} dbo.aplicar\_porcentaje(\\
\ind @precio \ \ \ \ \kw{DECIMAL}(10,2),\\
\ind @porcentaje \kw{DECIMAL}(5,2))\\
\kw{RETURNS DECIMAL}(10,2)\\
\kw{AS}\\
\kw{BEGIN}\\
\ind\kw{DECLARE} @desc \kw{DECIMAL}(10,2);\\
\ind\kw{SET} @desc = @precio * @porcentaje / 100;\\
\ind\kw{RETURN} @precio - @desc;\\
\kw{END};\\
\dentro{\kw{GO}}\\[2pt]
\kw{SELECT} dbo.aplicar\_porcentaje(120, 10) \kw{AS} a\_pagar;
\end{sqlmicro}
\end{columns}
\vspace{0.05cm}
\npieza{4}\ \small Comprueba que \texttt{a\_pagar} sea \textbf{108.00}.\quad\footnotesize\gris{Código en \texttt{ejemplos/cupones/2\_delimiter\_*.sql}. En SQL Server, \texttt{GO} hace de \texttt{DELIMITER}.}
\end{frame}

\begin{frame}{¿Cuándo conviene un procedimiento?}
\begin{columns}[T,onlytextwidth]
\column{.5\textwidth}
\centering\etiqueta{Úsalo cuando\ldots}\\[6pt]
\raggedright\small
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{@{}p{.06\textwidth}p{.84\textwidth}@{}}
\si & \textbf{cambias datos} con una regla\newline{\footnotesize\gris{\texttt{desactivar\_cupon}}} \\
\si & son \textbf{varios pasos} que van juntos\newline{\footnotesize\gris{\texttt{registrar\_venta}, con su transacción}} \\
\si & \textbf{varias pantallas o aplicaciones} piden lo mismo\newline{\footnotesize\gris{\texttt{cupones\_vigentes}}} \\
\si & es una \textbf{tarea que se repite}\newline{\footnotesize\gris{\texttt{cerrar\_cupones\_vencidos}, cada noche}} \\
\end{tabular}
\column{.48\textwidth}
\centering\etiqueta{No hace falta cuando\ldots}\\[6pt]
\raggedright\small
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{@{}p{.06\textwidth}p{.84\textwidth}@{}}
\no & solo \textbf{calculas un valor}\newline{\footnotesize\gris{mejor una función: \texttt{precio\_final}}} \\
\no & es un \texttt{SELECT} sencillo de \textbf{una sola pantalla}\newline{\footnotesize\gris{puede quedarse en la aplicación}} \\
\no & la base es \textbf{SQLite}\newline{\footnotesize\gris{no tiene procedimientos}} \\
\end{tabular}
\end{columns}
\vspace{0.4cm}
\concepto{La pregunta clave}{\small ¿Esto debe hacerse \textbf{siempre igual}, venga de la aplicación que venga? Entonces va en la base, como procedimiento.}
\end{frame}

\begin{frame}{La solución: \texttt{DELIMITER}}
\begin{columns}[c,onlytextwidth]
\column{.55\textwidth}
\begin{sqlcode}[\col]
\dentro{\kw{DELIMITER} \$\$}\hfill\npieza{1}\\[4pt]
\kw{CREATE PROCEDURE} desactivar\_cupon(\\
\ind p\_codigo \kw{VARCHAR}(255))\\
\dentro{\kw{BEGIN}}\hfill\npieza{2}\\
\ind\kw{UPDATE} cupon \kw{SET} estado = 'inactivo'\\
\ind\kw{WHERE} codigo = p\_codigo;\\
\ind\kw{SELECT} 'Cupón desactivado' \kw{AS} mensaje;\\
\dentro{\kw{END}\$\$}\hfill\npieza{3}\\[4pt]
\dentro{\kw{DELIMITER} ;}\hfill\npieza{4}
\end{sqlcode}
\column{.42\textwidth}
\small\raggedright
\npieza{1} Desde aquí, una instrucción termina en \texttt{\$\$}, no en \texttt{;}.\\[7pt]
\npieza{2} \texttt{BEGIN} y \texttt{END} agrupan las instrucciones del procedimiento. Sus \texttt{;} ya no cortan nada.\\[7pt]
\npieza{3} \texttt{\$\$} termina el \texttt{CREATE}.\\[7pt]
\npieza{4} Vuelve al \texttt{;} de siempre.
\end{columns}
\vspace{0.3cm}
{\centering\footnotesize\gris{Selecciona desde \texttt{DELIMITER \$\$} hasta \texttt{DELIMITER ;} y pulsa el rayo \rayo. Si ya existía, antes ejecuta \texttt{DROP PROCEDURE IF EXISTS}.}\par}
\end{frame}

\begin{frame}{Con variables e \texttt{IF}: avisar si el cupón no existe}
\small Con un código mal escrito (\texttt{'VERANO1'}), el \texttt{UPDATE} no cambia nada y \textbf{no avisa}.
\begin{columns}[T,onlytextwidth]
\column{.58\textwidth}
\begin{sqlmini}[\col]
\kw{DELIMITER} \$\$\\
\kw{CREATE PROCEDURE} desactivar\_cupon(p\_codigo \kw{VARCHAR}(255))\\
\kw{BEGIN}\\
\ind\dentro{\kw{DECLARE} v\_cantidad \kw{INT};}\hfill\npieza{1}\\[1.5pt]
\ind\kw{SELECT} \kw{COUNT}(*) \kw{INTO} v\_cantidad\\
\ind\kw{FROM} cupon \kw{WHERE} codigo = p\_codigo;\hfill\npieza{2}\\[1.5pt]
\ind\dentro{\kw{IF} v\_cantidad = 0 \kw{THEN}}\hfill\npieza{3}\\
\ind\ind\kw{SELECT} 'El cupón no existe' \kw{AS} mensaje;\\
\ind\kw{ELSE}\\
\ind\ind\kw{UPDATE} cupon \kw{SET} estado = 'inactivo'\\
\ind\ind\kw{WHERE} codigo = p\_codigo;\\
\ind\ind\kw{SELECT} 'Cupón desactivado' \kw{AS} mensaje;\\
\ind\kw{END IF};\\
\kw{END}\$\$\\
\kw{DELIMITER} ;
\end{sqlmini}
\column{.39\textwidth}
\small\raggedright
\vspace{0.3cm}
\npieza{1} Una variable para guardar un número.\\[7pt]
\npieza{2} Cuenta los cupones con ese código: \textbf{0} o \textbf{1}.\\[7pt]
\npieza{3} Si es 0, avisa. Si no, lo desactiva y lo dice.\\[10pt]
\footnotesize\gris{Lo mismo que en las funciones: \texttt{DECLARE}, \texttt{SELECT \ldots{} INTO} e \texttt{IF}.}
\end{columns}
\end{frame}

\begin{frame}{Un procedimiento que muestra un resultado}
\small La tienda pregunta: con una compra de S/ 120, ¿cuánto pagaría con cada cupón \textbf{vigente}?
\vspace{0.1cm}
\begin{columns}[c,onlytextwidth]
\column{.56\textwidth}
\begin{sqlmini}[\col]
\kw{CREATE PROCEDURE} cupones\_vigentes(\\
\ind p\_precio \kw{DECIMAL}(10,2), p\_fecha \kw{DATETIME})\\
\kw{SELECT} codigo,\\
\ind\dentro{precio\_final(p\_precio, id)} \kw{AS} a\_pagar\\
\kw{FROM} cupon\\
\kw{WHERE} \dentro{situacion\_cupon(id, p\_fecha)} = 'vigente'\\
\kw{ORDER BY} id;
\end{sqlmini}
\column{.4\textwidth}
\small\raggedright
\npieza{1} Recibe \textbf{dos} parámetros: el precio y la fecha.\\[7pt]
\npieza{2} Por dentro es un \texttt{SELECT} que usa las \textbf{funciones} de la sesión 28.\\[7pt]
\footnotesize\gris{Una sola instrucción, sin \texttt{BEGIN}: no necesita \texttt{DELIMITER}.}
\end{columns}
\end{frame}

\begin{frame}{Procedimientos, uso 4: una tarea que se repite cada día}
\small Cada noche hay que desactivar los cupones que ya vencieron. Un procedimiento lo hace con \textbf{un} \texttt{CALL}:
\vspace{0.1cm}
\begin{columns}[c,onlytextwidth]
\column{.56\textwidth}
\begin{sqlcode}[\col]
\kw{CREATE PROCEDURE} cerrar\_cupones\_vencidos(\\
\ind p\_fecha \kw{DATETIME})\\
\kw{UPDATE} cupon \kw{SET} estado = 'inactivo'\\
\kw{WHERE} estado = 'activo'\\
\ind\kw{AND} vigente\_hasta <= p\_fecha;\\[6pt]
\dentro{\kw{CALL} cerrar\_cupones\_vencidos('2026-11-10');}\\
\com{-- 1 row(s) affected}
\end{sqlcode}
\column{.4\textwidth}
\res{ll}{\cab{codigo} & \cab{estado}}{BIENVENIDA10 & activo \\ NOVIEMBRE20 & activo \\ \fila VERANO15 & inactivo \\ NAVIDAD25 & activo \\ PRUEBA50 & inactivo \\}
\par\vspace{4pt}
{\centering\normalfont\scriptsize\gris{\texttt{VERANO15} venció el 2026-04-01.}\par}
\end{columns}
\vspace{0.3cm}
\concepto{Se usa para}{\small Mantenimiento: cerrar lo vencido, borrar datos de prueba, preparar el reporte del mes. Se programa para que se ejecute solo, a la misma hora.}
\end{frame}

\begin{frame}{¿Qué puede hacer cada uno?}
\vspace{-0.2cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.35}
\newcommand{\Si}{\textcolor{green!45!black}{\textbf{sí}}}
\newcommand{\No}{\textcolor{utpRojo}{\textbf{no}}}
\begin{tabular}{@{}>{\raggedright}p{6.4cm}>{\centering}p{3cm}>{\centering\arraybackslash}p{3.4cm}@{}}
\toprule
 & \bfseries Función & \bfseries Procedimiento \\
\midrule
Recibir parámetros & \Si & \Si \\
Usar variables e \texttt{IF} & \Si & \Si \\
\filagris Ir \textbf{dentro} de un \texttt{SELECT} o un \texttt{WHERE} & \Si & \No \\
\filagris Mostrar filas con un \texttt{SELECT} & \No & \Si \\
\filagris Cambiar datos: \texttt{INSERT}, \texttt{UPDATE}, \texttt{DELETE} & \No\,* & \Si \\
\filagris Usar una transacción: \texttt{COMMIT}, \texttt{ROLLBACK} & \No & \Si \\
\midrule
Cómo se ejecuta & \texttt{\footnotesize SELECT f(\ldots);} & \texttt{\footnotesize CALL p(\ldots);}\newline{\scriptsize\gris{SQL Server: \texttt{EXEC p \ldots;}}} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{-2pt}
{\centering\scriptsize\gris{* SQL Server no lo permite. MySQL sí, pero no se hace: una función solo debe calcular.}\par}
\vspace{0.05cm}
\concepto{Regla práctica}{\small ¿Necesitas \textbf{un valor} para usarlo en una consulta? \textbf{Función}. ¿Necesitas que \textbf{pase algo} en la base? \textbf{Procedimiento}.}
\end{frame}
