\begin{frame}[c]
\begin{block}{Logro de la sesión}
Al finalizar, podrás mirar una tabla y decir \textbf{qué columna depende de cuál}; con eso podrás \textbf{separarla en varias tablas} sin perder información.
\end{block}
\end{frame}

% 5 ---------------------------------------------------------------------
\begin{frame}{El problema: la tienda guarda todo en una sola tabla}
\small Los mismos tres pedidos, con una fila por cada línea de venta:
\vspace{0.2cm}
\begin{center}
\begin{tablaventa}
\begin{tabular}{ccccccc}
\toprule
\bfseries pedido\_id & \bfseries cliente\_id & \bfseries cliente\_nombre & \bfseries producto\_id & \bfseries producto\_nombre & \bfseries precio & \bfseries cantidad \\
\midrule
1 & 40285571 & Raúl    & 1 & Audífonos & 89.00 & 2 \\
1 & 40285571 & Raúl    & 2 & Cargador  & 35.00 & 1 \\
2 & 16458744 & Mariana & 1 & Audífonos & 89.00 & 1 \\
2 & 16458744 & Mariana & 3 & Mouse     & 45.00 & 3 \\
3 & 36547781 & Elena   & 2 & Cargador  & 35.00 & 2 \\
\bottomrule
\end{tabular}
\end{tablaventa}
\end{center}
\vspace{0.4cm}
\begin{alertblock}{Mira la repetición}
\small El producto 1 aparece dos veces, y con él su nombre y su precio \textbf{89.00}. Si Audífonos sube a 99.00 y olvidamos corregir una fila, la tienda tendrá \textbf{dos precios} para el mismo producto.
\end{alertblock}
\end{frame}

% 6 ---------------------------------------------------------------------
\begin{frame}{La pregunta de hoy: ``si sé esto, ¿sé aquello?''}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Sí funciona}
\small Te digo \texttt{producto\_id} = \textbf{1}.\\
¿Cuál es su precio? \textbf{89.00}, siempre.
\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{No funciona}
\small Te digo \texttt{precio} = \textbf{89.00}.\\
¿Qué producto es? No se puede saber.
\end{alertblock}
\end{columns}
\vspace{0.45cm}
\begin{block}{La prueba}
Elige dos columnas y tapa el resto. Si al fijar la primera siempre queda \textbf{un único} valor en la segunda, la primera manda sobre la segunda.
\end{block}
\end{frame}

% 7 ---------------------------------------------------------------------
\begin{frame}{Eso se llama dependencia funcional}
\concepto{Dependencia funcional}{Al fijar el valor de una columna, queda determinado \textbf{un único} valor de otra.}
\vspace{0.3cm}
\begin{columns}[c,onlytextwidth]
\column{.52\textwidth}
\begin{center}
\begin{tikzpicture}[node distance=8mm]
  \node[cajaverde,text width=2.6cm] (a) {\texttt{producto\_id}};
  \node[cajanaranja,text width=2.2cm,right=11mm of a] (b) {\texttt{precio}};
  \draw[flecha] (a)--(b);
  \node[font=\scriptsize,below=1mm of a] {determinante};
  \node[font=\scriptsize,below=1mm of b] {dependiente};
\end{tikzpicture}
\end{center}
\column{.44\textwidth}
\small Se escribe \texttt{producto\_id} \fdep{} \texttt{precio}\\[5pt]
Se lee de dos maneras:
\begin{itemize}\setlength\itemsep{4pt}\small
  \item ``sabiendo el \texttt{producto\_id} puedo saber el \texttt{precio}'';
  \item ``\texttt{producto\_id} determina a \texttt{precio}''.
\end{itemize}
\end{columns}
\vspace{0.4cm}
\small \textbf{Este es el único ejemplo que hay que tener claro hoy.} Lo demás es aplicar la misma prueba a otras columnas.
\end{frame}

\midrule
1 & 40285571 & Raúl    & 1 & Audífonos & 89.00 & 2 \\
1 & 40285571 & Raúl    & 2 & Cargador  & 35.00 & 1 \\
2 & 16458744 & Mariana & 1 & Audífonos & 89.00 & 1 \\
\bottomrule
\end{tabular}
\end{tablaventa}
\end{center}
\vspace{0.35cm}
\begin{columns}[T,onlytextwidth]
\column{.54\textwidth}
\small \textbf{Escribe el nombre de la columna que falta:}
\begin{enumerate}\setlength\itemsep{7pt}\small
  \item \texttt{producto\_id} \fdep{} \bl
  \item \texttt{cliente\_id} \fdep{} \bl
  \item \bl{} \fdep{} \texttt{cliente\_id}
\end{enumerate}
\column{.42\textwidth}
\small \textbf{Responde sí o no:}
\vspace{4pt}

\small ¿Se cumple \texttt{producto\_id} \fdep{} \texttt{cantidad}?\\[5pt]
Respuesta: \bl[1.6cm]
\vspace{7pt}

\small ¿Por qué? \bl[3cm]
\end{columns}
\end{frame}

% 9 ---------------------------------------------------------------------
\begin{frame}{A veces el determinante son dos columnas juntas}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\begin{alertblock}{Con una sola columna falla}
\small
\texttt{pedido\_id} \fdep{} \texttt{cantidad}: en el pedido 1 hay 2 y 1.\\[4pt]
\texttt{producto\_id} \fdep{} \texttt{cantidad}: del producto 1 hay 2 y 1.
\end{alertblock}
\column{.49\textwidth}
\begin{exampleblock}{Con las dos juntas funciona}
\small
\{\texttt{pedido\_id}, \texttt{producto\_id}\} \fdep{} \texttt{cantidad}\\[4pt]
``producto 1 en el pedido 1'' da \textbf{2} y nada más.
\end{exampleblock}
\end{columns}
\vspace{0.45cm}
\analogia{Es como un asiento en el cine: la fila sola no ubica a nadie y el número solo tampoco. ``Fila G, asiento 12'' sí señala un único asiento.}
\end{frame}

\begin{frame}{Cada dependencia se convierte en una tabla}
\small Regla: \textbf{cada determinante se lleva consigo a sus dependientes} y se vuelve la clave primaria (\texttt{id}) de su tabla.
\vspace{0.25cm}
\begin{columns}[T,onlytextwidth]
\column{.28\textwidth}
\footnotesize
\texttt{\textbf{cliente}} \hfill {\scriptsize \texttt{cliente\_id}\fdep{}\texttt{nombre}}
\begin{tabular}{cc}
\toprule
\bfseries id & \bfseries nombre \\
\midrule
40285571 & Raúl \\
16458744 & Mariana \\
36547781 & Elena \\
\bottomrule
\end{tabular}
\vspace{8pt}

\texttt{\textbf{pedido}} \hfill {\scriptsize \texttt{pedido\_id}\fdep{}\texttt{cliente\_id}}
\begin{tabular}{cc}
\toprule
\bfseries id & \bfseries cliente\_id \\
\midrule
1 & 40285571 \\
2 & 16458744 \\
3 & 36547781 \\
\bottomrule
\end{tabular}
\column{.30\textwidth}
\footnotesize
\texttt{\textbf{producto}} \hfill {\scriptsize \texttt{producto\_id}\fdep{}\texttt{precio}}
\begin{tabular}{ccc}
\toprule
\bfseries id & \bfseries nombre & \bfseries precio \\
\midrule
1 & Audífonos & 89.00 \\
2 & Cargador & 35.00 \\
3 & Mouse & 45.00 \\
\bottomrule
\end{tabular}
\column{.38\textwidth}
\footnotesize
\texttt{\textbf{detalle\_pedido}}
\begin{tabular}{ccc}
\toprule
\bfseries pedido\_id & \bfseries producto\_id & \bfseries cantidad \\
\midrule
1 & 1 & 2 \\
1 & 2 & 1 \\
2 & 1 & 1 \\
2 & 3 & 3 \\
3 & 2 & 2 \\
\bottomrule
\end{tabular}
{\scriptsize \{\texttt{pedido\_id},\texttt{producto\_id}\}\fdep{}\texttt{cantidad}}
\end{columns}
\vspace{0.25cm}
\small El precio \textbf{89.00} queda escrito una sola vez. Dentro de su tabla el determinante se llama \texttt{id}; cuando aparece en otra lleva el nombre de su tabla más \texttt{\_id} y es una \textbf{clave foránea}, como \texttt{cliente\_id} o \texttt{producto\_id}.
\end{frame}

% 13 ---------------------------------------------------------------------
\begin{frame}{Para saber más: tipos y práctica real\apoyo}
\begin{center}
\footnotesize
\begin{tabular}{>{\bfseries}p{2.1cm}>{\raggedright\arraybackslash}p{3.5cm}>{\raggedright\arraybackslash}p{5.6cm}}
\toprule
Tipo & En qué consiste & En la tabla de la tienda \\
\midrule
Plena & Se necesita el determinante completo & \{\texttt{pedido\_id}, \texttt{producto\_id}\} \fdep{} \texttt{cantidad} \\
Transitiva & Se llega pasando por una tercera columna & \texttt{pedido\_id} \fdep{} \texttt{cliente\_id} \fdep{} \texttt{cliente\_nombre} \\
Trivial & Lo pedido ya venía incluido en lo dado & \{\texttt{pedido\_id}, \texttt{producto\_id}\} \fdep{} \texttt{pedido\_id} \\
Multivaluada & Un valor lleva a una lista independiente de otra & un cliente guarda varias direcciones y varias tarjetas, sin relación entre sí \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.25cm}
\begin{block}{¿Esto se usa en la práctica?}
\small Nadie dice ``esta dependencia es transitiva''; lo que sí se hace es \textbf{preguntar si el dato depende de toda la clave}.
\end{block}
\end{frame}
