\begin{frame}{Revisión: ¿qué puede hacer cada uno?}
\vspace{-0.2cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.35}
\newcommand{\Si}{\textcolor{green!45!black}{\textbf{sí}}}
\newcommand{\No}{\textcolor{utpRojo}{\textbf{no}}}
\begin{tabular}{@{}>{\raggedright}p{6.4cm}>{\centering}p{3cm}>{\centering\arraybackslash}p{3.4cm}@{}}
\toprule
 & \bfseries Función & \bfseries Procedimiento \\
\midrule
Recibir parámetros & \Si & \Si \\
Usar variables e \texttt{IF} & \Si & \Si \\
\filagris Ir \textbf{dentro} de un \texttt{SELECT} o un \texttt{WHERE} & \Si & \No \\
\filagris Mostrar filas con un \texttt{SELECT} & \No & \Si \\
\filagris Cambiar datos: \texttt{INSERT}, \texttt{UPDATE}, \texttt{DELETE} & \No\,* & \Si \\
\filagris Usar una transacción: \texttt{COMMIT}, \texttt{ROLLBACK} & \No & \Si \\
\midrule
Cómo se ejecuta & \texttt{\footnotesize SELECT f(\ldots);} & \texttt{\footnotesize CALL p(\ldots);}\newline{\scriptsize\gris{SQL Server: \texttt{EXEC p \ldots;}}} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{-2pt}
{\centering\scriptsize\gris{* SQL Server no lo permite. MySQL sí, pero no se hace: una función solo debe calcular.}\par}

\concepto{Regla práctica}{\small ¿Necesitas \textbf{un valor} para usarlo en una consulta? \textbf{Función}. ¿Necesitas que \textbf{pase algo} en la base? \textbf{Procedimiento}.}
\end{frame}

\begin{frame}{MySQL: recibir un dato, \texttt{IN}}
\small El administrador quiere dejar \textbf{300} entradas a la venta.
\vspace{0.2cm}
\begin{columns}[c,onlytextwidth]
\column{.55\textwidth}
\begin{sqlchico}[\col]
\kw{CREATE PROCEDURE} cambiar\_disponibles(\\
\ind\dentro{\kw{IN} p\_disponibles \kw{INT}})\\
\kw{UPDATE} entrada\\
\kw{SET} disponibles = \dentro{p\_disponibles}\\
\kw{WHERE} id = 1;\\[6pt]
\kw{CALL} cambiar\_disponibles(\dentro{300});\\
\kw{CALL} ver\_entrada();
\end{sqlchico}
\resg{lrr}{\cab{nombre} & \cab{precio} & \cab{disponibles}}{\fila Entrada Fogón 2026 & 45.00 & 300 \\}
\column{.42\textwidth}
\small\raggedright
\npieza{1} \texttt{IN}: el dato \textbf{entra}.\\[6pt]
\npieza{2} \texttt{p\_disponibles}: su nombre.\\[6pt]
\npieza{3} \texttt{INT}: su tipo.\\[6pt]
\npieza{4} En el \texttt{CALL} va el valor: \texttt{300}.\\[10pt]
\footnotesize\gris{Con varios datos, se separan con comas.}
\end{columns}
\end{frame}

\begin{frame}{MySQL: varias instrucciones, \texttt{BEGIN}\ldots\texttt{END}}
\small Cerrar la venta y mostrar cómo quedó: \textbf{dos} instrucciones.
\vspace{0.2cm}
\begin{columns}[c,onlytextwidth]
\column{.55\textwidth}
\begin{sqlchico}[\col]
\dentro{\kw{DELIMITER} //}\hfill\npieza{1}\\[2pt]
\kw{CREATE PROCEDURE} cerrar\_venta()\\
\dentro{\kw{BEGIN}}\hfill\npieza{2}\\
\ind\kw{UPDATE} entrada \kw{SET} venta\_activa = 0\\
\ind\kw{WHERE} id = 1;\\
\ind\kw{SELECT} nombre, venta\_activa \kw{FROM} entrada;\\
\dentro{\kw{END} //}\hfill\npieza{3}\\[2pt]
\dentro{\kw{DELIMITER} ;}\hfill\npieza{4}\\[6pt]
\kw{CALL} cerrar\_venta();
\end{sqlchico}
\resg{lr}{\cab{nombre} & \cab{venta\_activa}}{Entrada Fogón 2026 & 0 \\}
\column{.42\textwidth}
\small\raggedright
\npieza{1} El fin ahora es \texttt{//}.\\[6pt]
\npieza{2} \texttt{BEGIN}: aquí empieza.\\[6pt]
\npieza{3} \texttt{END //}: aquí termina.\\[6pt]
\npieza{4} El fin vuelve a ser \texttt{;}.\\[10pt]
\footnotesize\gris{Sin \texttt{DELIMITER}, el primer \texttt{;} cortaría el procedimiento a la mitad.}
\end{columns}
\end{frame}

\begin{frame}{MySQL: variables}
\small ¿Cuánto se recaudaría si se vendieran todas las entradas?
\vspace{0.2cm}
\begin{columns}[c,onlytextwidth]
\column{.56\textwidth}
\begin{sqlchico}[\col]
\kw{DELIMITER} //\\
\kw{CREATE PROCEDURE} recaudacion\_posible()\\
\kw{BEGIN}\\
\ind\dentro{\kw{DECLARE} v\_precio \kw{DECIMAL}(10, 2);}\hfill\npieza{1}\\
\ind\dentro{\kw{DECLARE} v\_disponibles \kw{INT};}\\
\ind\dentro{\kw{DECLARE} v\_total \kw{DECIMAL}(12, 2);}\\[2pt]
\ind\kw{SELECT} precio, disponibles\hfill\npieza{2}\\
\ind\dentro{\kw{INTO} v\_precio, v\_disponibles}\\
\ind\kw{FROM} entrada \kw{WHERE} id = 1;\\[2pt]
\ind\dentro{\kw{SET} v\_total = v\_precio * v\_disponibles;}\hfill\npieza{3}\\
\ind\kw{SELECT} v\_total \kw{AS} recaudacion\_posible;\\
\kw{END} //\\
\kw{DELIMITER} ;
\end{sqlchico}
\column{.41\textwidth}
\small\raggedright
\npieza{1} \texttt{DECLARE}: crea una variable.\\[6pt]
\npieza{2} \texttt{INTO}: guarda lo leído en las variables.\\[6pt]
\npieza{3} \texttt{SET}: guarda un cálculo.\\[10pt]
\begin{sqlcode}[\col]
\kw{CALL} recaudacion\_posible();
\end{sqlcode}
\resg{r}{\cab{recaudacion\_posible}}{22500.00 \\}
\end{columns}
\end{frame}

\begin{frame}{MySQL: decidir con \texttt{IF}}
\small Mostrar un aviso según cuántas entradas quedan.
\vspace{0.2cm}
\begin{columns}[c,onlytextwidth]
\column{.56\textwidth}
\begin{sqlchico}[\col]
\kw{DELIMITER} //\\
\kw{CREATE PROCEDURE} estado\_venta()\\
\kw{BEGIN}\\
\ind\kw{DECLARE} v\_disponibles \kw{INT};\\
\ind\kw{SELECT} disponibles \kw{INTO} v\_disponibles\\
\ind\kw{FROM} entrada \kw{WHERE} id = 1;\\[2pt]
\ind\dentro{\kw{IF}} v\_disponibles = 0 \dentro{\kw{THEN}}\\
\ind\ind\kw{SELECT} 'Agotado' \kw{AS} estado;\\
\ind\dentro{\kw{ELSEIF}} v\_disponibles < 50 \dentro{\kw{THEN}}\\
\ind\ind\kw{SELECT} 'Últimas entradas' \kw{AS} estado;\\
\ind\dentro{\kw{ELSE}}\\
\ind\ind\kw{SELECT} 'Disponible' \kw{AS} estado;\\
\ind\dentro{\kw{END IF};}\\
\kw{END} //\\
\kw{DELIMITER} ;
\end{sqlchico}
\column{.41\textwidth}
\small\raggedright
Se ejecuta \textbf{un solo} camino: el primero que se cumple.\\[10pt]
\renewcommand{\arraystretch}{1.2}
\footnotesize
\begin{tabular}{@{}rl@{}}
\toprule
\bfseries disponibles & \bfseries \texttt{CALL estado\_venta();} \\
\midrule
500 & \texttt{Disponible} \\
20 & \texttt{Últimas entradas} \\
0 & \texttt{Agotado} \\
\bottomrule
\end{tabular}
\end{columns}
\end{frame}

\begin{frame}{MySQL: repetir con \texttt{WHILE}}
\small Cargar ventas de prueba: el mismo \texttt{INSERT}, varias veces.
\vspace{0.15cm}
\begin{columns}[c,onlytextwidth]
\column{.58\textwidth}
\begin{sqlmini}[\col]
\kw{DELIMITER} //\\
\kw{CREATE PROCEDURE} cargar\_ventas\_prueba(\kw{IN} p\_cantidad \kw{INT})\\
\kw{BEGIN}\\
\ind\dentro{\kw{DECLARE} v\_i \kw{INT DEFAULT} 1;}\hfill\npieza{1}\\
\ind\dentro{\kw{WHILE} v\_i <= p\_cantidad \kw{DO}}\hfill\npieza{2}\\
\ind\ind\kw{INSERT INTO} venta (dni, nombre, correo, precio,\\
\ind\ind\ind referencia\_pago)\\
\ind\ind\kw{VALUES} (\kw{CONCAT}('7000000', v\_i), \kw{CONCAT}('Prueba ', v\_i),\\
\ind\ind\ind \kw{CONCAT}('prueba', v\_i, '@example.com'), 45.00,\\
\ind\ind\ind \kw{CONCAT}('PRUEBA-', v\_i));\\
\ind\ind\dentro{\kw{SET} v\_i = v\_i + 1;}\hfill\npieza{3}\\
\ind\dentro{\kw{END WHILE};}\hfill\npieza{4}\\
\kw{END} //\\
\kw{DELIMITER} ;
\end{sqlmini}
\column{.39\textwidth}
\small\raggedright
\npieza{1} \texttt{DEFAULT 1}: empieza en 1.\\[5pt]
\npieza{2} Repite \textbf{mientras} se cumpla.\\[5pt]
\npieza{3} Suma 1 en cada vuelta.\\[5pt]
\npieza{4} Aquí termina el bucle.\\[8pt]
{\footnotesize\texttt{CALL cargar\_ventas\_prueba(3);}\\\texttt{SELECT dni, nombre, referencia\_pago}\\\texttt{FROM venta;}}\par
\vspace{3pt}
{\scriptsize\resg{lll}{\cab{dni} & \cab{nombre} & \cab{referencia\_pago}}{70000001 & Prueba 1 & PRUEBA-1 \\ 70000002 & Prueba 2 & PRUEBA-2 \\ 70000003 & Prueba 3 & PRUEBA-3 \\}}
\end{columns}
\vspace{0.1cm}
{\centering\footnotesize\gris{Sin el \texttt{SET} de \npieza{3}, \texttt{v\_i} no cambia y el bucle no termina nunca.}\par}
\end{frame}

\begin{frame}{MySQL: otras formas de repetir, \texttt{REPEAT} y \texttt{LOOP}}
\small Las mismas ventas de prueba, con las otras dos formas:
\vspace{0.2cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{REPEAT}
\begin{sqlchico}[\col]
\kw{DECLARE} v\_i \kw{INT DEFAULT} 1;\\
\dentro{\kw{REPEAT}}\\
\ind\kw{INSERT INTO} venta \ldots;\\
\ind\kw{SET} v\_i = v\_i + 1;\\
\dentro{\kw{UNTIL} v\_i > p\_cantidad}\\
\dentro{\kw{END REPEAT};}
\end{sqlchico}
\raggedright\small Primero hace una vuelta, después pregunta. Repite \textbf{hasta} (\texttt{UNTIL}) que se cumpla.\\[4pt]
\footnotesize\gris{Siempre da al menos una vuelta.}
\column{.48\textwidth}
\centering\etiqueta{LOOP}
\begin{sqlchico}[\col]
\kw{DECLARE} v\_i \kw{INT DEFAULT} 1;\\
\dentro{repetir: \kw{LOOP}}\\
\ind\kw{IF} v\_i > p\_cantidad \kw{THEN}\\
\ind\ind\dentro{\kw{LEAVE} repetir;}\\
\ind\kw{END IF};\\
\ind\kw{INSERT INTO} venta \ldots;\\
\ind\kw{SET} v\_i = v\_i + 1;\\
\dentro{\kw{END LOOP};}
\end{sqlchico}
\raggedright\small Repite sin condición. \texttt{repetir} es el nombre del bucle; \texttt{LEAVE repetir} sale de él.\\[4pt]
\footnotesize\gris{\texttt{ITERATE repetir} salta a la siguiente vuelta.}
\end{columns}
\vspace{0.25cm}
{\centering\small Las tres formas cargan las mismas ventas. \texttt{WHILE} es la más usada.\par}
\end{frame}

\begin{frame}{MySQL: detener con un error, \texttt{SIGNAL}}

\vspace{0.2cm}
\begin{columns}[c,onlytextwidth]
\column{.58\textwidth}
\begin{sqlchico}[\col]
\kw{DELIMITER} //\\
\kw{CREATE PROCEDURE} fijar\_disponibles(\kw{IN} p\_disponibles \kw{INT})\\
\kw{BEGIN}\\
\ind\kw{IF} p\_disponibles < 0 \kw{THEN}\\
\ind\ind\dentro{\kw{SIGNAL SQLSTATE} '45000'}\\
\ind\ind\ind\dentro{\kw{SET MESSAGE\_TEXT} = 'Las entradas}\\
\ind\ind\ind\dentro{disponibles no pueden ser negativas.';}\\
\ind\kw{END IF};\\
\ind\kw{UPDATE} entrada \kw{SET} disponibles = p\_disponibles\\
\ind\kw{WHERE} id = 1;\\
\kw{END} //\\
\kw{DELIMITER} ;
\end{sqlchico}
\column{.39\textwidth}
\small\raggedright
\texttt{SIGNAL} detiene el procedimiento y muestra el mensaje. El \texttt{UPDATE} \textbf{no} se ejecuta.\\[10pt]
\footnotesize\texttt{CALL fijar\_disponibles(-5);}\\[2pt]
\errlinea{Error Code: 1644. Las entradas disponibles no pueden ser negativas.}\\[10pt]
\gris{\texttt{'45000'} es el código para un error propio.}
\end{columns}
\end{frame}

\begin{columns}[c,onlytextwidth]
\column{.58\textwidth}

\kw{CREATE OR ALTER PROCEDURE} dbo.cargar\_ventas\_prueba\\
\ind @cantidad \kw{INT}\\
\kw{AS}\\
\kw{BEGIN}\\
\ind\dentro{\kw{DECLARE} @i \kw{INT} = 1;}\hfill\npieza{1}\\
\ind\dentro{\kw{WHILE} @i <= @cantidad}\hfill\npieza{2}\\
\ind\dentro{\kw{BEGIN}}\\

\column{.58\textwidth}
\begin{sqlchico}[\col]

\ind @disponibles \kw{INT}\\
\kw{AS}\\

\begin{frame}{SQL Server: una transacción, \texttt{TRY}\ldots\texttt{CATCH}}
\small Si un paso falla, se deshace \textbf{todo}.
\vspace{0.1cm}
\begin{columns}[c,onlytextwidth]
\column{.5\textwidth}
\begin{sqlmini}[\col]
\kw{AS}\\
\kw{BEGIN}\\
\ind\dentro{\kw{BEGIN TRY}}\hfill\npieza{1}\\
\ind\ind\kw{BEGIN TRANSACTION};\\
\ind\ind\com{-- UPDATE, INSERT\ldots}\\
\ind\ind\kw{COMMIT};\\
\ind\dentro{\kw{END TRY}}\\
\ind\dentro{\kw{BEGIN CATCH}}\hfill\npieza{2}\\
\ind\ind\kw{IF} @@TRANCOUNT > 0\\
\ind\ind\ind\kw{ROLLBACK};\\
\ind\ind\kw{THROW};\\
\ind\dentro{\kw{END CATCH};}\\
\kw{END};
\end{sqlmini}
\column{.47\textwidth}
\small\raggedright
\npieza{1} \texttt{TRY}: los pasos. Si nada falla, llega al \texttt{COMMIT}.\\[8pt]
\npieza{2} \texttt{CATCH}: si algo falla, salta aquí. \texttt{ROLLBACK} deshace y \texttt{THROW} muestra el error.\\[10pt]
\footnotesize\gris{\texttt{@@TRANCOUNT > 0}: hay una transacción abierta.}
\end{columns}
\end{frame}

\seccion{Triggers}
\begin{frame}{Introducción: ¿qué es un trigger?}
\small Un \textbf{trigger} es un bloque de SQL guardado en la base que se ejecuta \textbf{solo}, cuando pasa algo en una tabla: un \texttt{INSERT}, un \texttt{UPDATE} o un \texttt{DELETE}. Nadie lo llama con \texttt{CALL} ni con \texttt{EXEC}.

\concepto{Regla}{\small La \textbf{fecha de registro} no puede ser \textbf{futura}. En el modelo es un \texttt{CHECK} que compara con la fecha de hoy.}
\end{frame}

\begin{frame}{Agenda: en MySQL ese \texttt{CHECK} no se puede crear}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlmini}[\col]
\kw{CREATE TABLE} contacto (\\
\ind id \kw{INT AUTO\_INCREMENT PRIMARY KEY},\\
\ind nombre \kw{VARCHAR}(100) \kw{NOT NULL},\\
\ind telefono \kw{VARCHAR}(20),\\
\ind correo \kw{VARCHAR}(255) \kw{UNIQUE},\\
\ind es\_favorito \kw{BOOLEAN NOT NULL DEFAULT FALSE},\\
\ind fecha\_registro \kw{DATE NOT NULL}\\
\ind\ind\kw{DEFAULT} (\kw{CURRENT\_DATE}),\\
\ind\mal{\kw{CHECK} (fecha\_registro <= \kw{CURRENT\_DATE})}\\
);
\end{sqlmini}
\vspace{-4pt}
\raggedright\errlinea{Error Code: 3814. An expression of a check constraint 'contacto\_chk\_1' contains disallowed function: curdate.}
\column{.48\textwidth}
\centering\etiqueta{SQL Server}
\begin{sqlmini}[\col]
\kw{CREATE TABLE} contacto (\\
\ind id \kw{INT IDENTITY}(1,1) \kw{PRIMARY KEY},\\
\ind nombre \kw{NVARCHAR}(100) \kw{NOT NULL},\\
\ind telefono \kw{VARCHAR}(20),\\
\ind correo \kw{VARCHAR}(255) \kw{UNIQUE},\\
\ind es\_favorito \kw{BIT NOT NULL DEFAULT} 0,\\
\ind fecha\_registro \kw{DATE NOT NULL}\\
\ind\ind\kw{DEFAULT CAST}(\kw{GETDATE}() \kw{AS DATE}),\\
\ind\dentro{\kw{CHECK} (fecha\_registro <= \kw{CAST}(\kw{GETDATE}() \kw{AS DATE}))}\\
);
\end{sqlmini}
\vspace{-4pt}
\raggedright{\footnotesize Se crea. Con una fecha futura:}\\
\errlinea{Msg 547 \ldots{} The INSERT statement conflicted with the CHECK constraint \ldots}
\end{columns}
\vspace{0.1cm}
\concepto{¿Por qué?}{\small En MySQL un \texttt{CHECK} no puede usar funciones cuyo resultado cambia, como \texttt{CURRENT\_DATE}. SQL Server sí lo permite. \textbf{En MySQL, esta regla se hace con un trigger.}}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Agenda: la regla con un trigger en MySQL}
\begin{columns}[T,onlytextwidth]
\column{.52\textwidth}
\begin{sqlmini}[\col]
\kw{DELIMITER} //\\
\kw{CREATE TRIGGER} contacto\_fecha\_insert\\
\dentro{\kw{BEFORE INSERT ON} contacto}\\
\kw{FOR EACH ROW}\\
\kw{BEGIN}\\
\ind\kw{IF} \dentro{NEW}.fecha\_registro > \kw{CURRENT\_DATE} \kw{THEN}\\
\ind\ind\kw{SIGNAL SQLSTATE} '45000'\\
\ind\ind\ind\kw{SET MESSAGE\_TEXT} =\\
\ind\ind\ind'La fecha de registro no puede ser futura.';\\
\ind\kw{END IF};\\
\kw{END} //\\
\kw{DELIMITER} ;
\end{sqlmini}
\column{.45\textwidth}
\vspace{4pt}
\footnotesize
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}>{\ttfamily\bfseries}l>{\raggedright\arraybackslash}p{3.65cm}@{}}
BEFORE INSERT & Se ejecuta antes de guardar cada fila nueva. \\
ON contacto & La tabla a la que queda unido. \\
FOR EACH ROW & Una vez por cada fila. \\
NEW & La fila que se va a guardar. \\
SIGNAL & Detiene el \texttt{INSERT}: la fila no se guarda. \\
\end{tabular}
\end{columns}
\begin{sqlmini}
\kw{INSERT INTO} contacto (nombre, correo, fecha\_registro)\\
\kw{VALUES} ('Carlos Díaz', 'carlos.diaz@example.com', \mal{'2027-01-15'});
\end{sqlmini}
\vspace{-4pt}
{\centering\errlinea{Error Code: 1644. La fecha de registro no puede ser futura.}\par}
{\centering\footnotesize Un trigger atiende un solo evento: para \texttt{UPDATE}, otro igual con \texttt{BEFORE UPDATE}.\par}
\end{frame}

\begin{tabular}{>{\raggedright\bfseries}p{3.6cm}>{\raggedright\arraybackslash}p{9.4cm}}

Parámetros & Entran con \texttt{IN} (MySQL) o solo \texttt{@nombre} (SQL Server). Salen con \texttt{OUT} u \texttt{OUTPUT}. \\
