\begin{frame}{Paso 4: ¿quién es cada fila?}
\begin{columns}[c,onlytextwidth]
\column{.58\textwidth}
\begin{sqlcode}[\textwidth]
\kw{CREATE TABLE} cliente\\
(\\
\ind \nuevo{id \ \ \ \ \ INT \ \ \ \ \ \ \ \ \kw{PRIMARY KEY},}\\
\ind nombre \ VARCHAR(80),\\
\ind correo \ VARCHAR(120)\\
);
\end{sqlcode}
\column{.38\textwidth}
\begin{center}
\begin{tikzpicture}
  \node[cajaverde,text width=2.7cm,font=\scriptsize,align=left]
    {\textbf{cliente}\\[3pt] \texttt{id} \ \pk\\ \texttt{nombre}\\ \texttt{correo}};
\end{tikzpicture}
\end{center}
\vspace{5pt}
\centering\scriptsize la \pk{} del dibujo\\ es esta línea
\end{columns}
\vspace{0.5cm}
\small Sin el \texttt{id} hay un problema: si dos clientes se llaman igual, \textbf{no se pueden distinguir}. La \textbf{clave primaria} es la columna que identifica a una sola fila; no se repite y nunca queda vacía.
\vspace{0.2cm}

\small Y con eso la instrucción está completa: \textbf{eso es todo un \texttt{CREATE TABLE}}.
\end{frame}

\begin{frame}{Las restricciones: reglas que toda fila cumple}
\begin{center}
\small
\begin{tabular}{>{\ttfamily}p{2.9cm}>{\raggedright\arraybackslash}p{4.2cm}>{\raggedright\arraybackslash}p{4.6cm}}
\toprule
\normalfont\bfseries Restricción & \bfseries Qué obliga & \bfseries En la tienda \\
\midrule
NOT NULL & No puede quedar vacía & Un cliente sin nombre no sirve \\[5pt]
UNIQUE & No se repite en dos filas & Dos clientes no comparten correo \\[5pt]
AUTO\_INCREMENT & El gestor numera solo & \texttt{id} 1, 2, 3\ldots sin pensarlo \\[5pt]
DEFAULT \textit{valor} & Si no se indica, usa ese & \texttt{activo BOOLEAN DEFAULT TRUE} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.4cm}
\small Se escriben \textbf{después del tipo}, en la misma línea. Son opcionales: una columna con nombre y tipo ya funciona.
\vspace{0.3cm}

\small \texttt{PRIMARY KEY} ya incluye \texttt{NOT NULL} y \texttt{UNIQUE}; por eso casi siempre se escribe junto con \texttt{AUTO\_INCREMENT} y nada más.
\end{frame}

\begin{frame}{\texttt{FOREIGN KEY}: la conexión, por fin escrita}
\begin{center}
\begin{tikzpicture}

  \node[cajanaranja,text width=2.1cm,font=\tiny,align=center,right=22mm of cli] (ped) {\textbf{pedido}\\ \texttt{cliente\_id} \fk};
  \draw[relacion] (cli)--(ped);
  \pic at ($(cli.east)!2.6mm!(ped.west)$) {uno};
  \pic at ($(ped.west)!2.6mm!(cli.east)$) {muchos};
\end{tikzpicture}
\end{center}
\vspace{0.2cm}
\begin{sqlcode}[.72\textwidth]
\begin{sqltab}
\sqlline{\kw{CREATE TABLE} pedido (}\\
\ind id          & INT  & \kw{PRIMARY KEY} \kw{AUTO\_INCREMENT},\\
\ind fecha       & DATE & \kw{NOT NULL},\\
\ind cliente\_id  & INT  & \kw{NOT NULL},\\
\sqlline{\ind \nuevo{\kw{FOREIGN KEY} (cliente\_id) \kw{REFERENCES} cliente(id)}}\\
\sqlline{);}\\
\end{sqltab}
\end{sqlcode}
\vspace{0.15cm}
\small \textbf{Son dos líneas, no una:} arriba se declara \texttt{cliente\_id} como cualquier columna; abajo se dice \textbf{a dónde apunta}. Esa segunda línea conecta cada pedido con un cliente existente.
\end{frame}

);
\end{sqlcode}
\column{.49\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{CREATE TABLE} pedido (\\
\ind id \nuevo{INTEGER} \kw{PRIMARY KEY} \nuevo{\kw{AUTOINCREMENT},}\\
\ind fecha \nuevo{TEXT} \kw{NOT NULL},\\
\ind cliente\_id INTEGER \kw{NOT NULL},\\
\ind \kw{FOREIGN KEY} (cliente\_id)\\
\ind\ind \kw{REFERENCES} cliente(id)\\
);\\[7pt]
\kw{CREATE TABLE} producto (\\
\ind id \nuevo{INTEGER} \kw{PRIMARY KEY} \nuevo{\kw{AUTOINCREMENT},}\\
\ind precio \nuevo{REAL} \kw{NOT NULL},\\
\ind categoria\_id INTEGER \kw{NOT NULL},\\
\ind \kw{FOREIGN KEY} (categoria\_id)\\
\ind\ind \kw{REFERENCES} categoria(id)\\
);
\end{sqlcode}
\end{columns}
\vspace{0.15cm}
\small SQLite tiene \textbf{muy pocos tipos}: \texttt{INTEGER}, \texttt{TEXT} y \texttt{REAL} cubren todo, y las fechas se guardan como texto. \textbf{Y las claves foráneas vienen apagadas}: sin el \texttt{PRAGMA} se aceptan y no se vigilan.
\end{frame}

\begin{frame}{Cada gestor tiene su propia notación}
\begin{block}{Lo que hay que llevarse}
El \textbf{SQL que aprendimos hoy es el mismo en todos}. Lo que cambia son \textbf{detalles}: cómo se numera solo el \texttt{id}, cómo se llama el tipo de texto, si existe o no \texttt{CREATE DATABASE}.
\end{block}
\vspace{0.4cm}
\begin{center}
\small
\begin{tabular}{>{\raggedright\arraybackslash}p{3.1cm}>{\ttfamily}p{3.0cm}>{\ttfamily}p{2.9cm}>{\ttfamily}p{2.4cm}}
\toprule
& \normalfont\bfseries MySQL & \normalfont\bfseries SQL Server & \normalfont\bfseries SQLite \\
\midrule
Numerar solo el \texttt{id} & AUTO\_INCREMENT & IDENTITY(1,1) & AUTOINCREMENT \\[4pt]
Texto & VARCHAR(80) & NVARCHAR(80) & TEXT \\[4pt]
Decimales & DECIMAL(10,2) & DECIMAL(10,2) & REAL \\[4pt]
Crear la base & CREATE DATABASE & CREATE DATABASE & \normalfont es un archivo \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.4cm}
\small Por eso no hay que aprender tres idiomas: se aprende \textbf{uno}, y al cambiar de gestor se consulta la tabla de equivalencias. \texttt{CREATE TABLE}, \texttt{PRIMARY KEY}, \texttt{FOREIGN KEY}, \texttt{NOT NULL} y \texttt{UNIQUE} son iguales en los tres.
\end{frame}

\begin{frame}{\ins{Práctica 5} Ahora prueba guardar y consultar}
\small Las tablas ya existen, pero están \textbf{vacías}. Escribe estas dos instrucciones y mira qué responde el gestor:
\vspace{0.3cm}
\begin{sqlcode}[.80\textwidth]
\textcolor{utpGris}{-- 1. guardar una fila}\\
\kw{INSERT INTO} cliente (nombre, correo)\\
\kw{VALUES} ('Raúl', 'raul@mail.com');\\[9pt]
\textcolor{utpGris}{-- 2. verla}\\
\kw{SELECT} * \kw{FROM} cliente;
\end{sqlcode}
\vspace{0.15cm}
\small \texttt{INSERT INTO} dice \textbf{en qué tabla y en qué columnas}; \texttt{VALUES} trae los datos, en el mismo orden. \texttt{SELECT * FROM} es «muéstrame todo de esta tabla».
\vspace{0.3cm}
\begin{block}{Dos cosas para comprobar}
\small ¿Qué valor le tocó al \texttt{id}, si no lo escribimos? \ Y prueba un pedido con \texttt{cliente\_id = 97}: \textbf{¿qué contesta?}
\end{block}
\end{frame}
