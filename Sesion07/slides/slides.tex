\begin{frame}[c]
\begin{block}{Logro de la sesión}
Al finalizar, podrás mirar una tabla y decir \textbf{qué columna depende de cuál}; con eso podrás \textbf{separarla en varias tablas} sin perder información.
\end{block}
\end{frame}

% 3 ---------------------------------------------------------------------
\begin{frame}{Repaso: lo que ya sabemos escribir}
\begin{columns}[T,onlytextwidth]
\column{.43\textwidth}
\small \textbf{Las reglas de nombrado}
\begin{itemize}\setlength\itemsep{3pt}\small
  \item Tablas y columnas en \textbf{minúscula}, en \textbf{singular} y sin tildes: \texttt{cliente}, \texttt{metodo\_pago}.
  \item La clave primaria siempre se llama \texttt{id}.
  \item La clave foránea lleva el nombre de la tabla a la que apunta más \texttt{\_id}: \texttt{cliente\_id}.
\end{itemize}
\vspace{5pt}
\small \textbf{Los cinco tipos}: \texttt{INT}, \texttt{VARCHAR(80)}, \texttt{DECIMAL(10,2)}, \texttt{DATE}, \texttt{BOOLEAN}.
\column{.52\textwidth}
\begin{sqlcode}[.95\textwidth]
\scriptsize
\begin{tabular}{@{\hspace{3pt}}l@{\hspace{1.2em}}l@{\hspace{1.2em}}l@{}}
\multicolumn{3}{@{\hspace{3pt}}l}{CREATE TABLE cliente (}\\
\hspace{1em}id      & INT         & PRIMARY KEY AUTO\_INCREMENT,\\
\hspace{1em}nombre  & VARCHAR(80) & NOT NULL,\\
\hspace{1em}correo  & VARCHAR(120)& UNIQUE\\
\multicolumn{3}{@{\hspace{3pt}}l}{);}\\
\end{tabular}
\end{sqlcode}
\end{columns}
\vspace{0.3cm}
\small \textbf{Y el dibujo antes de las tablas:} cada \textbf{entidad} es una tabla, cada \textbf{atributo} una columna y cada \textbf{instancia} una fila. La \textbf{cardinalidad} ---las dos preguntas, una por lado--- decide cómo se conectan.
\end{frame}

% 12 ---------------------------------------------------------------------
\begin{frame}{La pregunta de hoy: ``si sé esto, ¿sé aquello?''}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Sí funciona}
\small Te digo \texttt{producto\_id} = \textbf{1}.\\
¿Cuál es su precio? \textbf{89.00}, siempre.
\end{exampleblock}
\column{.48\textwidth}
\begin{alertblock}{No funciona}
\small Te digo \texttt{precio} = \textbf{89.00}.\\
¿Qué producto es? No se puede saber.
\end{alertblock}
\end{columns}
\vspace{0.45cm}
\begin{block}{La prueba}
Elige dos columnas y tapa el resto. Si al fijar la primera siempre queda \textbf{un único} valor en la segunda, la primera manda sobre la segunda.
\end{block}
\end{frame}

% 13 ---------------------------------------------------------------------
\begin{frame}{Ejemplo 1: solo \texttt{producto\_id} y \texttt{precio}}
\small Tapa el resto de la tabla de la tienda. Quedan estas dos columnas, en el mismo orden de filas:
\vspace{0.25cm}
\begin{center}
\small
\begin{tabular}{cc}
\toprule
\bfseries producto\_id & \bfseries precio \\
\midrule
1 & 89.00 \\
2 & 35.00 \\
1 & 89.00 \\
3 & 45.00 \\
2 & 35.00 \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.25cm}
\begin{exampleblock}{Sí manda}
\small El \textbf{1} aparece dos veces y las dos dice \textbf{89.00}. El \textbf{2} aparece dos veces y las dos dice \textbf{35.00}. Al fijar \texttt{producto\_id} queda \textbf{un único} precio: \texttt{producto\_id} \fdep{} \texttt{precio}.
\end{exampleblock}
\end{frame}

\end{frame}

% 16 ---------------------------------------------------------------------
\begin{frame}{Eso se llama dependencia funcional}
\concepto{Dependencia funcional}{Al fijar el valor de una columna, queda determinado \textbf{un único} valor de otra.}
\vspace{0.3cm}
\begin{columns}[c,onlytextwidth]
\column{.52\textwidth}
\begin{center}
\begin{tikzpicture}[node distance=8mm]
  \node[cajaverde,text width=2.6cm] (a) {\texttt{producto\_id}};
  \node[cajanaranja,text width=2.2cm,right=11mm of a] (b) {\texttt{precio}};
  \draw[flecha] (a)--(b);
  \node[font=\scriptsize,below=1mm of a] {determinante};
  \node[font=\scriptsize,below=1mm of b] {dependiente};
\end{tikzpicture}
\end{center}
\column{.43\textwidth}
\small Se escribe \texttt{producto\_id} \fdep{} \texttt{precio}\\[5pt]
Se lee de dos maneras:
\begin{itemize}\setlength\itemsep{4pt}\small
  \item ``sabiendo el \texttt{producto\_id} puedo saber el \texttt{precio}'';
  \item ``\texttt{producto\_id} determina a \texttt{precio}''.
\end{itemize}
\end{columns}
\vspace{0.4cm}
\small \textbf{Este es el único ejemplo que hay que tener claro hoy.} Lo demás es aplicar la misma prueba a otras columnas.
\end{frame}

% 18 ---------------------------------------------------------------------
\begin{frame}{A veces el determinante son dos columnas juntas}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\begin{alertblock}{Con una sola columna falla}
\small
\texttt{pedido\_id} \fdep{} \texttt{cantidad}: en el pedido 1 hay 2 y 1.\\[4pt]
\texttt{producto\_id} \fdep{} \texttt{cantidad}: del producto 1 hay 2 y 1.
\end{alertblock}
\column{.49\textwidth}
\begin{exampleblock}{Con las dos juntas funciona}
\small
\{\texttt{pedido\_id}, \texttt{producto\_id}\} \fdep{} \texttt{cantidad}\\[4pt]
``producto 1 en el pedido 1'' da \textbf{2} y nada más.
\end{exampleblock}
\end{columns}
\vspace{0.45cm}
\analogia{Es como un asiento en el cine: la fila sola no ubica a nadie y el número solo tampoco. ``Fila G, asiento 12'' sí señala un único asiento.}
\end{frame}

\small Regla: \textbf{cada determinante se lleva consigo a sus dependientes} y se vuelve la clave primaria (\texttt{id}) de su tabla.

\end{tabular}
\vspace{8pt}

\texttt{\textbf{pedido}} \hfill {\scriptsize \texttt{pedido\_id}\fdep{}\texttt{cliente\_id}}

\bfseries id & \bfseries cliente\_id \\
\midrule
1 & 40285571 \\
2 & 16458744 \\

\bottomrule
\end{tabular}

\begin{frame}{Lo que acabas de separar: dependencia transitiva}
\small En el catálogo, para llegar a la ciudad hay que \textbf{pasar por el proveedor}:
\vspace{0.3cm}
\begin{center}
\begin{tikzpicture}
  \node[cajaverde,font=\footnotesize] (a) {\texttt{producto\_id}};
  \node[caja,font=\footnotesize,right=22mm of a] (b) {\texttt{proveedor\_id}};
  \node[caja,font=\footnotesize,right=22mm of b] (c) {\texttt{proveedor\_ciudad}};
  \draw[flecha] (a)--(b) node[midway,above,font=\scriptsize]{directa};
  \draw[flecha] (b)--(c) node[midway,above,font=\scriptsize]{directa};
  \draw[flecha,draw=utpRojo] (a) to[out=-32,in=-148]
    node[midway,below,font=\scriptsize,text=utpRojo]{transitiva: se llega dando un rodeo} (c);
\end{tikzpicture}
\end{center}
\vspace{0.25cm}
\small Con datos: Audífonos $\rightarrow$ proveedor \textbf{10} $\rightarrow$ \textbf{Lima}. El \texttt{producto\_id} sí determina la ciudad, pero \textbf{no por sí mismo}: la hereda del proveedor.
\vspace{0.3cm}
\begin{alertblock}{Por qué molesta}
\small \textbf{Lima} aparece en dos filas del catálogo. Si SonidoTop se muda a Trujillo, hay que corregir las dos. Al mandar la ciudad a la tabla \texttt{proveedor}, se escribe \textbf{una sola vez}.
\end{alertblock}
\end{frame}

\begin{frame}{Dos listas que no se hablan: dependencia multivaluada}
\small Raúl tiene \textbf{dos direcciones} y \textbf{dos tarjetas}. Todo en una sola tabla:
\vspace{0.25cm}
\begin{center}
\footnotesize
\begin{tabular}{ccc}
\toprule
\bfseries cliente\_id & \bfseries direccion & \bfseries tarjeta \\
\midrule
40285571 & Av. Arequipa 100 & Visa \\
40285571 & Av. Arequipa 100 & Mastercard \\
40285571 & Jr. Lima 250     & Visa \\
40285571 & Jr. Lima 250     & Mastercard \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.25cm}
\small Las cuatro filas son todas las \textbf{combinaciones}, y ninguna significa nada: la Visa no se usa más en Av. Arequipa que en Jr. Lima. \textbf{Las dos listas no se relacionan entre sí.}
\vspace{0.3cm}
\begin{alertblock}{La señal, y la salida}
\small Si Raúl agrega una \textbf{tercera} tarjeta hay que escribir \textbf{dos} filas más, una por cada dirección. Dos listas independientes en la misma tabla se multiplican. La salida es separarlas: una tabla \texttt{direccion} y otra \texttt{tarjeta}, cada una con su \texttt{cliente\_id}.
\end{alertblock}
\end{frame}

\begin{frame}{Para saber más: tipos y práctica real\apoyo}
\begin{center}
\footnotesize\hyphenpenalty=10000\exhyphenpenalty=10000
\begin{tabular}{>{\bfseries\raggedright\arraybackslash}p{2.5cm}>{\raggedright\arraybackslash}p{3.3cm}>{\raggedright\arraybackslash}p{5.3cm}}
\toprule
Tipo & En qué consiste & En la tabla de la tienda \\
\midrule
Dependencia completa & Se necesita el determinante entero \textnormal{(también: plena)} & \{\texttt{pedido\_id}, \texttt{producto\_id}\} \fdep{} \texttt{cantidad} \\[4pt]
Dependencia transitiva & Se llega pasando por una tercera columna & \texttt{pedido\_id} \fdep{} \texttt{cliente\_id} \fdep{} \texttt{cliente\_nombre} \\[4pt]
Dependencia trivial & Lo pedido ya venía incluido en lo dado & \{\texttt{pedido\_id}, \texttt{producto\_id}\} \fdep{} \texttt{pedido\_id} \\[4pt]
Dependencia multivaluada & Un valor lleva a una lista independiente de otra & un cliente guarda varias direcciones y varias tarjetas, sin relación entre sí \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.25cm}
\begin{block}{¿Esto se usa en la práctica?}
\small Nadie dice ``esta dependencia es transitiva''; lo que sí se hace es \textbf{preguntar si el dato depende de toda la clave}.
\end{block}
\end{frame}
