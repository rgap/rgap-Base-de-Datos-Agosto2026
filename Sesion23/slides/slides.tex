\begin{block}{Logro de la sesión}
Al finalizar, podrás \textbf{conectarte} a un gestor sabiendo qué datos necesitas y escribir \textbf{subconsultas}: una consulta dentro de otra, para responder preguntas que se resuelven en \textbf{dos pasos}.

\begin{frame}{Revisión de la sesión anterior: el gestor es un servicio}
\begin{columns}[c,onlytextwidth]
\column{.47\textwidth}
\small El gestor es un programa que \textbf{no ves}. Funciona como \textbf{servicio}:
\vspace{0.2cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item no tiene ventana,
  \item trabaja en \textbf{segundo plano},
  \item arranca solo al encender Windows,
  \item y se queda \textbf{esperando pedidos}.
\end{itemize}
\vspace{0.3cm}
\small Los ves con \textbf{Windows + R} $\rightarrow$ \texttt{services.msc}.
\column{.5\textwidth}
\centering
\begin{tikzpicture}[font=\scriptsize]
  \node[caja,text width=1.9cm] (wb) at (0,0) {MySQL\\Workbench};
  \node[caja,text width=1.9cm] (ss) at (2.6,0) {SSMS};
  \node[caja,cajanaranja,text width=1.9cm] (my) at (0,-2.3) {\texttt{\textbf{mysqld.exe}}\\{\tiny servicio \texttt{MySQL80}}};
  \node[caja,cajanaranja,text width=1.9cm] (sq) at (2.6,-2.3) {\texttt{\textbf{sqlservr.exe}}\\{\tiny servicio \texttt{SQL Server}}};
  \draw[flechavuelta] (wb)--(my); \draw[flechavuelta] (ss)--(sq);
  \draw[utpGrisMedio,thick,dashed] (-1.5,-1.15)--(4.1,-1.15);
  \node[anchor=south west,font=\scriptsize\bfseries] at (-1.5,0.5) {Clientes: piden};
  \node[anchor=north west,font=\scriptsize\bfseries,text=utpRojo] at (-1.5,-2.95) {Servicios: atienden};
\end{tikzpicture}
\end{columns}
\vspace{0.3cm}
\concepto{Servicio}{\small Un programa sin ventana que arranca con Windows y siempre está listo para atender. Si el servicio está apagado, \textbf{nadie} puede conectarse.}
\end{frame}

\begin{frame}{Revisión: los clientes}
\concepto{Cliente}{\small \textbf{Cualquier programa} que se conecta al gestor para enviarle consultas y recibir resultados. Puede tener ventana (Workbench), ser una terminal (\texttt{mysql}) o no tener ventana (una app web, un programa en Node).}

\vspace{0.05cm}

\begin{frame}{Los cinco datos de una conexión}
\small ¿Qué necesitas para conectarte? Cinco datos:
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{@{}>{\bfseries}l>{\raggedright}p{5.0cm}>{\raggedright\arraybackslash}p{3.6cm}@{}}
\toprule
Dato & \bfseries Pregunta & \bfseries En clase \\
\midrule
Host & ¿en qué computadora? & \texttt{localhost} \\

Usuario & ¿quién eres? & \texttt{root} \\

\bottomrule

\end{center}

\begin{frame}{Paso a paso: con dos consultas}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\npieza{1}\ \small ¿Cuál es el promedio?
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT AVG}(precio) \kw{FROM} servicio;
\end{sqlcode}
\vspace{-0.1cm}
\res{r}{\cab{AVG(precio)}}{46.25 \\}
\column{.49\textwidth}
\npieza{2}\ \small ¿Qué servicios cuestan más?
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT} nombre, precio \kw{FROM} servicio\\
\kw{WHERE} precio > \mal{46.25};
\end{sqlcode}
\vspace{-0.1cm}
\res{lr}{\cab{nombre} & \cab{precio}}{Corte de pelo & 55.00 \\ Baño completo con corte & 80.00 \\}
\end{columns}
\vspace{0.3cm}
\ojo{Funciona, pero copiaste \texttt{46.25} \textbf{a mano}. Si mañana cambia un precio, ese número queda viejo.}
\end{frame}

\begin{frame}{En una sola: la subconsulta}
\small En lugar de escribir el número, escribe \textbf{la consulta que lo calcula}:
\vspace{0.2cm}
\begin{center}
\begin{tikzpicture}
  \node[codigo,anchor=west] (a) at (0,0) {\kw{WHERE} precio >};
  \node[codigo,anchor=west,draw=utpRojo,thick,fill=utpRojo!18] (n) at ([xshift=2pt]a.east) {46.25};
  \node[codigo,anchor=west] (b) at (0,-1.5) {\kw{WHERE} precio >};
  \node[sub,anchor=west] (s) at ([xshift=2pt]b.east) {(\kw{SELECT AVG}(precio) \kw{FROM} servicio)};
  \draw[flecha] (n.south)--node[right,font=\scriptsize]{se reemplaza por}(n.south|-s.north);
\end{tikzpicture}
\end{center}
\vspace{0.1cm}
\begin{sqlcode}[.62\textwidth]
\kw{SELECT} nombre, precio\\
\kw{FROM} servicio\\
\kw{WHERE} precio > \dentro{(\kw{SELECT AVG}(precio) \kw{FROM} servicio)};
\end{sqlcode}
\vspace{-0.1cm}
\concepto{Subconsulta}{\small Una consulta escrita \textbf{dentro de otra}, entre \textbf{paréntesis}.}
\end{frame}

\begin{frame}{Una consulta dentro de otra}
\begin{center}
\begin{tikzpicture}
  \node[codigo,anchor=west] (o1) at (0,0) {\kw{SELECT} nombre, precio \kw{FROM} servicio};
  \node[codigo,anchor=west] (o2) at (0,-0.75) {\kw{WHERE} precio >};
  \node[sub,anchor=west,inner sep=6pt] (in) at ([xshift=3pt]o2.east) {(\kw{SELECT AVG}(precio) \kw{FROM} servicio)};
  \node[font=\small\bfseries,text=utpRojo,anchor=north] (lin) at ([yshift=-4pt]in.south) {subconsulta (la de adentro)};
  \node[draw=utpNegro,very thick,rounded corners=4pt,inner sep=12pt,fit=(o1)(o2)(in)(lin)] (out) {};
  \node[num,minimum size=5mm,font=\bfseries\scriptsize] at (in.north east) {1};
  \node[num,minimum size=5mm,font=\bfseries\scriptsize] at (out.north west) {2};
  \node[font=\small\bfseries,anchor=south west] at ([xshift=8pt,yshift=2pt]out.north west) {consulta principal (la de afuera)};
\end{tikzpicture}
\end{center}
\vspace{0.4cm}
\begin{itemize}\setlength\itemsep{8pt}
  \item \npieza{1} La \textbf{subconsulta} se resuelve primero y entrega un resultado: \texttt{46.25}.
  \item \npieza{2} La \textbf{consulta principal} usa ese resultado como si lo hubieras escrito tú.
\end{itemize}
\end{frame}

\begin{frame}{¿Qué devuelve la de adentro?}
\small Lo primero es mirar la \textbf{forma} de su resultado. Eso decide cómo se usa.
\vspace{0.3cm}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}
\centering
{\large\bfseries Un valor}\\[8pt]
\forma{|c|}{\cellcolor{yellow!45}46.25 \\ \hline}\\[8pt]
\footnotesize una fila, una columna\\[10pt]
\etiqueta{= \ > \ < }\\[6pt]
\scriptsize se compara
\column{.31\textwidth}
\centering
{\large\bfseries Una lista}\\[8pt]
\forma{|c|}{\cellcolor{yellow!45}1 \\ \hline \cellcolor{yellow!45}2 \\ \hline}\\[8pt]
\footnotesize una columna, varias filas\\[10pt]
\etiqueta{IN \ NOT IN}\\[6pt]
\scriptsize se busca dentro
\column{.31\textwidth}
\centering
{\large\bfseries Una tabla}\\[8pt]
\forma{|c|c|}{\cellcolor{yellow!45}1 & \cellcolor{yellow!45}3 \\ \hline \cellcolor{yellow!45}2 & \cellcolor{yellow!45}2 \\ \hline}\\[8pt]
\footnotesize varias columnas y filas\\[10pt]
\etiqueta{FROM}\\[6pt]
\scriptsize se usa como tabla
\end{columns}
\vspace{0.5cm}
\small Veamos un ejemplo de cada una.
\end{frame}

\begin{frame}{Una lista: ¿qué empleados tienen citas?}
\small \texttt{IN} ya lo conoces: \texttt{WHERE id IN (1, 2)}. Ahora la lista la escribe \textbf{una consulta}.
\vspace{0.2cm}
\begin{center}
\begin{tikzpicture}
  \node (ci) at (0,0) {\mini{cita}{rr}{\cab{id} & \cab{empleado\_id}}{%
    1 & 1 \\ 2 & 2 \\ 3 & 1 \\ 4 & 2 \\ 5 & 1 \\}};
  \node (li) at (3.6,0) {\forma{|c|}{\cellcolor{yellow!45}1 \\ \hline \cellcolor{yellow!45}2 \\ \hline}};
  \node[font=\scriptsize\bfseries,above=2pt of li] {la lista};
  \node (em) at (7.6,0) {\mini{empleado}{rll}{\cab{id} & \cab{nombre} & }{%
    \fila 1 & Ana Torres & \si \\
    \fila 2 & Bruno Díaz & \si \\
    3 & Carla Núñez & \no \\}};
  \draw[flecha] (ci.east)--node[above,font=\scriptsize]{\npieza{1}}(li.west);
  \draw[flecha] (li.east)--node[above,font=\scriptsize]{\npieza{2}}node[below,font=\scriptsize]{¿está?}(em.west);
\end{tikzpicture}
\end{center}
\vspace{0.1cm}
\begin{sqlcode}[.62\textwidth]
\kw{SELECT} nombre \kw{FROM} empleado\\
\kw{WHERE} id \kw{IN} \dentro{(\kw{SELECT} empleado\_id \kw{FROM} cita)};
\end{sqlcode}
\vspace{-0.1cm}
\centering\small Resultado: \textbf{Ana Torres} y \textbf{Bruno Díaz}.
\end{frame}

\begin{frame}{Ojo: el signo \texttt{=} espera un solo valor}
\begin{sqlcode}[.8\textwidth]
\kw{SELECT} nombre \kw{FROM} empleado\\
\kw{WHERE} id \mal{=} \dentro{(\kw{SELECT} empleado\_id \kw{FROM} cita)};\\[4pt]
\color{utpRojo}ERROR 1242 (21000): Subquery returns more than 1 row
\end{sqlcode}
\vspace{0.2cm}
\begin{center}
\begin{tikzpicture}[font=\small]
  \node (l) {\forma{|c|}{1 \\ \hline 2 \\ \hline 1 \\ \hline}};
  \node[right=0.6cm of l,font=\Large\ttfamily] (q) {= ?};
  \node[right=0.6cm of q,text width=6cm] {¿Con cuál de todos compara? El gestor no puede elegir y se detiene.};
\end{tikzpicture}
\end{center}
\vspace{0.2cm}
\concepto{Regla}{\small Si la de adentro devuelve \textbf{un valor}, usa \texttt{=}, \texttt{>}, \texttt{<}. Si devuelve \textbf{una lista}, usa \texttt{IN}.}
\end{frame}

\begin{frame}{Una tabla: ¿cuántas citas tiene el que más tiene?}
\small \texttt{MAX(COUNT(*))} \textbf{no se puede} escribir. Primero cuentas; luego buscas el máximo.
\vspace{0.2cm}
\begin{center}
\begin{tikzpicture}
  \node (ci) at (0,0) {\mini{cita}{rr}{\cab{id} & \cab{empleado\_id}}{%
    1 & 1 \\ 2 & 2 \\ 3 & 1 \\ 4 & 2 \\ 5 & 1 \\}};
  \node[draw=utpRojo,very thick,dashed,rounded corners=3pt,fill=yellow!20,inner sep=6pt] (co) at (4.6,0)
    {\mini{conteo}{rr}{\cab{empleado\_id} & \cab{total}}{1 & 3 \\ 2 & 2 \\}};
  \node[font=\scriptsize,text=utpRojo,below=2pt of co] {tabla temporal};
  \node[caja,cajaverde,font=\large\bfseries,minimum width=1.2cm] (r) at (8.6,0) {3};
  \draw[flecha] (ci.east)--node[above,font=\scriptsize]{\npieza{1} contar}(co.west);
  \draw[flecha] (co.east)--node[above,font=\scriptsize]{\npieza{2} \texttt{MAX}}(r.west);
\end{tikzpicture}
\end{center}
\vspace{0.1cm}
\begin{sqlcode}[.62\textwidth]
\kw{SELECT MAX}(total)\\
\kw{FROM} \dentro{(\kw{SELECT} empleado\_id, \kw{COUNT}(*) \kw{AS} total}\\
\ind\ind\dentro{\kw{FROM} cita \kw{GROUP BY} empleado\_id)} \kw{AS} conteo;
\end{sqlcode}
\vspace{-0.3cm}
\ojo{En \texttt{FROM}, la subconsulta \textbf{necesita un nombre} (\texttt{AS conteo}). Sin él: \texttt{ERROR 1248: Every derived table must have its own alias}.}
\end{frame}

\begin{frame}{¿Dónde puede ir una subconsulta?}
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{@{}>{\bfseries}l>{\raggedright}p{3.4cm}>{\raggedright\arraybackslash}p{7.0cm}@{}}
\toprule
Dónde & \bfseries Qué devuelve & \bfseries Ejemplo \\
\midrule
WHERE & un valor & \footnotesize\ttfamily \kw{WHERE} precio > \dentro{(\kw{SELECT AVG}(precio) \ldots)} \\
WHERE & una lista & \footnotesize\ttfamily \kw{WHERE} id \kw{IN} \dentro{(\kw{SELECT} empleado\_id \ldots)} \\
FROM & una tabla & \footnotesize\ttfamily \kw{FROM} \dentro{(\kw{SELECT} \ldots)} \kw{AS} conteo \\
SELECT & un valor & \footnotesize\ttfamily \kw{SELECT} nombre, \dentro{(\kw{SELECT AVG}(precio) \ldots)} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.3cm}
\small En \texttt{SELECT}, el valor aparece como \textbf{una columna más}, repetido en cada fila:
\begin{sqlcode}[.8\textwidth]
\kw{SELECT} nombre, precio, \dentro{(\kw{SELECT AVG}(precio) \kw{FROM} servicio)} \kw{AS} promedio \kw{FROM} servicio;
\end{sqlcode}
\end{frame}

\begin{frame}{¿Subconsulta o \texttt{JOIN}?}
\small Muchas preguntas se pueden responder de las dos formas:
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\centering\etiqueta{Con subconsulta}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT} nombre \kw{FROM} empleado\\
\kw{WHERE} id \kw{IN} (\kw{SELECT} empleado\_id\\
\ind\ind\ind\ind\kw{FROM} cita);
\end{sqlcode}
\column{.49\textwidth}
\centering\etiqueta{Con JOIN}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT DISTINCT} e.nombre\\
\kw{FROM} empleado e\\
\kw{JOIN} cita c \kw{ON} c.empleado\_id = e.id;
\end{sqlcode}
\end{columns}
\vspace{0.1cm}
\centering\small Las dos dan \textbf{Ana Torres} y \textbf{Bruno Díaz}. Sin \texttt{DISTINCT}, el \texttt{JOIN} repite a Ana tres veces.
\vspace{0.3cm}
\raggedright
\begin{itemize}\setlength\itemsep{6pt}
  \item La \textbf{subconsulta} se lee casi como la pregunta: «empleados cuyo id está en las citas».
  \item El \textbf{JOIN} conviene cuando quieres ver columnas de \textbf{las dos} tablas.
\end{itemize}
\end{frame}
