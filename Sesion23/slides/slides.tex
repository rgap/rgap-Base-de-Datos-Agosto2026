\begin{frame}[c]
\begin{block}{Logro de la sesión}
Al finalizar, podrás \textbf{conectarte} a un gestor desde distintos clientes, sabiendo qué \textbf{cinco datos} necesitas y qué revisar cuando la conexión \textbf{falla}.
\end{block}
\end{frame}
% =======================================================================
% Revisión de la sesión anterior
% ---------------------------------------------------------------------
\begin{frame}{Revisión de la sesión anterior: el gestor es un servicio}
\begin{columns}[c,onlytextwidth]
\column{.47\textwidth}
\small El gestor es un programa que \textbf{no ves}. Funciona como \textbf{servicio}:
\vspace{0.2cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item no tiene ventana,
  \item trabaja en \textbf{segundo plano},
  \item arranca solo al encender Windows,
  \item y se queda \textbf{esperando pedidos}.
\end{itemize}
\vspace{0.3cm}
\small Los ves con \textbf{Windows + R} $\rightarrow$ \texttt{services.msc}.
\column{.5\textwidth}
\centering
\begin{tikzpicture}[font=\scriptsize]
  \node[caja,text width=1.9cm] (wb) at (0,0) {MySQL\\Workbench};
  \node[caja,text width=1.9cm] (ss) at (2.6,0) {SSMS};
  \node[caja,cajanaranja,text width=1.9cm] (my) at (0,-2.3) {\texttt{\textbf{mysqld.exe}}\\{\tiny servicio \texttt{MySQL80}}};
  \node[caja,cajanaranja,text width=1.9cm] (sq) at (2.6,-2.3) {\texttt{\textbf{sqlservr.exe}}\\{\tiny servicio \texttt{SQL Server}}};
  \draw[flechavuelta] (wb)--(my); \draw[flechavuelta] (ss)--(sq);
  \draw[utpGrisMedio,thick,dashed] (-1.5,-1.15)--(4.1,-1.15);
  \node[anchor=south west,font=\scriptsize\bfseries] at (-1.5,0.5) {Clientes: piden};
  \node[anchor=north west,font=\scriptsize\bfseries,text=utpRojo] at (-1.5,-2.95) {Servicios: atienden};
\end{tikzpicture}
\end{columns}
\vspace{0.3cm}
\concepto{Servicio}{\small Un programa sin ventana que arranca con Windows y siempre está listo para atender. Si el servicio está apagado, \textbf{nadie} puede conectarse.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Revisión: los clientes}
\concepto{Cliente}{\small \textbf{Cualquier programa} que se conecta al gestor para enviarle consultas y recibir resultados. Puede tener ventana (Workbench), ser una terminal (\texttt{mysql}) o no tener ventana (una app web, un programa en Node).}
\small Cada gestor tiene su cliente:
\begin{center}
\begin{tikzpicture}[font=\scriptsize]
  \foreach \i/\c/\g in {0/SSMS/SQL Server, 1/MySQL Workbench/MySQL,
      2/DB Browser for SQLite/SQLite, 3/pgAdmin/PostgreSQL} {
    \node[caja,text width=2.5cm] (c\i) at (3.3*\i,0) {\textbf{\c}};
    \node[caja,cajanaranja,text width=2.5cm] (g\i) at (3.3*\i,-1.2) {\g};
    \draw[flecha] (c\i)--(g\i);
  }
\end{tikzpicture}
\end{center}
\vspace{0.1cm}
\small Hay otros que sirven para \textbf{varios} gestores o para \textbf{dibujar} el modelo:\\[5pt]
\etiqueta{DataGrip} \etiqueta{DBeaver} \etiqueta{Luna Modeler} \etiqueta{Visual Paradigm} \etiqueta{HeidiSQL} {\footnotesize \ldots entre otros.}
\vspace{0.05cm}
\ojo{Si cierras el cliente, los datos \textbf{no} se borran: siguen en el gestor.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Los cinco datos de una conexión}
\small ¿Qué necesitas para conectarte? Cinco datos:
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{@{}>{\bfseries}l>{\raggedright}p{5.0cm}>{\raggedright\arraybackslash}p{3.6cm}@{}}
\toprule
Dato & \bfseries Pregunta & \bfseries En clase \\
\midrule
Host & ¿en qué computadora? & \texttt{localhost} \\
Puerto & ¿a qué programa? & \texttt{3306} \\
Usuario & ¿quién eres? & \texttt{root} \\
Contraseña & ¿cómo lo demuestras? & la de la instalación \\
Base de datos & ¿con cuál empiezas? \textit{(opcional)} & \texttt{peluqueria} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.3cm}
\concepto{Idea clave}{\small Todos los clientes piden \textbf{estos cinco datos}. Solo cambia dónde se escriben.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Los cinco datos en SQL Server con Windows}
\begin{center}
\small
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{@{}>{\bfseries}l>{\raggedright}p{3.0cm}c>{\raggedright\arraybackslash}p{7.2cm}@{}}
\toprule
Dato & \etiqueta{MySQL} & & \etiqueta{SQL Server con autenticación de Windows} \\
\midrule
Host & \texttt{localhost} & $\rightarrow$ & \textbf{Server Name}: \texttt{localhost} o \texttt{localhost\bs SQLEXPRESS} \\
Puerto & \texttt{3306} & $\rightarrow$ & no se escribe: usa \texttt{1433} solo \\
Usuario & \texttt{root} & $\rightarrow$ & \mal{tu usuario de Windows}: se llena solo \\
Contraseña & la de la instalación & $\rightarrow$ & \mal{no se pide} \\
Base de datos & \texttt{peluqueria} & $\rightarrow$ & \textbf{Database Name}, o luego \texttt{USE peluqueria;} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.2cm}
\concepto{Autenticación de Windows}{\small Windows ya comprobó quién eres cuando iniciaste sesión en la laptop. SQL Server \textbf{confía} en eso: por eso no pide contraseña.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Los cinco datos en SQLite}
\begin{center}
\small
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{@{}>{\bfseries}l>{\raggedright}p{3.0cm}c>{\raggedright\arraybackslash}p{7.2cm}@{}}
\toprule
Dato & \etiqueta{MySQL} & & \etiqueta{SQLite} \\
\midrule
Host & \texttt{localhost} & $\rightarrow$ & \mal{no hay}: el archivo está en tu computadora \\
Puerto & \texttt{3306} & $\rightarrow$ & \mal{no hay}: no hay servidor esperando \\
Usuario & \texttt{root} & $\rightarrow$ & \mal{no hay} \\
Contraseña & la de la instalación & $\rightarrow$ & \mal{no hay} \\
Base de datos & \texttt{peluqueria} & $\rightarrow$ & \textbf{el archivo}: \texttt{peluqueria.db} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{DB Browser for SQLite}\\[6pt]
\small \textit{Abrir base de datos} $\rightarrow$ eliges \texttt{peluqueria.db}
\column{.48\textwidth}
\centering\etiqueta{Terminal}
\begin{sqlcode}[.8\textwidth]
\tu{sqlite3 peluqueria.db}
\end{sqlcode}
\end{columns}
\vspace{0.1cm}
\concepto{Idea clave}{\small En SQLite, conectarse es \textbf{abrir un archivo}. Solo necesitas un dato: \textbf{dónde está}.}
\end{frame}

\begin{frame}{¿Qué pasa al conectarse?}
\begin{center}
\begin{tikzpicture}[font=\scriptsize]
  \node[caja,text width=2cm] (c) at (0,0) {\textbf{Cliente}\\{\tiny Workbench}};
  \node[caja,cajanaranja,text width=2.3cm] (s) at (6.3,0) {\textbf{Servidor}\\{\tiny MySQL en localhost:3306}};
  \draw[vida] (c.south)--(0,-3.9); \draw[vida] (s.south)--(6.3,-3.9);
  \ida{-0.95}{1. llega a localhost, puerto 3306}
  \vuelta{-1.5}{2. «¿quién eres?»}
  \ida{-2.05}{3. usuario y contraseña}
  \vuelta{-2.6}{4. «aceptado»: conexión abierta}
  \ida{-3.15}{5. consultas \ldots}
  \vuelta{-3.7}{6. \ldots{} y resultados}
\end{tikzpicture}
\end{center}
\vspace{0.2cm}
\small Si falla el paso 1, el problema es el host o el puerto. Si falla el 4, el usuario o la contraseña.
\end{frame}

\begin{frame}{Desde la terminal}
\small La \textbf{terminal} es una ventana para escribir órdenes (PowerShell o cmd).
\vspace{0.3cm}
\begin{center}
\renewcommand{\piezax}{3.5}
\begin{tikzpicture}
  \pieza{0}{mysql}{el programa cliente}
  \pieza{-0.68}{-h localhost}{host}
  \pieza{-1.36}{-P 3306}{puerto}
  \pieza{-2.04}{-u root}{usuario}
  \pieza{-2.72}{-p}{pide la contraseña}
  \pieza{-3.4}{peluqueria}{base de datos}
\end{tikzpicture}
\end{center}
\vspace{0.2cm}
\begin{sqlcode}[.7\textwidth]
\tu{mysql -h localhost -P 3306 -u root -p peluqueria}
\end{sqlcode}
\end{frame}

\begin{frame}{Cuando no conecta}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{@{}>{\raggedright}p{6.4cm}>{\raggedright\arraybackslash}p{4.6cm}@{}}
\toprule
\bfseries Mensaje & \bfseries Qué revisar \\
\midrule
\texttt{ERROR 2003: Can't connect to MySQL server on 'localhost:3306'} & ¿el servicio está encendido? ¿host y puerto? \\
\texttt{ERROR 1045: Access denied for user 'root'@'localhost'} & usuario y contraseña \\
\texttt{ERROR 1049: Unknown database 'peluqueria'} & ¿existe la base? \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.3cm}
\concepto{¿Está encendido el servicio?}{\small \textbf{Windows + R} $\rightarrow$ \texttt{services.msc} $\rightarrow$ \texttt{MySQL80}. Debe decir \textbf{En ejecución}; si no, clic derecho $\rightarrow$ \textit{Iniciar}.}
\end{frame}
\begin{frame}{Cierre}
\begin{center}
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{>{\raggedright\bfseries}p{3.4cm}>{\raggedright\arraybackslash}p{8.4cm}}
\toprule
Servicio & Programa sin ventana que arranca con Windows: \texttt{mysqld.exe}, \texttt{sqlservr.exe}. \\
Cliente & Cualquier programa que se conecta al gestor: SSMS, Workbench, la terminal, una app. \\
Conectarse & Host, puerto, usuario, contraseña y base. \\
SQL Server con Windows & No pide contraseña: confía en tu sesión de Windows. \\
SQLite & Conectarse es abrir un archivo. \\
Cadena de conexión & Los cinco datos en una sola línea. \\
Cuando no conecta & Revisa el servicio, el host y el puerto, y luego el usuario y la contraseña. \\
\bottomrule
\end{tabular}
\end{center}
\end{frame}
