\begin{frame}{Revisión: ¿qué son los triggers?}
\small Un bloque de SQL unido a una tabla. Se ejecuta \textbf{solo} cuando la tabla cambia.\par
\vspace{0.2cm}
{\footnotesize\textbf{Ejemplo, la agenda:} la fecha de registro no puede ser futura. En MySQL,
\mal{\texttt{\kw{CHECK} (fecha\_registro <= \kw{CURRENT\_DATE})}} da el error 3814; por eso la regla va en un trigger.\par}
\vspace{0.05cm}
\begin{center}
\begin{tikzpicture}
  \node[caja,font=\ttfamily\footnotesize,align=left] (i) at (0,0) {\kw{INSERT INTO} contacto \ldots\\fecha\_registro = \mal{'2027-01-15'}};
  \node[caja,cajanaranja,minimum height=1cm] (t) at (5.4,0) {\textbf{trigger} de \tbl{contacto}\\[-1pt]{\scriptsize ¿la fecha es futura?}};
  \node[caja,draw=utpRojo,align=left,font=\ttfamily\scriptsize,text=utpRojo] (e) at (10,0) {Error Code: 1644.\\La fecha de registro\\no puede ser futura.};
  \draw[flecha] (i) -- node[above,font=\scriptsize]{dispara} (t);
  \draw[flecha] (t) -- node[above,font=\scriptsize]{sí} (e);
\end{tikzpicture}
\end{center}
\vspace{0.2cm}
\begin{center}
\begin{tikzpicture}[ficha/.style={caja,minimum width=3.8cm,minimum height=2.05cm,text width=3.5cm,align=center,execute at begin node={\hyphenpenalty=10000}}]
  \node[ficha] at (0,0) {\textbf{Nombre}\\[4pt]\footnotesize\textit{trigger} = gatillo\\en español: \textbf{disparador}};
  \node[ficha,minimum width=8cm,text width=7.6cm] at (6.3,0) {\textbf{Origen y para qué}\\[4pt]\footnotesize Varios programas usan la misma base. Con un trigger, la regla se escribe \textbf{una vez}, en la base, y se cumple \textbf{siempre}, venga de donde venga el cambio.};
\end{tikzpicture}
\end{center}
\end{frame}

\begin{frame}{Paso 1: completar los datos antes de guardar}
\small El correo se guarda en minúsculas y el precio es el de la entrada, escriba lo que escriba quien inserta.
\vspace{0.05cm}
\begin{columns}[T,onlytextwidth]
\column{.55\textwidth}
\begin{sqlmini}[\col]
\kw{DELIMITER} //\\
\kw{CREATE TRIGGER} venta\_completa\_datos\\
\dentro{\kw{BEFORE INSERT}} \kw{ON} venta\hfill\npieza{1}\\
\kw{FOR EACH ROW}\\
\kw{BEGIN}\\
\ind\kw{SET} \dentro{NEW.correo} = \kw{LOWER}(\kw{TRIM}(NEW.correo));\hfill\npieza{2}\\
\ind\kw{SET} \dentro{NEW.precio} = (\kw{SELECT} precio \kw{FROM} entrada\\
\ind\ind\kw{WHERE} id = NEW.entrada\_id);\\
\kw{END} //\\
\kw{DELIMITER} ;
\end{sqlmini}
\column{.42\textwidth}
\footnotesize\raggedright
\vspace{4pt}
\npieza{1} \textbf{Antes} de guardar: la fila todavía se puede cambiar.\\[3pt]
\npieza{2} \texttt{SET NEW.columna}: cambia lo que se va a guardar.\\[3pt]
{\scriptsize\gris{Dos instrucciones: van \texttt{BEGIN}\ldots\texttt{END} y \texttt{DELIMITER}.}\par}
\begin{sqlmicro}[\col]
\kw{INSERT INTO} venta (dni, nombre, correo,\\
\ind precio, referencia\_pago)\\
\kw{VALUES} ('70123457', 'María Torres',\\
\ind\mal{'Maria.Torres@Example.com'}, \mal{1.00},\\
\ind'PAGO-0002');
\end{sqlmicro}
\end{columns}
\vspace{-0.1cm}
\begin{center}
\tablita{\texttt{SELECT dni, nombre, correo, precio FROM venta;}}{llrr}{\cab{dni} & \cab{nombre} & \cab{correo} & \cab{precio}}{\fila 70123457 & María Torres & maria.torres@example.com & 45.00 \\}
\end{center}
\end{frame}

\begin{frame}{Paso 2: detener una operación}
\small Una venta no se borra: queda como registro de lo que se cobró.
\vspace{0.15cm}
\begin{columns}[T,onlytextwidth]
\column{.52\textwidth}
\begin{sqlchico}[\col]
\kw{DELIMITER} //\\
\kw{CREATE TRIGGER} venta\_no\_se\_borra\\
\dentro{\kw{BEFORE DELETE}} \kw{ON} venta\\
\kw{FOR EACH ROW}\\
\kw{BEGIN}\\
\ind\dentro{\kw{SIGNAL SQLSTATE} '45000'}\\
\ind\ind\kw{SET MESSAGE\_TEXT} =\\
\ind\ind'Una venta no se puede borrar.';\\
\kw{END} //\\
\kw{DELIMITER} ;
\end{sqlchico}
\column{.45\textwidth}
\begin{sqlchico}[\col]
\kw{DELETE FROM} venta\\
\kw{WHERE} dni = '70123457';
\end{sqlchico}
\vspace{-4pt}
{\centering\errlinea{\footnotesize Error Code: 1644. Una venta no se puede borrar.}\par}
\vspace{6pt}
\begin{sqlchico}[\col]
\kw{SELECT} \kw{COUNT}(*) \kw{AS} ventas \kw{FROM} venta;
\end{sqlchico}
\resg{r}{\cab{ventas}}{1 \\}
\end{columns}
\vspace{0.2cm}
\concepto{SIGNAL en un trigger BEFORE}{\small Cancela la operación: la fila queda como estaba. Es la regla del flujo de caja: \textit{«Un movimiento anotado no se puede editar ni eliminar»}.}
\end{frame}

\end{center}
\vspace{-0.1cm}
\concepto{Paso 3}{\small Los pasos 1 y 2 cambiaban o detenían la \textbf{misma} fila; este escribe en \textbf{otra} tabla.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Paso 3: el historial de precios, \texttt{OLD} y \texttt{NEW}}
\small El trigger de la lámina anterior, en MySQL.
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{.55\textwidth}
\begin{sqlmini}[\col]
\kw{DELIMITER} //\\
\kw{CREATE TRIGGER} entrada\_guarda\_cambio\_precio\\
\dentro{\kw{AFTER UPDATE}} \kw{ON} entrada\\
\kw{FOR EACH ROW}\\
\kw{BEGIN}\\
\ind\kw{IF} \dentro{OLD}.precio <> \dentro{NEW}.precio \kw{THEN}\\
\ind\ind\kw{INSERT INTO} cambio\_precio\\
\ind\ind\ind(precio\_anterior, precio\_nuevo)\\
\ind\ind\kw{VALUES} (OLD.precio, NEW.precio);\\
\ind\kw{END IF};\\
\kw{END} //\\
\kw{DELIMITER} ;
\end{sqlmini}
\small\raggedright
\texttt{OLD}: la fila \textbf{antes} del \texttt{UPDATE}.\\
\texttt{NEW}: la fila \textbf{después}.\\[3pt]
{\footnotesize El \texttt{IF}: si el \texttt{UPDATE} cambia otra columna, como \texttt{disponibles}, no se anota nada.\par}
\column{.42\textwidth}
\footnotesize La tabla del historial:
\begin{sqlmicro}[\col]
\kw{CREATE TABLE} cambio\_precio (\\
\ind id \kw{INT AUTO\_INCREMENT PRIMARY KEY},\\
\ind precio\_anterior \kw{DECIMAL}(10, 2) \kw{NOT NULL},\\
\ind precio\_nuevo \kw{DECIMAL}(10, 2) \kw{NOT NULL},\\
\ind fecha \kw{DATETIME NOT NULL}\\
\ind\ind\kw{DEFAULT} (\kw{UTC\_TIMESTAMP}())\\
);
\end{sqlmicro}
\begin{sqlmini}[\col]
\kw{UPDATE} entrada \kw{SET} precio = 50.00\\
\kw{WHERE} id = 1;\\[2pt]
\kw{SELECT} precio\_anterior, precio\_nuevo\\
\kw{FROM} cambio\_precio;
\end{sqlmini}
\resg{rr}{\cab{precio\_anterior} & \cab{precio\_nuevo}}{45.00 & 50.00 \\}
\end{columns}
\end{frame}

\begin{frame}{MySQL: las piezas de un trigger}
\vspace{-0.1cm}
\begin{center}
\renewcommand{\piezax}{6.4}
\begin{tikzpicture}

  \pieza{-2.25}{\kw{ON} venta}{\textbf{en qué tabla}}

\begin{frame}{¿Por qué \texttt{FOR EACH ROW}?}
\small Un \texttt{INSERT} puede guardar \textbf{varias} filas: el trigger se ejecuta \textbf{una vez por cada una}.
\vspace{0.05cm}
\begin{sqlmini}[.9\textwidth]
\kw{INSERT INTO} venta (dni, nombre, correo, referencia\_pago)\\
\kw{VALUES} \dentro{('70123461', 'Ana Ruiz', 'Ana.Ruiz@Example.com', 'PAGO-0011')},\\
\ind\ind\dentro{('70123462', 'Jorge Soto', 'JORGE.SOTO@EXAMPLE.COM', 'PAGO-0012')},\\
\ind\ind\dentro{('70123463', 'Lucía Vega', ' Lucia.Vega@Example.com ', 'PAGO-0013')};
\end{sqlmini}
\vspace{-0.25cm}
\begin{columns}[c,onlytextwidth]
\column{.42\textwidth}
\centering
\begin{tikzpicture}[node distance=.15cm,
  f/.style={caja,font=\scriptsize,minimum width=1.3cm,minimum height=.55cm},
  t/.style={caja,cajanaranja,font=\ttfamily\scriptsize,minimum height=.55cm}]
  \node[f] (f1) {fila 1};
  \node[f,below=of f1] (f2) {fila 2};
  \node[f,below=of f2] (f3) {fila 3};
  \node[t,right=.6cm of f1] (t1) {venta\_completa\_datos};
  \node[t,right=.6cm of f2] (t2) {venta\_completa\_datos};
  \node[t,right=.6cm of f3] (t3) {venta\_completa\_datos};
  \draw[flecha] (f1)--(t1); \draw[flecha] (f2)--(t2); \draw[flecha] (f3)--(t3);
\end{tikzpicture}\\[3pt]
\small Un \texttt{INSERT}, 3 filas: \textbf{3 veces}.
\column{.55\textwidth}
\begin{sqlmini}[\col]
\kw{SELECT} dni, correo, precio \kw{FROM} venta\\
\kw{WHERE} dni > '70123460';
\end{sqlmini}
\resg{llr}{\cab{dni} & \cab{correo} & \cab{precio}}{70123461 & ana.ruiz@example.com & 50.00 \\ 70123462 & jorge.soto@example.com & 50.00 \\ 70123463 & lucia.vega@example.com & 50.00 \\}
\end{columns}
\vspace{0.05cm}
\concepto{¿Por qué así?}{\small Cada fila tiene su propio \texttt{NEW}: su correo, su precio. Así el trigger completa \textbf{cada una}.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{MySQL: \texttt{BEFORE} o \texttt{AFTER}, \texttt{NEW} y \texttt{OLD}}
\begin{columns}[T,onlytextwidth]
\column{.46\textwidth}
\centering\etiqueta{BEFORE}\par\vspace{4pt}
\raggedright\small La fila \textbf{todavía no} se guardó.\\[4pt]
\npieza{1} Completarla con \texttt{SET NEW.\ldots} \gris{(paso 1)}.\\[3pt]
\npieza{2} Detener la operación con \texttt{SIGNAL} \gris{(paso 2)}.\par\vspace{12pt}
\centering\etiqueta{AFTER}\par\vspace{4pt}
\raggedright La fila \textbf{ya} se guardó.\\[4pt]
\npieza{1} Cambiar \textbf{otra} tabla \gris{(paso 3)}.
\column{.5\textwidth}
\centering\etiqueta{NEW y OLD}\par\vspace{4pt}
\footnotesize
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{@{}>{\ttfamily\bfseries}l>{\raggedright}p{2cm}>{\raggedright\arraybackslash}p{2cm}@{}}
\toprule
 & \texttt{OLD} & \texttt{NEW} \\
\midrule
INSERT & \gris{no existe} & la fila nueva \\
UPDATE & la fila antes & la fila después \\
DELETE & la fila que se borra & \gris{no existe} \\
\bottomrule
\end{tabular}\\[8pt]
\small\raggedright Se usan con el nombre de la columna: \texttt{NEW.precio}, \texttt{OLD.precio}.
\end{columns}
\vspace{0.4cm}
\concepto{Regla práctica}{\small ¿Revisar o completar la \textbf{misma} fila? \texttt{BEFORE}. ¿Cambiar \textbf{otra} tabla? \texttt{AFTER}.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{SQL Server: las piezas de un trigger}
\begin{columns}[c,onlytextwidth]
\column{.62\textwidth}
\renewcommand{\piezax}{5.3}
\begin{tikzpicture}
  \pieza{0}{\kw{CREATE OR ALTER TRIGGER}}{crea o reemplaza}
  \pieza{-0.72}{dbo.entrada\_guarda\_cambio\_precio}{su \textbf{nombre}}
  \pieza{-1.44}{\kw{ON} entrada}{\textbf{en qué tabla}}
  \pieza{-2.16}{\kw{AFTER UPDATE}}{\textbf{cuándo} y \textbf{ante qué}}
  \pieza{-2.88}{\kw{AS}}{aquí empieza}
  \pieza{-3.6}{\kw{BEGIN} \ldots{} \kw{END};}{lo que \textbf{hace}}
  \pieza{-4.32}{\kw{GO}}{aquí termina}
\end{tikzpicture}
\column{.36\textwidth}
\small\raggedright
Lo que cambia frente a MySQL:\\[6pt]
\npieza{1} \texttt{ON entrada} va \textbf{antes} del momento.\\[5pt]
\npieza{2} No hay \texttt{BEFORE}: hay \texttt{AFTER} e \texttt{INSTEAD OF}.\\[5pt]
\npieza{3} No hay \texttt{FOR EACH ROW}: se ejecuta \textbf{una vez por instrucción}.\\[5pt]
\npieza{4} No hay \texttt{NEW} ni \texttt{OLD}: hay dos tablas, \texttt{inserted} y \texttt{deleted}.
\end{columns}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{SQL Server: el historial de precios}
\small El paso 3: \texttt{deleted} tiene el precio de antes; \texttt{inserted}, el de después.
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlmini}[\col]
\kw{CREATE TRIGGER} entrada\_guarda\_cambio\_precio\\
\kw{AFTER UPDATE ON} entrada\\
\kw{FOR EACH ROW}\\
\kw{BEGIN}\\
\ind\kw{IF} \dentro{OLD.precio <> NEW.precio} \kw{THEN}\\
\ind\ind\kw{INSERT INTO} cambio\_precio\\
\ind\ind\ind(precio\_anterior, precio\_nuevo)\\
\ind\ind\kw{VALUES} (OLD.precio, NEW.precio);\\
\ind\kw{END IF};\\
\kw{END} //
\end{sqlmini}
\resg{rr}{\cab{precio\_anterior} & \cab{precio\_nuevo}}{45.00 & 50.00 \\}
\par\vspace{3pt}
{\centering\scriptsize\gris{Mismo resultado en los dos gestores.}\par}
\column{.48\textwidth}
\centering\etiqueta{SQL Server}
\begin{sqlmini}[\col]
\kw{CREATE OR ALTER TRIGGER}\\
\ind dbo.entrada\_guarda\_cambio\_precio\\
\kw{ON} entrada \kw{AFTER UPDATE}\\
\kw{AS}\\
\kw{BEGIN}\\
\ind\kw{SET NOCOUNT ON};\\
\ind\kw{INSERT INTO} cambio\_precio\\
\ind\ind(precio\_anterior, precio\_nuevo)\\
\ind\kw{SELECT} d.precio, i.precio\\
\ind\kw{FROM} \dentro{deleted d}\\
\ind\kw{JOIN} \dentro{inserted i} \kw{ON} i.id = d.id\\
\ind\kw{WHERE} \dentro{d.precio <> i.precio};\\
\kw{END};\\
\kw{GO}
\end{sqlmini}
\raggedright\footnotesize En vez del \texttt{IF}, un \texttt{WHERE}: guarda solo las filas cuyo precio cambió, sean una o muchas.
\end{columns}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{SQL Server: \texttt{INSTEAD OF}, en lugar de la instrucción}
\small No hay \texttt{BEFORE}: para detener una operación, el trigger se ejecuta \textbf{en lugar} de ella. El paso 2:
\vspace{0cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlmini}[\col]
\kw{CREATE TRIGGER} venta\_no\_se\_borra\\
\dentro{\kw{BEFORE DELETE}} \kw{ON} venta\\
\kw{FOR EACH ROW}\\
\kw{BEGIN}\\
\ind\dentro{\kw{SIGNAL SQLSTATE} '45000'}\\
\ind\ind\kw{SET MESSAGE\_TEXT} =\\
\ind\ind'Una venta no se puede borrar.';\\
\kw{END} //
\end{sqlmini}
\column{.48\textwidth}
\centering\etiqueta{SQL Server}
\begin{sqlmini}[\col]
\kw{CREATE OR ALTER TRIGGER} dbo.venta\_no\_se\_borra\\
\kw{ON} venta\\
\dentro{\kw{INSTEAD OF DELETE}}\\
\kw{AS}\\
\kw{BEGIN}\\
\ind\dentro{\kw{THROW} 50000,}\\
\ind\ind N'Una venta no se puede borrar.', 1;\\
\kw{END};\\
\kw{GO}
\end{sqlmini}
\raggedright\scriptsize\texttt{DELETE FROM venta WHERE dni = '70123457';}\\[2pt]
\errlinea{Msg 50000, Level 16, State 1, Procedure venta\_no\_se\_borra, Line 12\\Una venta no se puede borrar.}
\end{columns}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{MySQL y SQL Server: lo que cambia}
\vspace{-0.15cm}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}>{\bfseries}p{3.1cm}>{\raggedright}p{4.9cm}>{\raggedright\arraybackslash}p{5.2cm}@{}}
\toprule
 & MySQL & SQL Server \\
\midrule
Crear & \texttt{CREATE TRIGGER t} & \texttt{CREATE OR ALTER TRIGGER dbo.t} \\
Tabla y momento & \texttt{BEFORE INSERT ON venta} & \texttt{ON venta AFTER INSERT} \\
Momentos & \texttt{BEFORE}, \texttt{AFTER} & \texttt{AFTER}, \texttt{INSTEAD OF} \\
Cuántas veces & Una por \textbf{fila}: \texttt{FOR EACH ROW} & Una por \textbf{instrucción} \\
La fila que cambia & \texttt{NEW.columna}, \texttt{OLD.columna} & Tablas \texttt{inserted} y \texttt{deleted} \\
Completar un dato & \texttt{BEFORE} con \texttt{SET NEW.precio = \ldots} & \texttt{INSTEAD OF} que hace el \texttt{INSERT} \\
Detener & \texttt{SIGNAL SQLSTATE '45000'} & \texttt{THROW 50000, N'\ldots', 1;} \\
Cuerpo & \texttt{DELIMITER //} \ldots{} \texttt{BEGIN} \ldots{} \texttt{END //} & \texttt{AS BEGIN} \ldots{} \texttt{END; GO} \\
Ver & \texttt{SHOW TRIGGERS;} & \texttt{SELECT name FROM sys.triggers;} \\
Borrar & \texttt{DROP TRIGGER IF EXISTS t;} & \texttt{DROP TRIGGER IF EXISTS dbo.t;} \\
\bottomrule
\end{tabular}
\end{center}
\end{frame}

\concepto{Copia de seguridad (\textit{backup})}{\small Un \textbf{archivo} con todo lo necesario para volver a crear la base: las tablas, los datos y los triggers.}

\begin{frame}{MySQL: sacar la copia con \texttt{mysqldump}}
\small MySQL no tiene una instrucción \texttt{BACKUP}: la copia la saca \texttt{mysqldump}, un programa que viene con MySQL. Se ejecuta en \textbf{PowerShell}, no en Workbench, dentro de la carpeta \texttt{C:\textbackslash Program Files\textbackslash MySQL\textbackslash MySQL Server 8.4\textbackslash bin}:
\vspace{0.15cm}
\begin{sqlmicro}[\textwidth]
mkdir \$HOME\textbackslash Desktop\textbackslash backups\\
.\textbackslash mysqldump -u root -p --databases agenda --result-file=\$HOME\textbackslash Desktop\textbackslash backups\textbackslash agenda.sql
\end{sqlmicro}
\vspace{0.05cm}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}>{\ttfamily}l>{\raggedright\arraybackslash}p{6.6cm}@{}}
\toprule
\normalfont\bfseries Pieza & \bfseries Qué hace \\
\midrule
-u root -p & Entra como \texttt{root} y pide su contraseña. \\
--databases agenda & La base que se copia. \\
--result-file=\ldots & El archivo de la copia, en la carpeta \texttt{backups} del Escritorio. \texttt{\$HOME} es tu carpeta de usuario. \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.1cm}
\concepto{¿Salió bien?}{\small Después de la contraseña no muestra nada: si no aparece un error, la copia ya está en \texttt{backups}. El \texttt{mkdir} solo hace falta la primera vez.}
\end{frame}

\begin{frame}{¿Qué hay dentro de la copia?}
\small Es un \textbf{script SQL}: se puede abrir con el Bloc de notas. Un resumen de \texttt{agenda.sql}:
\vspace{0.15cm}
\begin{columns}[c,onlytextwidth]
\column{.62\textwidth}
\begin{sqlmicro}[\col]
\kw{CREATE DATABASE} \ldots{} \bq agenda\bq{} \ldots;\hfill\npieza{1}\\
\kw{USE} \bq agenda\bq;\\[4pt]
\kw{DROP TABLE IF EXISTS} \bq contacto\bq;\hfill\npieza{2}\\
\kw{CREATE TABLE} \bq contacto\bq{} ( \ldots{} );\\[4pt]
\kw{INSERT INTO} \bq contacto\bq{} \kw{VALUES} (1,'Ana Torres',\ldots),\ldots;\hfill\npieza{3}\\[4pt]
\ldots{} \kw{TRIGGER} \bq contacto\_fecha\_no\_futura\_al\_registrar\bq\hfill\npieza{4}\\

\com{-- y lo mismo con contacto\_grupo y grupo}
\end{sqlmicro}

\small\raggedright
\npieza{1} Crea la base, si no existe.\\[6pt]
\npieza{2} Borra la tabla y la crea de nuevo.\\[6pt]
\npieza{3} Vuelve a poner los datos.\\[6pt]
\npieza{4} Vuelve a crear los triggers.

\concepto{Buenas costumbres}{\small Sacar la copia \textbf{seguido}, de forma automática; guardarla en \textbf{otro lugar} (otro disco o la nube); y \textbf{probar} que se puede volver a ella.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{SQL Server: \texttt{BACKUP} y \texttt{RESTORE}}
\small Si la agenda estuviera en SQL Server, la copia se saca con una instrucción, desde SSMS. El archivo es un \texttt{.bak}: no es un script, solo SQL Server lo puede leer.
\vspace{0.15cm}
\begin{columns}[T,onlytextwidth]
\column{.46\textwidth}
\centering\etiqueta{Sacar la copia}
\begin{sqlchico}[\col]
\dentro{\kw{BACKUP DATABASE}} agenda\\
\kw{TO DISK} = 'agenda.bak'\\
\kw{WITH INIT};
\end{sqlchico}
\raggedright\footnotesize
\npieza{1} Sin ruta, el archivo va a la carpeta \texttt{Backup} de SQL Server.\\[3pt]
\npieza{2} \texttt{WITH INIT}: reemplaza lo que ya tenía el archivo.
\column{.5\textwidth}
\centering\etiqueta{Volver a la copia}
\begin{sqlchico}[\col]
\kw{USE} master;\\
\kw{GO}\\
\kw{ALTER DATABASE} agenda\\
\ind\kw{SET SINGLE\_USER WITH ROLLBACK IMMEDIATE};\\
\dentro{\kw{RESTORE DATABASE}} agenda\\
\kw{FROM DISK} = 'agenda.bak'\\
\kw{WITH REPLACE};
\end{sqlchico}
\raggedright\footnotesize
\npieza{3} Desde \texttt{master}, sin nadie conectado a la base.\\[3pt]
\npieza{4} \texttt{WITH REPLACE}: reemplaza la base que ya existe.
\end{columns}
\vspace{0.2cm}
{\centering\footnotesize\gris{En SSMS: clic derecho en la base, Tasks, Back Up\ldots{} y Restore.}\par}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{MySQL y SQL Server: la copia}
\small SQL Server tiene dos instrucciones: \texttt{BACKUP} y \texttt{RESTORE}. MySQL \textbf{no} las tiene: usa un programa y un script. La idea es la misma; cambia \textbf{cómo} se hace.
\vspace{0.05cm}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{@{}>{\bfseries\raggedright}p{2.6cm}>{\raggedright}p{5cm}>{\raggedright\arraybackslash}p{5cm}@{}}
\toprule
 & \bfseries MySQL & \bfseries SQL Server \\
\midrule
Sacar la copia & El programa \texttt{mysqldump}, en PowerShell & La instrucción \texttt{BACKUP DATABASE}, en SSMS \\
Volver a la copia & Ejecutar su script, con \texttt{source} & La instrucción \texttt{RESTORE DATABASE} \\
El archivo & \texttt{.sql}: un script, se lee con el Bloc de notas & \texttt{.bak}: solo SQL Server lo lee \\
Qué guarda & Tablas, datos y triggers. Procedimientos y funciones, solo con \texttt{--routines} & Toda la base \\
Con el mouse & Workbench: Server, Data Export y Data Import & SSMS: Tasks, Back Up\ldots{} y Restore \\
\bottomrule
\end{tabular}
\end{center}
\vspace{-0.1cm}
\concepto{En los dos}{\small La copia se saca \textbf{seguido}, se guarda en \textbf{otro lugar} y se \textbf{prueba}.}
\end{frame}
% =======================================================================
\seccion{Usuarios y permisos}
\begin{frame}{¿Qué es un usuario? ¿Y un permiso?}
\small Para entrar a MySQL hace falta un \textbf{usuario}: un nombre y una contraseña. Cada usuario tiene \textbf{permisos}: lo que puede hacer en la base.\par
\vspace{0.2cm}
La aplicación de la agenda entra hoy como \texttt{root}, el usuario que puede hacer \textbf{todo} (en \texttt{src/db.js}):
\begin{sqlchico}[.7\textwidth]
mysql://\mal{root}@localhost:3306/agenda
\end{sqlchico}
\vspace{0.05cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{Si entra como root}\par\vspace{4pt}
\raggedright\small Un error en la aplicación o una contraseña robada pueden borrar la base entera: \texttt{DROP DATABASE agenda;}
\column{.48\textwidth}
\centering\etiqueta{Si entra como app\_agenda}\par\vspace{4pt}
\raggedright\small Solo puede leer, agregar, cambiar y borrar filas: lo que hace la aplicación. No puede borrar tablas ni la base.
\end{columns}
\vspace{0.3cm}
\concepto{Mínimo privilegio}{\small Cada usuario recibe \textbf{solo} los permisos que necesita para su trabajo.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{MySQL: conectarse desde la terminal}
\small Al instalar MySQL ya existe un usuario: \texttt{root}. Con él entras desde PowerShell, en la carpeta \texttt{bin} de MySQL, la misma de \texttt{mysqldump}.
\vspace{0.05cm}
\begin{columns}[c,onlytextwidth]
\column{.5\textwidth}
\renewcommand{\piezax}{3.55}
\begin{tikzpicture}
  \pieza{0}{.\textbackslash mysql}{el programa}
  \pieza{-0.72}{-u root}{el \textbf{usuario}}
  \pieza{-1.44}{-p}{pide la \textbf{contraseña}}
  \pieza{-2.16}{agenda}{la \textbf{base}}
\end{tikzpicture}\\[6pt]
{\scriptsize\gris{Sin \texttt{-h} ni \texttt{-P}: entra a \texttt{localhost}, puerto \texttt{3306}.}\par}
\column{.47\textwidth}
\centering\etiqueta{Solo con el usuario}
\begin{sqlmini}[\col]
.\textbackslash mysql -u root -p\\
Enter password: ********\\
mysql> \dentro{\kw{USE} agenda;}
\end{sqlmini}
\vspace{-2pt}
\etiqueta{Con el usuario y la base}
\begin{sqlmini}[\col]
.\textbackslash mysql -u root -p \dentro{agenda}\\
Enter password: ********\\
mysql> \kw{SELECT} nombre \kw{FROM} contacto;
\end{sqlmini}
\vspace{-4pt}
{\scriptsize\gris{Para salir: \texttt{exit}.}\par}
\end{columns}
\vspace{0.15cm}
\concepto{Igual que la aplicación}{\small \texttt{mysql://\textbf{root}@localhost:3306/\textbf{agenda}}: el usuario y la base.}
\end{frame}

\begin{tabular}{>{\raggedright\bfseries}p{3.4cm}>{\raggedright\arraybackslash}p{9.6cm}}
\toprule
Trigger & Se ejecuta solo cuando una tabla cambia: \texttt{INSERT}, \texttt{UPDATE} o \texttt{DELETE}. Sirve para completar datos, detener una operación o guardar un historial. \\
