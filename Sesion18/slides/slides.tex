\begin{frame}{Revisión de la sesión anterior: ¿qué hace un \texttt{JOIN}?}
\begin{columns}[c,onlytextwidth]
\column{.20\textwidth}
\tablacliente\\[3pt]
{\normalfont\scriptsize\color{utpGris}tabla \textbf{principal}}
\column{.28\textwidth}
\tablapedido\\[3pt]
{\normalfont\scriptsize\color{utpGris}la \textbf{otra} tabla}
\column{.48\textwidth}
\small Tienes una \textbf{tabla principal} y buscas \textbf{coincidencias} en otra tabla, usando una \textbf{columna en común}.
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{ON} cliente.id = pedido.cliente\_id
\end{sqlcode}
\end{columns}
\vspace{0.25cm}
\begin{itemize}\setlength\itemsep{5pt}
  \item \textbf{Ana} tiene dos coincidencias y \textbf{Carla} una.
  \item \mal{Bruno} no tiene ninguna: no hizo pedidos.
  \item El \mal{Monitor} tampoco: su \texttt{cliente\_id} es 4 y ese cliente no existe.
\end{itemize}
\vspace{0.2cm}
\concepto{La pregunta de siempre}{\small Cada tipo de \texttt{JOIN} decide \textbf{qué hacer con las filas que no encontraron coincidencia}.}
\end{frame}

\begin{frame}{Revisión de la sesión anterior: los cuatro \texttt{JOIN}}
\begin{center}
\renewcommand{\arraystretch}{1.7}
\small
\begin{tabular}{>{\centering\arraybackslash}m{1.5cm}>{\ttfamily\bfseries\raggedright\arraybackslash}m{2.9cm}>{\raggedright\arraybackslash}m{3.5cm}>{\raggedright\arraybackslash}m{4.9cm}}
\toprule
& \normalfont\bfseries Tipo & \bfseries Qué conserva & \bfseries En el ejemplo \\
\midrule
\venn[0.42]{\voff}{\von}{\voff} & INNER JOIN & solo las coincidencias & Ana y Carla. Bruno no aparece. \\
\venn[0.42]{\von}{\von}{\voff} & LEFT JOIN & toda la tabla izquierda & Ana, Bruno y Carla. Bruno con \nulo. \\
\venn[0.42]{\voff}{\von}{\von} & RIGHT JOIN & toda la tabla derecha & todos los pedidos, también el Monitor. \\
\venn[0.42]{\von}{\von}{\von} & FULL OUTER JOIN & todo, de ambas tablas & todos; lo que no coincide va con \nulo. \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.15cm}
\begin{columns}[T,onlytextwidth]
\column{.32\textwidth}
\centering\etiqueta{LEFT JOIN}\\[5pt]
\small «mi tabla principal \textbf{manda}»
\column{.32\textwidth}
\centering\etiqueta{INNER JOIN}\\[5pt]
\small «solo me interesan las \textbf{coincidencias}»
\column{.32\textwidth}
\centering\etiqueta{FULL JOIN}\\[5pt]
\small «quiero \textbf{todo}, coincida o no»
\end{columns}
\end{frame}

\begin{frame}{Revisión de la sesión anterior: leer un \texttt{LEFT JOIN}}
\begin{columns}[c,onlytextwidth]
\column{.52\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} c.id, c.nombre, p.producto\\
\kw{FROM} cliente c\\
\nuevo{\kw{LEFT JOIN}} pedido p\\
\ind \kw{ON} c.id = p.cliente\_id;
\end{sqlcode}
\column{.42\textwidth}
\res{rll}{\cab{id} & \cab{nombre} & \cab{producto}}{%
  1 & Ana & Laptop \\
  1 & Ana & Mouse \\
  2 & \mal{Bruno} & \mal{\nulo} \\
  3 & Carla & Teclado \\}
\end{columns}
\vspace{0.2cm}
\analogia{«Mi tabla principal es \tbl{cliente}. Quiero conservar \textbf{todos} los clientes y buscar coincidencias en \tbl{pedido}. Si las encuentro, agrego esos datos. Si no, pongo \nulo.»}
\vspace{0.05cm}
\ojo{Una fila de la tabla principal \textbf{se repite} si tiene varias coincidencias: Ana sale \textbf{dos veces} porque tiene dos pedidos.}
\end{frame}

\begin{columns}[c,onlytextwidth]

\midrule

3 & Zoe \\
4 & Ana \\

\bottomrule

\begin{frame}{\texttt{NULL}: el dato que falta}
\begin{columns}[c,onlytextwidth]
\column{.56\textwidth}
\tablaempleado
\column{.40\textwidth}
\small
\begin{itemize}\setlength\itemsep{6pt}
  \item \nulo{} significa \textbf{«no se sabe»} o \textbf{«no hay dato»}.
  \item Bruno y Elena \textbf{no registraron} teléfono.
  \item Diego tiene \texttt{''}: un \textbf{texto vacío}. Parece lo mismo, pero \textbf{no es} \nulo.
\end{itemize}
\end{columns}
\vspace{0.0cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.05}
\begin{tabular}{>{\ttfamily\bfseries\raggedright\arraybackslash}p{2.2cm}>{\raggedright\arraybackslash}p{9.4cm}}
\toprule
\normalfont\bfseries Valor & \bfseries Qué dice \\
\midrule
0 & un número: «tiene cero» \\
'' & un texto sin letras: «lo dejó en blanco» \\
NULL & \textbf{ningún valor}: «no lo sabemos» \\
\bottomrule
\end{tabular}
\end{center}
\vspace{-0.15cm}
\analogia{En un examen: un \textbf{cero} es una nota. Un casillero \textbf{sin nota} es otra cosa: todavía no se sabe cuánto sacó.}
\end{frame}

\begin{frame}{\texttt{IS NULL}: preguntar si falta el dato}
\small \textbf{Pregunta:} ¿qué empleados \textbf{no tienen} teléfono registrado?
\vspace{0.05cm}
\begin{columns}[c,onlytextwidth]
\column{.56\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} nombre, telefono\\
\kw{FROM} empleado\\
\kw{WHERE} telefono \nuevo{\kw{IS NULL}};
\end{sqlcode}
\column{.38\textwidth}
\res{ll}{\cab{nombre} & \cab{telefono}}{%
  Bruno & \nulo \\
  Elena & \nulo \\}
\end{columns}
\vspace{0.2cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item Lo contrario es \texttt{IS NOT NULL}: los que \textbf{sí} tienen un valor.
  \item Diego \textbf{no sale}: su teléfono es \texttt{''}, y eso es un valor.
\end{itemize}
\vspace{0.15cm}
\ojo{\texttt{WHERE telefono = NULL} \textbf{no devuelve nada}. Comparar con algo desconocido nunca da «verdadero». Para los nulos se usa siempre \texttt{IS NULL}.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{COALESCE}: un valor de repuesto}
\begin{columns}[c,onlytextwidth]
\column{.58\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} nombre,\\
\ind \nuevo{\kw{COALESCE}(telefono, 'sin teléfono')} \kw{AS} contacto\\
\kw{FROM} empleado;
\end{sqlcode}
\column{.38\textwidth}
\res{ll}{\cab{nombre} & \cab{contacto}}{%
  Ana & 987111222 \\
  Bruno & \mal{sin teléfono} \\
  Carla & 955333444 \\
  Diego & '' \\
  Elena & \mal{sin teléfono} \\}
\end{columns}
\vspace{0.25cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item \texttt{COALESCE(a, b)} devuelve \texttt{a}; y si \texttt{a} es \nulo, devuelve \texttt{b}.
  \item Acepta más valores: \texttt{COALESCE(a, b, c)} devuelve \textbf{el primero que no sea} \nulo.
  \item Solo cambia lo que \textbf{se muestra}: la tabla sigue guardando \nulo.
\end{itemize}
\vspace{0.1cm}
\analogia{Es el \textbf{suplente}: si el titular no vino, entra el que sigue en la lista.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{NULLIF}: convertir un valor en \texttt{NULL}}
\begin{columns}[c,onlytextwidth]
\column{.58\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} nombre,\\
\ind \nuevo{\kw{NULLIF}(telefono, '')} \kw{AS} telefono\\
\kw{FROM} empleado;
\end{sqlcode}
\column{.38\textwidth}
\res{ll}{\cab{nombre} & \cab{telefono}}{%
  Ana & 987111222 \\
  Bruno & \nulo \\
  Carla & 955333444 \\
  Diego & \mal{\nulo} \\
  Elena & \nulo \\}
\end{columns}
\vspace{0.2cm}
\begin{itemize}\setlength\itemsep{5pt}
  \item \texttt{NULLIF(a, b)} devuelve \nulo{} si \texttt{a} \textbf{es igual} a \texttt{b}. Si no, devuelve \texttt{a}.
  \item Juntos: \texttt{COALESCE(NULLIF(telefono, ''), 'sin teléfono')} y Diego también sale «sin teléfono».
\end{itemize}
\vspace{0.1cm}
\enlarealidad{El uso más común es \textbf{evitar dividir entre cero}: \texttt{total / NULLIF(cantidad, 0)} da \nulo{} en vez de un error.}
\end{frame}

\begin{frame}{\ins{Práctica 2} Valores nulos en la tienda}
\begin{columns}[c,onlytextwidth]

  \item Una \textbf{expresión} combina columnas, valores y operadores: \texttt{+} \ \texttt{-} \ \texttt{*} \ \texttt{/}
  \item El cálculo se hace \textbf{en cada fila}.
  \item \texttt{AS} le pone nombre a la columna nueva.

\begin{frame}{Expresiones con fechas: la antigüedad}
\small \textbf{Pregunta:} ¿cuántos años lleva cada empleado en la empresa?
\vspace{0.05cm}
\begin{columns}[c,onlytextwidth]
\column{.64\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} nombre,\\
\ind \nuevo{\kw{CAST}(strftime('\%Y', 'now') \kw{AS} INTEGER) - anio\_ingreso}\\
\ind \ind \kw{AS} antiguedad\\
\kw{FROM} empleado;
\end{sqlcode}
\column{.32\textwidth}
\res{lr}{\cab{nombre} & \cab{antiguedad}}{%
  Ana & 8 \\
  Bruno & 5 \\
  Carla & 11 \\
  Diego & 3 \\
  Elena & 6 \\}
\end{columns}
\vspace{0.15cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{>{\ttfamily\raggedright\arraybackslash}p{5.2cm}>{\raggedright\arraybackslash}p{7.2cm}}
\toprule
\normalfont\bfseries Pieza & \bfseries Qué hace \\
\midrule
strftime('\%Y', 'now') & el año de hoy, como \textbf{texto}: \texttt{'2026'} \\
CAST(... AS INTEGER) & lo convierte en \textbf{número}: \texttt{2026} \\
- anio\_ingreso & le resta el año en que entró cada uno \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.05cm}
\small Así se escribe en \textbf{SQLite}. En MySQL la misma idea es \texttt{YEAR(CURDATE()) - anio\_ingreso}.
\end{frame}

\begin{frame}{Concatenar: pegar textos con \texttt{||}}
\begin{columns}[c,onlytextwidth]
\column{.60\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} \nuevo{nombre || ' (' || area || ')'} \kw{AS} info\\
\kw{FROM} empleado;
\end{sqlcode}
\column{.34\textwidth}
\res{l}{\cab{info}}{%
  Ana (Ventas) \\
  Bruno (Ventas) \\
  Carla (Sistemas) \\
  Diego (Sistemas) \\
  Elena (Almacén) \\}
\end{columns}
\vspace{0.25cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item \texttt{||} \textbf{pega} un texto a continuación del otro.
  \item Lo que va entre comillas se escribe \textbf{tal cual}: aquí, los paréntesis y el espacio.
  \item También pega números: \texttt{nombre || ' - ' || sueldo} da \texttt{Ana - 2500}.
\end{itemize}
\vspace{0.1cm}
\ojo{Si \textbf{una} de las piezas es \nulo, \textbf{todo} el resultado es \nulo. Se arregla con \texttt{COALESCE(telefono, 'sin teléfono')}.}
\vspace{0.05cm}
\small En MySQL se usa \texttt{CONCAT(nombre, ' (', area, ')')}.
\end{frame}

\begin{frame}{\texttt{CASE WHEN}: decidir fila por fila}
\begin{columns}[c,onlytextwidth]
\column{.56\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} nombre, sueldo,\\
\ind \nuevo{\kw{CASE}}\\
\ind \ind \nuevo{\kw{WHEN} sueldo >= 3000 \kw{THEN} 'alto'}\\
\ind \ind \nuevo{\kw{WHEN} sueldo >= 2000 \kw{THEN} 'medio'}\\
\ind \ind \nuevo{\kw{ELSE} 'bajo'}\\
\ind \nuevo{\kw{END}} \kw{AS} nivel\\
\kw{FROM} empleado;
\end{sqlcode}
\column{.40\textwidth}
\res{lrl}{\cab{nombre} & \cab{sueldo} & \cab{nivel}}{%
  Ana & 2500 & medio \\
  Bruno & 1800 & bajo \\
  Carla & 4200 & alto \\
  Diego & 3100 & alto \\
  Elena & 1500 & bajo \\}
\end{columns}
\vspace{0.2cm}
\begin{itemize}\setlength\itemsep{5pt}
  \item Las condiciones se revisan \textbf{en orden}: se queda con \textbf{la primera} que se cumple.
  \item \texttt{ELSE} atrapa a todos los demás. Sin \texttt{ELSE}, a esos les toca \nulo.
  \item Siempre se cierra con \texttt{END}.
\end{itemize}
\vspace{0.05cm}
\analogia{Es la regla del profesor: «de 18 para arriba, sobresaliente; de 11 para arriba, aprobado; \textbf{si no}, desaprobado».}
\end{frame}

\begin{frame}{Funciones de texto}
\small Una \textbf{función} recibe un valor y devuelve otro. Estas trabajan con \textbf{texto}.
\begin{center}
\small
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{>{\ttfamily\bfseries\raggedright\arraybackslash}p{1.7cm}>{\raggedright\arraybackslash}p{4.3cm}>{\ttfamily\raggedright\arraybackslash}p{4.7cm}>{\ttfamily\raggedright\arraybackslash}p{1.5cm}}
\toprule
\normalfont\bfseries Función & \bfseries Qué hace & \normalfont\bfseries Con \texttt{nombre = 'Carla'} & \normalfont\bfseries Resultado \\
\midrule
UPPER & pasa todo a mayúsculas & UPPER(nombre) & CARLA \\
LOWER & pasa todo a minúsculas & LOWER(nombre) & carla \\
LENGTH & cuenta los caracteres & LENGTH(nombre) & 5 \\
SUBSTR & saca un pedazo: desde dónde y cuántos & SUBSTR(nombre, 1, 3) & Car \\
REPLACE & cambia un texto por otro & REPLACE(nombre, 'a', 'o') & Corlo \\
\bottomrule
\end{tabular}
\end{center}
\vspace{-0.1cm}
\begin{sqlcode}[.80\textwidth]
\scriptsize
\kw{SELECT} \kw{UPPER}(nombre) \kw{AS} nombre, \kw{SUBSTR}(area, 1, 3) \kw{AS} sigla \ \ \com{CARLA | Sis}\\
\kw{FROM} empleado;
\end{sqlcode}
\vspace{-0.1cm}
\small En \texttt{SUBSTR} las posiciones empiezan en \textbf{1}, no en 0.
\end{frame}

\begin{frame}{Funciones numéricas}

\column{.38\textwidth}
\res{lrr}{\cab{nombre} & \cab{sueldo\_diario} & \cab{distancia}}{%

  Diego & 103.33 & 480 \\

\end{columns}
\vspace{0.15cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{>{\ttfamily\bfseries\raggedright\arraybackslash}p{1.5cm}>{\raggedright\arraybackslash}p{5.6cm}>{\ttfamily\raggedright\arraybackslash}p{3.4cm}>{\ttfamily\raggedright\arraybackslash}p{1.5cm}}
\toprule
\normalfont\bfseries Función & \bfseries Qué hace & \normalfont\bfseries Ejemplo & \normalfont\bfseries Resultado \\

\begin{frame}{Agregaciones: muchas filas, un solo valor}
\begin{columns}[c,onlytextwidth]
\column{.54\textwidth}
\tablaempleado
\column{.44\textwidth}
\small
\begin{itemize}\setlength\itemsep{6pt}
  \item Una expresión calcula \textbf{fila por fila}.
  \item Una \textbf{agregación} toma \textbf{toda una columna} y devuelve \textbf{un solo valor}.
\end{itemize}
\end{columns}
\vspace{0.0cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.08}
\begin{tabular}{>{\ttfamily\bfseries\raggedright\arraybackslash}p{1.5cm}>{\raggedright\arraybackslash}p{4.6cm}>{\ttfamily\raggedright\arraybackslash}p{3.4cm}>{\ttfamily\raggedleft\arraybackslash}p{1.7cm}}
\toprule
\normalfont\bfseries Función & \bfseries Qué responde & \normalfont\bfseries Ejemplo & \normalfont\bfseries Resultado \\
\midrule
COUNT & ¿cuántas filas hay? & COUNT(*) & 5 \\
SUM & ¿cuánto suman? & SUM(sueldo) & 13100 \\
AVG & ¿cuál es el promedio? & AVG(sueldo) & 2620.0 \\
MIN & ¿cuál es el menor? & MIN(sueldo) & 1500 \\
MAX & ¿cuál es el mayor? & MAX(sueldo) & 4200 \\
\bottomrule
\end{tabular}
\end{center}
\vspace{-0.1cm}
\begin{sqlcode}[.80\textwidth]
\scriptsize
\kw{SELECT} \kw{COUNT}(*) \kw{AS} cantidad, \kw{AVG}(sueldo) \kw{AS} sueldo\_promedio \kw{FROM} empleado;
\end{sqlcode}
\end{frame}

\begin{frame}{Las agregaciones y los \texttt{NULL}}
\begin{columns}[c,onlytextwidth]
\column{.56\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} \kw{COUNT}(*) \kw{AS} empleados,\\
\ind \nuevo{\kw{COUNT}(telefono)} \kw{AS} con\_telefono\\
\kw{FROM} empleado;
\end{sqlcode}
\column{.40\textwidth}
\res{rr}{\cab{empleados} & \cab{con\_telefono}}{%
  5 & \mal{3} \\}
\end{columns}
\vspace{0.3cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{>{\ttfamily\bfseries\raggedright\arraybackslash}p{3.4cm}>{\raggedright\arraybackslash}p{8.6cm}}
\toprule
\normalfont\bfseries Se escribe & \bfseries Cuenta \\
\midrule
COUNT(*) & \textbf{todas} las filas \\
COUNT(telefono) & solo las filas donde \texttt{telefono} \textbf{no es} \nulo \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.15cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item Bruno y Elena no se cuentan. Diego \textbf{sí}: su \texttt{''} es un valor.
  \item \texttt{SUM}, \texttt{AVG}, \texttt{MIN} y \texttt{MAX} también \textbf{saltan} los \nulo: no los toman como cero.
\end{itemize}
\end{frame}

\begin{frame}{\texttt{GROUP BY}: un resultado por grupo}
\small \textbf{Pregunta:} ¿cuántos empleados hay y cuál es el sueldo promedio \textbf{en cada área}?
\vspace{0.05cm}
\begin{columns}[c,onlytextwidth]
\column{.50\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} area,\\
\ind \kw{COUNT}(*) \kw{AS} cantidad,\\
\ind \kw{AVG}(sueldo) \kw{AS} sueldo\_promedio\\
\kw{FROM} empleado\\
\nuevo{\kw{GROUP BY} area};
\end{sqlcode}
\column{.48\textwidth}
\res{lrr}{\cab{area} & \cab{cantidad} & \cab{sueldo\_promedio}}{%
  Almacén & 1 & 1500.0 \\
  Sistemas & 2 & 3650.0 \\
  Ventas & 2 & 2150.0 \\}
\end{columns}
\vspace{0.2cm}
\begin{enumerate}\setlength\itemsep{5pt}
  \item Se \textbf{juntan} las filas que tienen la misma \texttt{area}: salen tres grupos.
  \item La agregación se calcula \textbf{dentro de cada grupo}.
  \item El resultado tiene \textbf{una fila por grupo}.
\end{enumerate}
\vspace{0.1cm}
\ojo{En el \texttt{SELECT} solo van las columnas del \texttt{GROUP BY} y las agregaciones. Pedir también \texttt{nombre} no tiene sentido: ¿el de cuál de los dos de Ventas?}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{HAVING}: filtrar los grupos}
\small \textbf{Pregunta:} ¿qué áreas tienen \textbf{dos o más} empleados?
\vspace{0.05cm}
\begin{columns}[c,onlytextwidth]
\column{.56\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} area, \kw{COUNT}(*) \kw{AS} cantidad\\
\kw{FROM} empleado\\
\kw{GROUP BY} area\\
\nuevo{\kw{HAVING} \kw{COUNT}(*) >= 2};
\end{sqlcode}
\column{.38\textwidth}
\res{lr}{\cab{area} & \cab{cantidad}}{%
  Sistemas & 2 \\
  Ventas & 2 \\}
\end{columns}
\vspace{-0.05cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.1}
\begin{tabular}{>{\ttfamily\bfseries\raggedright\arraybackslash}p{1.7cm}>{\raggedright\arraybackslash}p{4.4cm}>{\raggedright\arraybackslash}p{5.8cm}}
\toprule
& \bfseries Qué filtra & \bfseries Cuándo \\
\midrule
WHERE & \textbf{filas} sueltas & \textbf{antes} de agrupar \\
HAVING & \textbf{grupos} completos & \textbf{después} de agrupar \\
\bottomrule
\end{tabular}
\end{center}
\vspace{-0.15cm}
\ojo{\texttt{WHERE COUNT(*) >= 2} da error: cuando actúa el \texttt{WHERE}, los grupos \textbf{todavía no existen}.}
\end{frame}

\begin{frame}{\ins{Práctica 4} Haz tú el \texttt{GROUP BY}, a mano}
