\begin{frame}{Una función en MySQL y en SQL Server}
\small El caso más simple: un cupón de \textbf{porcentaje}.
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlmini}[\col]
\kw{CREATE FUNCTION} aplicar\_porcentaje(\\
\ind\mal{p\_precio} \kw{DECIMAL}(10,2),\\
\ind\mal{p\_porcentaje} \kw{DECIMAL}(5,2))\\
\kw{RETURNS DECIMAL}(10,2)\\
\kw{DETERMINISTIC}\\
\kw{RETURN} p\_precio - p\_precio * p\_porcentaje / 100;\\[6pt]
\kw{SELECT} aplicar\_porcentaje(120, 10);\ \com{-- 108.00}
\end{sqlmini}
\column{.5\textwidth}
\centering\etiqueta{SQL Server}
\begin{sqlmini}[\col]
\kw{CREATE} \mal{\kw{OR ALTER}} \kw{FUNCTION} \mal{dbo.}aplicar\_porcentaje(\\
\ind\mal{@precio} \kw{DECIMAL}(10,2),\\
\ind\mal{@porcentaje} \kw{DECIMAL}(5,2))\\
\kw{RETURNS DECIMAL}(10,2)\\
\mal{\kw{AS}}\\
\mal{\kw{BEGIN}}\\
\ind\kw{RETURN} @precio - @precio * @porcentaje / 100;\\
\mal{\kw{END}};\\
\mal{\kw{GO}}\\[4pt]
\kw{SELECT} \mal{dbo.}aplicar\_porcentaje(120, 10);\ \com{-- 108.00}
\end{sqlmini}
\end{columns}
\vspace{0.1cm}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.1}
\begin{tabular}{@{}>{\bfseries}lll@{}}
\toprule
 & \bfseries MySQL & \bfseries SQL Server \\
\midrule
Parámetros & \texttt{p\_precio} (\texttt{p\_}: costumbre) & \texttt{@precio}: la \texttt{@} es obligatoria \\
Cuerpo & \texttt{RETURN \ldots;} & entre \texttt{AS BEGIN} y \texttt{END} \\
Para usarla & \texttt{aplicar\_porcentaje(\ldots)} & \texttt{dbo.aplicar\_porcentaje(\ldots)} \\

\begin{frame}{Usarla sobre las filas de una tabla}
\small En el panel: con una compra de S/ 120, ¿cuánto pagaría el cliente con cada cupón de porcentaje?
\begin{columns}[c,onlytextwidth]
\column{.55\textwidth}

\begin{frame}{Variables: guardar un valor para usarlo después}
\concepto{Variable}{\small Un \textbf{nombre} que guarda \textbf{un valor} mientras se ejecuta tu código. Se le da un valor con \texttt{SET}.}
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlchico}[\col]
\kw{SET} @precio = 120;\\
\kw{SET} @descuento = 20;\\
\kw{SELECT} @precio - @descuento \kw{AS} a\_pagar;\\
\com{-- 100}
\end{sqlchico}
\footnotesize Guardar lo que devuelve un \texttt{SELECT}:
\begin{sqlchico}[\col]
\kw{SELECT} valor\_descuento \dentro{\kw{INTO} @descuento}\\
\kw{FROM} cupon \kw{WHERE} codigo = 'NOVIEMBRE20';
\end{sqlchico}
\column{.48\textwidth}
\centering\etiqueta{SQL Server}
\begin{sqlchico}[\col]
\kw{DECLARE} @precio \kw{DECIMAL}(10,2) = 120;\\
\kw{DECLARE} @descuento \kw{DECIMAL}(10,2) = 20;\\
\kw{SELECT} @precio - @descuento \kw{AS} a\_pagar;\\
\com{-- 100.00}
\end{sqlchico}
\footnotesize Guardar lo que devuelve un \texttt{SELECT}:
\begin{sqlchico}[\col]
\kw{SELECT} \dentro{@descuento = valor\_descuento}\\
\kw{FROM} cupon \kw{WHERE} codigo = 'NOVIEMBRE20';
\end{sqlchico}
\end{columns}
\ojo{Dentro de una función de MySQL, las variables se crean con \texttt{DECLARE} y van \textbf{sin} \texttt{@}: \texttt{DECLARE v\_tipo VARCHAR(10);}}
\end{frame}

\begin{frame}{\texttt{IF}: elegir un camino}
\begin{columns}[c,onlytextwidth]
\column{.42\textwidth}
\centering
\begin{tikzpicture}
  \node[decision] (d) at (0,0) {¿el tipo es\\'porcentaje'?};
  \node[paso,text width=2.4cm] (si) at (-1.65,-1.9) {descuento =\\precio $\times$ valor / 100};
  \node[paso,text width=2.1cm] (no) at (1.65,-1.9) {descuento =\\valor};
  \draw[flecha] (d.west) -| node[above,pos=.25,font=\scriptsize\bfseries]{sí} (si.north);
  \draw[flecha] (d.east) -| node[above,pos=.25,font=\scriptsize\bfseries]{no} (no.north);
\end{tikzpicture}\\[8pt]
\small Se ejecuta \textbf{solo uno} de los dos caminos.
\column{.55\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlcode}[\col]
\kw{IF} v\_tipo = 'porcentaje' \mal{\kw{THEN}}\\
\ind\kw{SET} v\_descuento = p\_precio * v\_valor / 100;\\
\kw{ELSE}\\
\ind\kw{SET} v\_descuento = v\_valor;\\
\mal{\kw{END IF}};
\end{sqlcode}
\centering\etiqueta{SQL Server}
\begin{sqlcode}[\col]
\kw{IF} @tipo = 'porcentaje'\\
\ind\kw{SET} @descuento = @precio * @valor / 100;\\
\kw{ELSE}\\
\ind\kw{SET} @descuento = @valor;
\end{sqlcode}
\end{columns}
\analogia{En SQL Server no hay \texttt{THEN} ni \texttt{END IF}. Si un camino tiene \textbf{varias} instrucciones, van entre \texttt{BEGIN} y \texttt{END}.}
\end{frame}

\begin{frame}{Todo junto: \texttt{precio\_final} en MySQL}
\begin{columns}[c,onlytextwidth]
\column{.58\textwidth}
\begin{sqlmicro}[\col]
\dentro{\kw{DELIMITER} \$\$}\\
\kw{CREATE FUNCTION} precio\_final(p\_precio \kw{DECIMAL}(10,2),\\
\ind\ind\ind\ind\ind\ind\ind\ind\ind\ind p\_cupon\_id \kw{INT})\\
\kw{RETURNS DECIMAL}(10,2)\\
\kw{READS SQL DATA}\\
\kw{BEGIN}\\
\ind\kw{DECLARE} v\_tipo \kw{VARCHAR}(10);\\
\ind\kw{DECLARE} v\_valor \kw{DECIMAL}(10,2);\\
\ind\kw{DECLARE} v\_descuento \kw{DECIMAL}(10,2);\\[1.5pt]
\ind\kw{SELECT} tipo\_descuento, valor\_descuento \kw{INTO} v\_tipo, v\_valor\\
\ind\kw{FROM} cupon \kw{WHERE} id = p\_cupon\_id;\hfill\npieza{1}\\[1.5pt]
\ind\kw{IF} v\_tipo = 'porcentaje' \kw{THEN}\\
\ind\ind\kw{SET} v\_descuento = p\_precio * v\_valor / 100;\\
\ind\kw{ELSE}\\
\ind\ind\kw{SET} v\_descuento = v\_valor;\\
\ind\kw{END IF};\hfill\npieza{2}\\[1.5pt]
\ind\kw{IF} v\_descuento > p\_precio \kw{THEN}\\
\ind\ind\kw{SET} v\_descuento = p\_precio;\\
\ind\kw{END IF};\\[1.5pt]
\ind\kw{RETURN} p\_precio - v\_descuento;\hfill\npieza{3}\\
\kw{END}\dentro{\$\$}\\
\dentro{\kw{DELIMITER} ;}
\end{sqlmicro}
\column{.39\textwidth}
\raggedright\footnotesize
\textbf{\texttt{BEGIN} \ldots{} \texttt{END}}: agrupa varias instrucciones, cada una con su \texttt{;}.\\[5pt]
\textbf{\texttt{DELIMITER \$\$}}: le dice a MySQL que la función termina en \texttt{\$\$}, no en el primer \texttt{;}.\\[5pt]
\textbf{\texttt{READS SQL DATA}}: la función lee una tabla.\\[5pt]
El segundo \texttt{IF} evita que se pague menos de 0.
\vspace{0.2cm}
\begin{sqlmini}[\col]
\kw{SELECT} precio\_final(120, 1),\ \com{-- 108.00}\\
\ind\ind\ind precio\_final(120, 2),\ \com{-- 100.00}\\
\ind\ind\ind precio\_final(30, 5);\ \ \ \com{-- \ \ 0.00}
\end{sqlmini}
{\scriptsize\gris{Cupones 1, 2 y 5. Con \texttt{PRUEBA50} el descuento (50) es mayor que el precio (30): paga 0.}}
\end{columns}
\end{frame}

\begin{frame}{\texttt{IF} con varios caminos: \texttt{ELSEIF}}
\renewcommand{\arraystretch}{0.85}
\small ¿En qué situación está cada cupón? Se revisa \textbf{en orden}: gana la primera condición que se cumple.
\begin{columns}[T,onlytextwidth]
\column{.5\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlmini}[\col]
\kw{IF} v\_estado = 'inactivo' \kw{THEN}\\
\ind\kw{SET} v\_situacion = 'inactivo';\\
\dentro{\kw{ELSEIF}} p\_fecha < v\_desde \kw{THEN}\\
\ind\kw{SET} v\_situacion = 'por empezar';\\
\dentro{\kw{ELSEIF}} v\_hasta \kw{IS NOT NULL}\\
\ind\ind\kw{AND} p\_fecha >= v\_hasta \kw{THEN}\\
\ind\kw{SET} v\_situacion = 'vencido';\\
\kw{ELSE}\\
\ind\kw{SET} v\_situacion = 'vigente';\\
\kw{END IF};
\end{sqlmini}
\centering\etiqueta{SQL Server}
\begin{sqlmini}[\col]
\kw{IF} @estado = 'inactivo' \ldots\\
\dentro{\kw{ELSE IF}} @fecha < @desde \ldots\\
\dentro{\kw{ELSE IF}} @hasta \kw{IS NOT NULL AND} \ldots\\
\kw{ELSE} \ldots
\end{sqlmini}
\column{.47\textwidth}
\begin{sqlmini}[\col]
\kw{SELECT} codigo,\\
\ind situacion\_cupon(id, '2026-11-10') \kw{AS} situacion\\
\kw{FROM} cupon \kw{ORDER BY} id;
\end{sqlmini}
\res{ll}{\cab{codigo} & \cab{situacion}}{BIENVENIDA10 & vigente \\ NOVIEMBRE20 & vigente \\ VERANO15 & vencido \\ NAVIDAD25 & por empezar \\ PRUEBA50 & inactivo \\}
\par\vspace{4pt}
{\centering\normalfont\scriptsize\color{utpGris}La fecha es fija para que todos veamos lo mismo.\\En la aplicación se usa la fecha de hoy.\par}
\end{columns}
\end{frame}

\seccion{Procedimientos almacenados}
\begin{frame}{Desactivar un cupón no es un cálculo}
\small El responsable de promociones quiere \textbf{desactivar} \texttt{VERANO15}. No hay nada que calcular: hay que \textbf{cambiar} una fila.
\vspace{0.3cm}
\begin{center}
\begin{tikzpicture}
  \tikzset{q/.style={caja,text width=5.3cm,minimum height=2.3cm,font=\small,align=left,anchor=north}}
  \node[q] (f) at (0,0) {\textbf{Función}\\[4pt]
    \footnotesize Recibe datos, \textbf{calcula} y \textbf{devuelve un valor}.\\[4pt]
    Se usa \textbf{dentro} de un \texttt{SELECT}:\\
    {\ttfamily\scriptsize \kw{SELECT} precio\_final(120, 1);}};
  \node[q,cajanaranja] (p) at (6.2,0) {\textbf{Procedimiento almacenado}\\[4pt]
    \footnotesize \textbf{Hace algo}: cambia datos (\texttt{UPDATE}, \texttt{INSERT}\ldots) o muestra un resultado.\\[4pt]
    Se ejecuta \textbf{solo}, con \texttt{CALL} o \texttt{EXEC}:\\
    {\ttfamily\scriptsize \kw{CALL} desactivar\_cupon('VERANO15');}};
\end{tikzpicture}
\end{center}
\vspace{0.3cm}
\concepto{Procedimiento almacenado}{\small Un grupo de instrucciones con \textbf{nombre}, guardado \textbf{en la base}. Se ejecuta con \texttt{CALL} (MySQL) o \texttt{EXEC} (SQL Server).}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Un procedimiento en MySQL y en SQL Server}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlmini}[\col]
\kw{CREATE} \dentro{\kw{PROCEDURE}} desactivar\_cupon(\\
\ind p\_codigo \kw{VARCHAR}(255))\\
\kw{UPDATE} cupon \kw{SET} estado = 'inactivo'\\
\kw{WHERE} codigo = p\_codigo;\\[6pt]
\dentro{\kw{CALL}} desactivar\_cupon('VERANO15');
\end{sqlmini}
\column{.5\textwidth}
\centering\etiqueta{SQL Server}
\begin{sqlmini}[\col]
\kw{CREATE OR ALTER} \dentro{\kw{PROCEDURE}} dbo.desactivar\_cupon(\\
\ind @codigo \kw{NVARCHAR}(255))\\
\kw{AS}\\
\kw{BEGIN}\\
\ind\kw{UPDATE} cupon \kw{SET} estado = 'inactivo'\\
\ind\kw{WHERE} codigo = @codigo;\\
\kw{END};\\
\kw{GO}\\[4pt]
\dentro{\kw{EXEC}} dbo.desactivar\_cupon 'VERANO15';
\end{sqlmini}
\end{columns}
\vspace{0.1cm}
\begin{columns}[c,onlytextwidth]
\column{.55\textwidth}
\footnotesize
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}>{\bfseries}lll@{}}
\toprule
 & \bfseries MySQL & \bfseries SQL Server \\
\midrule
Crear & \texttt{CREATE PROCEDURE} & \texttt{CREATE OR ALTER PROCEDURE} \\
Ejecutar & \texttt{CALL nombre(\ldots);} & \texttt{EXEC dbo.nombre \ldots;} \\
\bottomrule
\end{tabular}
\column{.42\textwidth}
\centering
\begin{sqlmini}[\col]
\kw{SELECT} codigo, estado \kw{FROM} cupon\\
\kw{WHERE} codigo = 'VERANO15';
\end{sqlmini}
\vspace{-6pt}
\res{ll}{\cab{codigo} & \cab{estado}}{VERANO15 & inactivo \\}
\end{columns}
\vspace{0.15cm}
{\centering\scriptsize\gris{No devuelve nada: solo cambia la fila. En SQL Server los parámetros van entre \texttt{EXEC} y \texttt{;}, sin paréntesis.}\par}
\end{frame}

\begin{frame}{Con variables e \texttt{IF}: avisar si el cupón no existe}
\small Con un código mal escrito (\texttt{'VERANO1'}), el \texttt{UPDATE} no cambia nada y \textbf{no avisa}.
\begin{columns}[T,onlytextwidth]
\column{.58\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlmini}[\col]
\kw{DELIMITER} \$\$\\
\kw{CREATE PROCEDURE} desactivar\_cupon(p\_codigo \kw{VARCHAR}(255))\\
\kw{BEGIN}\\
\ind\kw{DECLARE} v\_cantidad \kw{INT};\\[1.5pt]
\ind\kw{SELECT} \kw{COUNT}(*) \kw{INTO} v\_cantidad\\
\ind\kw{FROM} cupon \kw{WHERE} codigo = p\_codigo;\hfill\npieza{1}\\[1.5pt]
\ind\dentro{\kw{IF} v\_cantidad = 0 \kw{THEN}}\\
\ind\ind\kw{SELECT} 'El cupón no existe' \kw{AS} mensaje;\\
\ind\kw{ELSE}\\
\ind\ind\kw{UPDATE} cupon \kw{SET} estado = 'inactivo'\\
\ind\ind\kw{WHERE} codigo = p\_codigo;\\
\ind\ind\kw{SELECT} 'Cupón desactivado' \kw{AS} mensaje;\\
\ind\kw{END IF};\hfill\npieza{2}\\
\kw{END}\$\$\\
\kw{DELIMITER} ;
\end{sqlmini}
\column{.39\textwidth}
\footnotesize\raggedright
\npieza{1} Cuenta los cupones con ese código.\\[3pt]
\npieza{2} Si es 0, avisa. Si no, lo desactiva.
\vspace{0.15cm}
\begin{sqlmini}[\col]
\kw{CALL} desactivar\_cupon('VERANO1');
\end{sqlmini}
\vspace{-8pt}
\res{l}{\cab{mensaje}}{El cupón no existe \\}
\vspace{0pt}
\begin{sqlmini}[\col]
\kw{CALL} desactivar\_cupon('VERANO15');
\end{sqlmini}
\vspace{-8pt}
\res{l}{\cab{mensaje}}{Cupón desactivado \\}
\end{columns}
\vspace{0.1cm}
{\centering\scriptsize\gris{En SQL Server es igual, con \texttt{@cantidad}. El \texttt{ELSE} tiene dos instrucciones: van entre \texttt{BEGIN} y \texttt{END}.}\par}
\end{frame}

\begin{frame}{Un procedimiento que muestra un resultado}
\small La tienda pregunta: con una compra de S/ 120, ¿cuánto pagaría con cada cupón \textbf{vigente}?
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlmini}[\col]
\kw{CREATE PROCEDURE} cupones\_vigentes(\\
\ind p\_precio \kw{DECIMAL}(10,2), p\_fecha \kw{DATETIME})\\
\kw{SELECT} codigo,\\
\ind\dentro{precio\_final(p\_precio, id)} \kw{AS} a\_pagar\\
\kw{FROM} cupon\\
\kw{WHERE} \dentro{situacion\_cupon(id, p\_fecha)} = 'vigente'\\
\kw{ORDER BY} id;\\[6pt]
\kw{CALL} cupones\_vigentes(120, '2026-11-10');
\end{sqlmini}
\column{.48\textwidth}
\centering\etiqueta{SQL Server}
\begin{sqlmicro}[\col]
\kw{CREATE OR ALTER PROCEDURE} dbo.cupones\_vigentes(\\
\ind @precio \kw{DECIMAL}(10,2), @fecha \kw{DATETIME2})\\
\kw{AS}\\
\kw{BEGIN}\\
\ind\kw{SELECT} codigo,\\
\ind\ind dbo.precio\_final(@precio, id) \kw{AS} a\_pagar\\
\ind\kw{FROM} cupon\\
\ind\kw{WHERE} dbo.situacion\_cupon(id, @fecha) = 'vigente'\\
\ind\kw{ORDER BY} id;\\
\kw{END};\\
\kw{GO}\\[3pt]
\kw{EXEC} dbo.cupones\_vigentes 120, '2026-11-10';
\end{sqlmicro}
\end{columns}
\begin{columns}[c,onlytextwidth]
\column{.35\textwidth}
\res{lr}{\cab{codigo} & \cab{a\_pagar}}{BIENVENIDA10 & 108.00 \\ NOVIEMBRE20 & 100.00 \\}
\column{.62\textwidth}
\concepto{Idea clave}{\small El procedimiento \textbf{usa las funciones} de antes. La tienda ya no escribe la consulta: solo ejecuta \texttt{cupones\_vigentes}.}
\end{columns}
\end{frame}

\begin{frame}{¿Función o procedimiento?}
\vspace{0.1cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{@{}>{\raggedright\bfseries}p{2.6cm}>{\raggedright}p{4.9cm}>{\raggedright\arraybackslash}p{5cm}@{}}
\toprule
 & \bfseries Función & \bfseries Procedimiento almacenado \\
\midrule
Para qué & \textbf{calcular} un valor & \textbf{hacer} algo: cambiar datos o mostrar un resultado \\
Devuelve & un valor, con \texttt{RETURN} & nada, o las filas de sus \texttt{SELECT} \\
Cómo se usa & dentro de un \texttt{SELECT} & solo: \texttt{CALL} o \texttt{EXEC} \\

\bottomrule
\end{tabular}
