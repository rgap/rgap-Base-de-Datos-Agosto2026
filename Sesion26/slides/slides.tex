\begin{frame}{Revisión: ¿qué hace \texttt{OVER}?}
\renewcommand{\arraystretch}{0.85}
\small Los 3 pagos de hoy y su promedio, de dos formas:
\vspace{0.2cm}
\begin{columns}[T,onlytextwidth]
\column{.42\textwidth}
\centering\etiqueta{Sin OVER}
\begin{sqlchico}[\col]
\kw{SELECT AVG}(monto) \kw{AS} promedio\\
\kw{FROM} pago \kw{WHERE} id >= 6;
\end{sqlchico}
\resg{r}{\cab{promedio}}{35.000000 \\}
\vspace{6pt}
\small Junta las filas: queda \textbf{1}.
\column{.54\textwidth}
\centering\etiqueta{Con OVER}
\begin{sqlchico}[\col]
\kw{SELECT} id, monto, \dentro{\kw{AVG}(monto) \kw{OVER} ()} \kw{AS} promedio\\
\kw{FROM} pago \kw{WHERE} id >= 6 \kw{ORDER BY} id;
\end{sqlchico}
\resg{rrr}{\cab{id} & \cab{monto} & \cab{promedio}}{6 & 15.00 & \cellcolor{yellow!45}35.000000 \\ 7 & 35.00 & \cellcolor{yellow!45}35.000000 \\ 8 & 55.00 & \cellcolor{yellow!45}35.000000 \\}
\vspace{6pt}
\small No junta: quedan las \textbf{3}, cada una con el promedio al lado.
\end{columns}
\vspace{0.3cm}
\concepto{\texttt{OVER}}{\small Calcula sobre varias filas \textbf{sin juntarlas}.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Revisión: lo que va dentro de \texttt{OVER}}
\renewcommand{\arraystretch}{0.85}
\begin{center}
\small
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}ll@{}}
\toprule
\bfseries Se escribe & \bfseries Calcula sobre \\
\midrule
\texttt{OVER ()} & todas las filas \\
\texttt{OVER (PARTITION BY metodo)} & cada grupo por separado \\
\texttt{OVER (ORDER BY id)} & las filas en orden, sumando lo anterior \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.1cm}
\begin{columns}[c,onlytextwidth]
\column{.55\textwidth}
\begin{sqlchico}[\col]
\kw{SELECT} id, monto,\\
\ind\dentro{\kw{SUM}(monto) \kw{OVER} (\kw{ORDER BY} id)} \kw{AS} acumulado\\
\kw{FROM} pago \kw{WHERE} id >= 6 \kw{ORDER BY} id;
\end{sqlchico}
\column{.42\textwidth}
\centering
\tabla{resultado}{rrr}{\cab{id} & \cab{monto} & \cab{acumulado}}{6 & 15.00 & 15.00 \\ 7 & 35.00 & 50.00 \\ 8 & 55.00 & 105.00 \\}\\[4pt]
{\footnotesize\gris{15, luego 15 + 35, luego 50 + 55}}
\end{columns}
\vspace{0.2cm}
{\centering\small También van con \texttt{OVER}: \texttt{ROW\_NUMBER}, \texttt{RANK}, \texttt{DENSE\_RANK}, \texttt{LAG} y \texttt{LEAD}.\par}
\end{frame}

\begin{frame}{Variables: guardar un valor para usarlo después}
\concepto{Variable}{\small Un \textbf{nombre} que guarda \textbf{un valor} mientras se ejecuta tu código. Se le da un valor con \texttt{SET}.}
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlchico}[\col]
\kw{SET} @precio = 120;\\
\kw{SET} @descuento = 20;\\
\kw{SELECT} @precio - @descuento \kw{AS} a\_pagar;\\
\com{-- 100}
\end{sqlchico}
\footnotesize Guardar lo que devuelve un \texttt{SELECT}:
\begin{sqlchico}[\col]
\kw{SELECT} valor\_descuento \dentro{\kw{INTO} @descuento}\\
\kw{FROM} cupon \kw{WHERE} codigo = 'NOVIEMBRE20';
\end{sqlchico}
\column{.48\textwidth}
\centering\etiqueta{SQL Server}
\begin{sqlchico}[\col]
\kw{DECLARE} @precio \kw{DECIMAL}(10,2) = 120;\\
\kw{DECLARE} @descuento \kw{DECIMAL}(10,2) = 20;\\
\kw{SELECT} @precio - @descuento \kw{AS} a\_pagar;\\
\com{-- 100.00}
\end{sqlchico}
\footnotesize Guardar lo que devuelve un \texttt{SELECT}:
\begin{sqlchico}[\col]
\kw{SELECT} \dentro{@descuento = valor\_descuento}\\
\kw{FROM} cupon \kw{WHERE} codigo = 'NOVIEMBRE20';
\end{sqlchico}
\end{columns}
\ojo{Dentro de una función de MySQL, las variables se crean con \texttt{DECLARE} y van \textbf{sin} \texttt{@}: \texttt{DECLARE v\_tipo VARCHAR(10);}}
\end{frame}

\begin{frame}{\texttt{IF}: elegir un camino}
\begin{columns}[c,onlytextwidth]
\column{.42\textwidth}
\centering
\begin{tikzpicture}
  \node[decision] (d) at (0,0) {¿el tipo es\\'porcentaje'?};
  \node[paso,text width=2.4cm] (si) at (-1.65,-1.9) {descuento =\\precio $\times$ valor / 100};
  \node[paso,text width=2.1cm] (no) at (1.65,-1.9) {descuento =\\valor};
  \draw[flecha] (d.west) -| node[above,pos=.25,font=\scriptsize\bfseries]{sí} (si.north);
  \draw[flecha] (d.east) -| node[above,pos=.25,font=\scriptsize\bfseries]{no} (no.north);
\end{tikzpicture}\\[8pt]
\small Se ejecuta \textbf{solo uno} de los dos caminos.
\column{.55\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlcode}[\col]
\kw{IF} v\_tipo = 'porcentaje' \mal{\kw{THEN}}\\
\ind\kw{SET} v\_descuento = p\_precio * v\_valor / 100;\\
\kw{ELSE}\\
\ind\kw{SET} v\_descuento = v\_valor;\\
\mal{\kw{END IF}};
\end{sqlcode}
\centering\etiqueta{SQL Server}
\begin{sqlcode}[\col]
\kw{IF} @tipo = 'porcentaje'\\
\ind\kw{SET} @descuento = @precio * @valor / 100;\\
\kw{ELSE}\\
\ind\kw{SET} @descuento = @valor;
\end{sqlcode}
\end{columns}
\analogia{En SQL Server no hay \texttt{THEN} ni \texttt{END IF}. Si un camino tiene \textbf{varias} instrucciones, van entre \texttt{BEGIN} y \texttt{END}.}
\end{frame}

\begin{frame}{Todo junto: \texttt{precio\_final} en MySQL}
\begin{columns}[c,onlytextwidth]
\column{.58\textwidth}
\begin{sqlmicro}[\col]
\dentro{\kw{DELIMITER} \$\$}\\
\kw{CREATE FUNCTION} precio\_final(p\_precio \kw{DECIMAL}(10,2),\\
\ind\ind\ind\ind\ind\ind\ind\ind\ind\ind p\_cupon\_id \kw{INT})\\
\kw{RETURNS DECIMAL}(10,2)\\
\kw{READS SQL DATA}\\
\kw{BEGIN}\\
\ind\kw{DECLARE} v\_tipo \kw{VARCHAR}(10);\\
\ind\kw{DECLARE} v\_valor \kw{DECIMAL}(10,2);\\
\ind\kw{DECLARE} v\_descuento \kw{DECIMAL}(10,2);\\[1.5pt]
\ind\kw{SELECT} tipo\_descuento, valor\_descuento \kw{INTO} v\_tipo, v\_valor\\
\ind\kw{FROM} cupon \kw{WHERE} id = p\_cupon\_id;\hfill\npieza{1}\\[1.5pt]
\ind\kw{IF} v\_tipo = 'porcentaje' \kw{THEN}\\
\ind\ind\kw{SET} v\_descuento = p\_precio * v\_valor / 100;\\
\ind\kw{ELSE}\\
\ind\ind\kw{SET} v\_descuento = v\_valor;\\
\ind\kw{END IF};\hfill\npieza{2}\\[1.5pt]
\ind\kw{IF} v\_descuento > p\_precio \kw{THEN}\\
\ind\ind\kw{SET} v\_descuento = p\_precio;\\
\ind\kw{END IF};\\[1.5pt]
\ind\kw{RETURN} p\_precio - v\_descuento;\hfill\npieza{3}\\
\kw{END}\dentro{\$\$}\\
\dentro{\kw{DELIMITER} ;}
\end{sqlmicro}
\column{.39\textwidth}
\raggedright\footnotesize
\textbf{\texttt{BEGIN} \ldots{} \texttt{END}}: agrupa varias instrucciones, cada una con su \texttt{;}.\\[5pt]
\textbf{\texttt{DELIMITER \$\$}}: le dice a MySQL que la función termina en \texttt{\$\$}, no en el primer \texttt{;}.\\[5pt]
\textbf{\texttt{READS SQL DATA}}: la función lee una tabla.\\[5pt]
El segundo \texttt{IF} evita que se pague menos de 0.
\vspace{0.2cm}
\begin{sqlmini}[\col]
\kw{SELECT} precio\_final(120, 1),\ \com{-- 108.00}\\
\ind\ind\ind precio\_final(120, 2),\ \com{-- 100.00}\\
\ind\ind\ind precio\_final(30, 5);\ \ \ \com{-- \ \ 0.00}
\end{sqlmini}
{\scriptsize\gris{Cupones 1, 2 y 5. Con \texttt{PRUEBA50} el descuento (50) es mayor que el precio (30): paga 0.}}
\end{columns}
\end{frame}

\begin{frame}{\texttt{IF} con varios caminos: \texttt{ELSEIF}}
\renewcommand{\arraystretch}{0.85}
\small ¿En qué situación está cada cupón? Se revisa \textbf{en orden}: gana la primera condición que se cumple.
\begin{columns}[T,onlytextwidth]
\column{.5\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlmini}[\col]
\kw{IF} v\_estado = 'inactivo' \kw{THEN}\\
\ind\kw{SET} v\_situacion = 'inactivo';\\
\dentro{\kw{ELSEIF}} p\_fecha < v\_desde \kw{THEN}\\
\ind\kw{SET} v\_situacion = 'por empezar';\\
\dentro{\kw{ELSEIF}} v\_hasta \kw{IS NOT NULL}\\
\ind\ind\kw{AND} p\_fecha >= v\_hasta \kw{THEN}\\
\ind\kw{SET} v\_situacion = 'vencido';\\
\kw{ELSE}\\
\ind\kw{SET} v\_situacion = 'vigente';\\
\kw{END IF};
\end{sqlmini}
\centering\etiqueta{SQL Server}
\begin{sqlmini}[\col]
\kw{IF} @estado = 'inactivo' \ldots\\
\dentro{\kw{ELSE IF}} @fecha < @desde \ldots\\
\dentro{\kw{ELSE IF}} @hasta \kw{IS NOT NULL AND} \ldots\\
\kw{ELSE} \ldots
\end{sqlmini}
\column{.47\textwidth}
\begin{sqlmini}[\col]
\kw{SELECT} codigo,\\
\ind situacion\_cupon(id, '2026-11-10') \kw{AS} situacion\\
\kw{FROM} cupon \kw{ORDER BY} id;
\end{sqlmini}
\res{ll}{\cab{codigo} & \cab{situacion}}{BIENVENIDA10 & vigente \\ NOVIEMBRE20 & vigente \\ VERANO15 & vencido \\ NAVIDAD25 & por empezar \\ PRUEBA50 & inactivo \\}

{\centering\normalfont\scriptsize\color{utpGris}La fecha es fija para que todos veamos lo mismo.\\En la aplicación se usa la fecha de hoy.\par}
\end{columns}
\end{frame}

\begin{frame}{Lo que viene: procedimientos almacenados y triggers}
\small La sintaxis de hoy no sirve solo para funciones. Con ella se programan las dos cosas que vienen:

\begin{center}
\begin{tikzpicture}
  \node[caja,align=left,font=\scriptsize,text width=3.7cm] (s) at (0,0) {%

    \texttt{SELECT} \ldots{} \texttt{INTO}\\

Transacción & Varias instrucciones que se guardan \textbf{todas o ninguna}. \\
Empezar & MySQL: \texttt{START TRANSACTION}; SQL Server y SQLite: \texttt{BEGIN TRANSACTION}. \\
\texttt{COMMIT} & Confirma y guarda los cambios. \\
\texttt{ROLLBACK} & Deshace los cambios: todo vuelve a como estaba. \\
Función del usuario & Tu propio cálculo, guardado en la base con \texttt{CREATE FUNCTION}. \\
Variable & Un nombre que guarda un valor: \texttt{DECLARE}, \texttt{SET}, \texttt{SELECT \ldots{} INTO}. \\
\texttt{IF} & Elige un camino. MySQL: \texttt{ELSEIF} y \texttt{END IF}; SQL Server: \texttt{ELSE IF}. \\
