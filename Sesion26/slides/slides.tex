\begin{frame}{Revisión: ¿qué hace \texttt{OVER}?}
\renewcommand{\arraystretch}{0.85}
\small Los 3 pagos de hoy y su promedio, de dos formas:
\vspace{0.2cm}
\begin{columns}[T,onlytextwidth]
\column{.42\textwidth}
\centering\etiqueta{Sin OVER}
\begin{sqlchico}[\col]
\kw{SELECT AVG}(monto) \kw{AS} promedio\\
\kw{FROM} pago \kw{WHERE} id >= 6;
\end{sqlchico}
\resg{r}{\cab{promedio}}{35.000000 \\}
\vspace{6pt}
\small Junta las filas: queda \textbf{1}.
\column{.54\textwidth}
\centering\etiqueta{Con OVER}
\begin{sqlchico}[\col]
\kw{SELECT} id, monto, \dentro{\kw{AVG}(monto) \kw{OVER} ()} \kw{AS} promedio\\
\kw{FROM} pago \kw{WHERE} id >= 6 \kw{ORDER BY} id;
\end{sqlchico}
\resg{rrr}{\cab{id} & \cab{monto} & \cab{promedio}}{6 & 15.00 & \cellcolor{yellow!45}35.000000 \\ 7 & 35.00 & \cellcolor{yellow!45}35.000000 \\ 8 & 55.00 & \cellcolor{yellow!45}35.000000 \\}
\vspace{6pt}
\small No junta: quedan las \textbf{3}, cada una con el promedio al lado.
\end{columns}
\vspace{0.3cm}
\concepto{\texttt{OVER}}{\small Calcula sobre varias filas \textbf{sin juntarlas}.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Revisión: lo que va dentro de \texttt{OVER}}
\renewcommand{\arraystretch}{0.85}
\begin{center}
\small
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{@{}ll@{}}
\toprule
\bfseries Se escribe & \bfseries Calcula sobre \\
\midrule
\texttt{OVER ()} & todas las filas \\
\texttt{OVER (PARTITION BY metodo)} & cada grupo por separado \\
\texttt{OVER (ORDER BY id)} & las filas en orden, sumando lo anterior \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.1cm}
\begin{columns}[c,onlytextwidth]
\column{.55\textwidth}
\begin{sqlchico}[\col]
\kw{SELECT} id, monto,\\
\ind\dentro{\kw{SUM}(monto) \kw{OVER} (\kw{ORDER BY} id)} \kw{AS} acumulado\\
\kw{FROM} pago \kw{WHERE} id >= 6 \kw{ORDER BY} id;
\end{sqlchico}
\column{.42\textwidth}
\centering
\tabla{resultado}{rrr}{\cab{id} & \cab{monto} & \cab{acumulado}}{6 & 15.00 & 15.00 \\ 7 & 35.00 & 50.00 \\ 8 & 55.00 & 105.00 \\}\\[4pt]
{\footnotesize\gris{15, luego 15 + 35, luego 50 + 55}}
\end{columns}
\vspace{0.2cm}
{\centering\small También van con \texttt{OVER}: \texttt{ROW\_NUMBER}, \texttt{RANK}, \texttt{DENSE\_RANK}, \texttt{LAG} y \texttt{LEAD}.\par}
\end{frame}

\begin{frame}{Caso de estudio: venta de entradas para una feria}
\begin{columns}[T,onlytextwidth]
\column{.54\textwidth}
\small La feria \textbf{Fogón 2026} dura tres días (4, 5 y 6 de diciembre) y vende su entrada en línea.
\vspace{0.05cm}
\begin{itemize}\setlength\itemsep{2pt}\footnotesize
  \item Hay \textbf{una sola entrada}, válida para los tres días: S/ 45, con \textbf{500 disponibles}.
  \item El \textbf{visitante} compra en la página principal: indica su \textbf{DNI}, su \textbf{nombre} y su \textbf{correo}, y paga con tarjeta.
  \item Cada DNI compra \textbf{una sola entrada}.
  \item Cada compra \textbf{descuenta una} de las entradas disponibles.
  \item Si el pago no se aprueba, \textbf{no} se registra la compra \textbf{ni} se descuenta la entrada.
  \item El \textbf{administrador} fija el precio y las entradas disponibles, abre o cierra la venta y consulta las ventas.
\end{itemize}

\begin{frame}{¿Y si el segundo cambio falla?}
\small Luis \textbf{ya compró} su entrada: quedan 499. Intenta comprar otra con el mismo DNI. Cada persona compra solo una: \texttt{dni} es \texttt{UNIQUE}.
\vspace{0.2cm}
\begin{sqlcode}[.9\textwidth]
\kw{UPDATE} entrada \kw{SET} disponibles = disponibles - 1 \kw{WHERE} id = 1;\hfill\textcolor{utpRojo}{\checkmark}\\[3pt]
\kw{INSERT INTO} venta (dni, nombre, correo, precio, referencia\_pago)\\
\kw{VALUES} (\mal{'70123456'}, 'Luis Rojas', 'luis.rojas@example.com',\\
\ind 45.00, 'PAGO-0002');\hfill\no\\[3pt]
\scriptsize\err{-- ERROR 1062 (23000): Duplicate entry '70123456' for key 'venta.dni'}
\end{sqlcode}
