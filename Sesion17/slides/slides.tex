\seccion{Unir tablas}
% ---------------------------------------------------------------------
\begin{frame}{Los datos viven en tablas separadas}
\begin{columns}[c,onlytextwidth]
\column{.22\textwidth}
\tablaclientes
\column{.36\textwidth}
\tablapedidos
\column{.38\textwidth}
\small \textbf{Pregunta:} ¿quién pidió el \textbf{Teclado}?
\begin{enumerate}\setlength\itemsep{4pt}
  \item En \tbl{pedidos}: el Teclado tiene \texttt{cliente\_id} = \textbf{2}.
  \item En \tbl{clientes}: el \texttt{id} \textbf{2} es \textbf{Luis}.
\end{enumerate}
\end{columns}
\vspace{0.35cm}
\concepto{La idea}{\small Tú lo resolviste \textbf{mirando dos tablas} y juntando lo que coincide. Eso mismo hace un \texttt{JOIN}, en \textbf{una sola consulta}.}
\vspace{0.1cm}
\analogia{Tienes una hoja con \textbf{códigos y notas}, y otra con \textbf{códigos y nombres}. Las pones lado a lado para saber de quién es cada nota.}
\end{frame}

\begin{frame}{Un \texttt{JOIN}, pieza por pieza}
\begin{center}
\begin{tikzpicture}[y=0.82cm]
  \renewcommand{\piezax}{6.0}
  \pieza{3}{\kw{SELECT} c.nombre, p.producto}{qué columnas quiero, y \textbf{de qué tabla}}
  \pieza{2}{\kw{FROM} clientes c}{la tabla de la \textbf{izquierda}: se escribe primero}
  \pieza{1}{\nuevo{\kw{JOIN}} pedidos p}{la tabla de la \textbf{derecha}: se escribe después}
  \pieza{0}{\nuevo{\kw{ON}} p.cliente\_id = c.id;}{cómo se \textbf{emparejan} las filas}
\end{tikzpicture}
\end{center}
\vspace{0.3cm}
\begin{itemize}\setlength\itemsep{7pt}
  \item \texttt{c} y \texttt{p} son \textbf{apodos} (alias) de las tablas: ahorran escribir el nombre completo.
  \item \texttt{c.nombre} se lee «la columna \texttt{nombre} de \tbl{clientes}».
  \item El \texttt{ON} casi siempre compara la \textbf{clave foránea} con la \textbf{clave primaria}.
\end{itemize}
\end{frame}

\seccion[0]{INNER JOIN}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{INNER JOIN}: solo los que tienen pareja}
\begin{columns}[c,onlytextwidth]
\column{.60\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} c.nombre, p.producto\\
\kw{FROM} clientes c\\
\nuevo{\kw{INNER JOIN}} pedidos p \kw{ON} p.cliente\_id = c.id;
\end{sqlcode}
\vspace{-0.15cm}
\centering\venn{\voff}{\von}{\voff}\\[3pt]
{\scriptsize\color{utpGris}solo lo que está en \textbf{las dos} tablas}
\column{.34\textwidth}
\resultado{\filasinner}
\end{columns}
\vspace{0.35cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item Los que no tienen pareja \textbf{quedan fuera}: no aparecen ni María ni el Monitor.
  \item Ana aparece \textbf{dos veces}: una fila por cada pedido suyo.
  \item Escribir solo \texttt{JOIN} es lo mismo que \texttt{INNER JOIN}. Es el más usado.
\end{itemize}
\vspace{0.15cm}
\analogia{Es una \textbf{fiesta solo para parejas}: el que llega solo, no entra.}
\end{frame}
% ---------------------------------------------------------------------

\seccion{LEFT y RIGHT}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{LEFT JOIN}: toda la tabla izquierda}
\begin{columns}[c,onlytextwidth]
\column{.60\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} c.nombre, p.producto\\
\kw{FROM} clientes c\\
\nuevo{\kw{LEFT JOIN}} pedidos p \kw{ON} p.cliente\_id = c.id;
\end{sqlcode}
\vspace{-0.15cm}
\centering\venn{\von}{\von}{\voff}\\[3pt]
{\scriptsize\color{utpGris}\textbf{toda} la izquierda, tenga pareja o no}
\column{.34\textwidth}
\resultado{\filasinner \mal{María} & \mal{\nulo} \\}
\end{columns}
\vspace{0.35cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item La \textbf{izquierda} es la tabla del \texttt{FROM}: \tbl{clientes}. \textbf{Ninguno} de sus clientes se pierde.
  \item María no tiene pedido: su \texttt{producto} se rellena con \nulo.
  \item El Monitor sigue fuera: está en la tabla de la \textbf{derecha}.
\end{itemize}
\vspace{0.15cm}
\analogia{Es la \textbf{lista de asistencia}: salen todos los alumnos, y al que faltó se le deja el casillero vacío.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{RIGHT JOIN}: toda la tabla derecha}
\begin{columns}[c,onlytextwidth]
\column{.60\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} c.nombre, p.producto\\
\kw{FROM} clientes c\\
\nuevo{\kw{RIGHT JOIN}} pedidos p \kw{ON} p.cliente\_id = c.id;
\end{sqlcode}
\vspace{-0.15cm}
\centering\venn{\voff}{\von}{\von}\\[3pt]
{\scriptsize\color{utpGris}\textbf{toda} la derecha, tenga pareja o no}
\column{.34\textwidth}
\resultado{\filasinner \mal{\nulo} & \mal{Monitor} \\}
\end{columns}
\vspace{0.35cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item La \textbf{derecha} es la tabla del \texttt{JOIN}: \tbl{pedidos}. \textbf{Ningún} pedido se pierde.
  \item El Monitor no tiene cliente: su \texttt{nombre} se rellena con \nulo.
  \item Ahora la que queda fuera es María.
\end{itemize}
\vspace{0.15cm}
\small Es el \textbf{mismo} \texttt{LEFT JOIN}, visto desde el otro lado.
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{LEFT} y \texttt{RIGHT}: lo mismo, al revés}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{Con RIGHT}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{FROM} clientes c\\
\kw{RIGHT JOIN} \nuevo{pedidos p}\\
\ind \kw{ON} p.cliente\_id = c.id;
\end{sqlcode}
\column{.48\textwidth}
\centering\etiqueta{Con LEFT}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{FROM} \nuevo{pedidos p}\\
\kw{LEFT JOIN} clientes c\\
\ind \kw{ON} p.cliente\_id = c.id;
\end{sqlcode}
\end{columns}
\vspace{0.1cm}
\begin{center}
\small Las dos conservan \textbf{todos los pedidos}. Solo cambió \textbf{el orden} en que se escriben las tablas.
\end{center}
\vspace{0.2cm}
\begin{itemize}\setlength\itemsep{7pt}
  \item \textbf{Truco:} pregúntate «¿de qué tabla \textbf{no quiero perder} ninguna fila?».
  \item Pon esa tabla en el \texttt{FROM} y usa \texttt{LEFT JOIN}.
\end{itemize}
\vspace{0.3cm}
\enlarealidad{Casi todos escriben siempre \texttt{LEFT JOIN} y acomodan el orden de las tablas. El \texttt{RIGHT JOIN} se ve muy poco.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Un clásico: encontrar a los que no tienen pareja}
\small \textbf{Pregunta:} ¿qué clientes \textbf{nunca} han hecho un pedido?
\vspace{0.1cm}
\begin{columns}[c,onlytextwidth]
\column{.60\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} c.nombre\\
\kw{FROM} clientes c\\
\kw{LEFT JOIN} pedidos p \kw{ON} p.cliente\_id = c.id\\
\nuevo{\kw{WHERE} p.id \kw{IS NULL};}
\end{sqlcode}
\column{.34\textwidth}
\centering{\scriptsize\color{utpGris}resultado}\\[3pt]
\footnotesize\ttfamily
\begin{tabular}{l}
\toprule
\normalfont\bfseries nombre \\
\midrule
María \\
\bottomrule
\end{tabular}
\end{columns}
\vspace{0.2cm}
\begin{enumerate}\setlength\itemsep{6pt}
  \item El \texttt{LEFT JOIN} trae a \textbf{todos} los clientes; a María le pone \nulo{} en el pedido.
  \item El \texttt{WHERE} se queda \textbf{solo con las filas que tienen ese \nulo}.
\end{enumerate}
\vspace{0.3cm}
\enlarealidad{Así se buscan los alumnos \textbf{sin matrícula}, los productos \textbf{que nadie compró} o los empleados \textbf{sin área}.}
\end{frame}

\seccion{FULL, CROSS y SELF}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{FULL OUTER JOIN}: nadie queda fuera}
\begin{columns}[c,onlytextwidth]
\column{.60\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} c.nombre, p.producto\\
\kw{FROM} clientes c\\
\nuevo{\kw{FULL OUTER JOIN}} pedidos p \kw{ON} p.cliente\_id = c.id;
\end{sqlcode}
\vspace{-0.15cm}
\centering\venn{\von}{\von}{\von}\\[3pt]
{\scriptsize\color{utpGris}\textbf{todo} lo de las dos tablas}
\column{.34\textwidth}
\resultado{\filasinner \mal{María} & \mal{\nulo} \\ \mal{\nulo} & \mal{Monitor} \\}
\end{columns}
\vspace{0.35cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item Es un \texttt{LEFT} y un \texttt{RIGHT} \textbf{juntos}: aparecen María \textbf{y} el Monitor.
  \item A cada uno se le rellena con \nulo{} el lado que le falta.
\end{itemize}
\vspace{0.2cm}
\ojo{\textbf{MySQL no tiene} \texttt{FULL OUTER JOIN}. Ahí se arma juntando un \texttt{LEFT JOIN} y un \texttt{RIGHT JOIN} con \texttt{UNION}.}
\end{frame}

\normalfont\bfseries talla & \normalfont\bfseries color \\
\midrule
S & rojo \\ S & azul \\ S & negro \\
M & rojo \\ M & azul \\ M & negro \\
\bottomrule
\end{tabular}
\end{columns}
\vspace{0.3cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item \textbf{No lleva} \texttt{ON}: no empareja nada, combina \textbf{cada fila con todas} las de la otra tabla.
  \item Las filas se \textbf{multiplican}: 2 tallas $\times$ 3 colores = \textbf{6 filas}.
\end{itemize}
\vspace{0.2cm}
\ojo{Con tablas grandes explota: 1\,000 $\times$ 1\,000 = \textbf{1\,000\,000} de filas. Se usa poco, y a propósito.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{SELF JOIN}: una tabla unida consigo misma}
\begin{columns}[c,onlytextwidth]
\column{.27\textwidth}
\centering\tbl{empleados}\\[4pt]
\scriptsize\ttfamily
\begin{tabular}{rlr}
\toprule
\normalfont\bfseries id & \normalfont\bfseries nombre & \normalfont\bfseries jefe\_id \\
\midrule
1 & Rosa & \nulo \\
2 & Carlos & 1 \\
3 & Elena & 1 \\
4 & Diego & 2 \\
\bottomrule
\end{tabular}
\column{.44\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} e.nombre \kw{AS} empleado,\\
\ind\ind\ \ j.nombre \kw{AS} jefe\\
\kw{FROM} empleados \nuevo{e}\\
\kw{JOIN} empleados \nuevo{j} \kw{ON} e.jefe\_id = j.id;
\end{sqlcode}
\column{.25\textwidth}
\centering\scriptsize\ttfamily
\begin{tabular}{ll}
\toprule
\normalfont\bfseries empleado & \normalfont\bfseries jefe \\
\midrule
Carlos & Rosa \\
Elena & Rosa \\
Diego & Carlos \\
\bottomrule
\end{tabular}
\end{columns}
\vspace{0.3cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item El jefe \textbf{también es un empleado}: está en la misma tabla.
  \item \textbf{No es una palabra nueva}: es un \texttt{JOIN} de siempre, con la tabla escrita \textbf{dos veces}.
  \item Los apodos son \textbf{obligatorios}: \texttt{e} es el empleado y \texttt{j} es su jefe.
\end{itemize}
\vspace{0.15cm}
\small Rosa no aparece como empleado porque no tiene jefe. ¿Con qué \texttt{JOIN} sí aparecería?
\end{frame}

\seccion[0]{Índices}
% ---------------------------------------------------------------------
\begin{frame}{Antes de empezar: tres preguntas}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}
\centering\etiqueta{Un diccionario}\\[6pt]
\raggedright\small ¿Cómo encuentras la palabra «zapato»? ¿Lees desde la «a»?
\column{.31\textwidth}
\centering\etiqueta{Un libro de 500 páginas}\\[6pt]
\raggedright\small ¿Cómo encuentras el capítulo de «JOIN»? ¿Pasas hoja por hoja?
\column{.31\textwidth}
\centering\etiqueta{Tu celular}\\[6pt]
\raggedright\small ¿Cómo encuentras a «Mamá» entre 400 contactos?
\end{columns}
\vspace{0.5cm}
\begin{itemize}\setlength\itemsep{7pt}
  \item En los tres casos hay algo \textbf{ordenado} que te dice \textbf{dónde ir}.
  \item Por eso \textbf{no revisas todo}: saltas casi directo al lugar correcto.
\end{itemize}
\vspace{0.4cm}
\concepto{La idea}{\small Una base de datos tiene el mismo problema cuando busca una fila, y lo resuelve igual: con un \textbf{índice}.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Buscar sin índice: fila por fila}
\begin{columns}[c,onlytextwidth]
\column{.30\textwidth}
\centering\tbl{clientes}\\[4pt]
\scriptsize\ttfamily
\begin{tabular}{rl}
\toprule
\normalfont\bfseries id & \normalfont\bfseries nombre \\
\midrule
1 & Rosa \\
2 & Beto \\
3 & Zoe \\
4 & Ana \\
5 & Luis \\
\bottomrule
\end{tabular}
\column{.66\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} * \kw{FROM} clientes\\
\kw{WHERE} nombre = 'Luis';
\end{sqlcode}
\vspace{-0.15cm}
\small
\begin{itemize}\setlength\itemsep{4pt}
  \item Los nombres están \textbf{desordenados}: el gestor pregunta fila por fila. ¿Rosa? no. ¿Beto? no. ¿Zoe? no\ldots
  \item Tiene que revisar \textbf{toda la tabla}.
\end{itemize}
\end{columns}
\vspace{0.1cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{>{\raggedright\arraybackslash}p{4.6cm}>{\raggedright\arraybackslash}p{6.6cm}}
\toprule
\bfseries Si la tabla tiene\ldots & \bfseries Revisa\ldots \\
\midrule
5 filas & 5 filas: no se nota \\
1\,000\,000 de filas & \mal{1\,000\,000 de filas}, en \textbf{cada} búsqueda \\
\bottomrule
\end{tabular}
\end{center}
\vspace{-0.1cm}
\analogia{Es buscar una palabra en un libro \textbf{sin índice}: toca leerlo completo, página por página.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{¿Qué es un índice?}
\begin{columns}[c,onlytextwidth]
\column{.52\textwidth}
\centering
\begin{tikzpicture}[y=0.56cm]
  \tikzset{
    celda/.style={draw=utpNegro,thick,font=\scriptsize\ttfamily,minimum height=0.48cm,
      inner sep=2.5pt,align=left},
    idx/.style={celda,fill=yellow!35,text width=1.15cm},
    dato/.style={celda,fill=utpGrisClaro,text width=1.75cm}
  }
  \node[idx] (i1) at (0,0) {Ana};
  \node[idx] (i2) at (0,-1) {Beto};
  \node[idx] (i3) at (0,-2) {Luis};
  \node[idx] (i4) at (0,-3) {Rosa};
  \node[idx] (i5) at (0,-4) {Zoe};
  \node[dato] (t1) at (4.3,0) {1 \ Rosa};
  \node[dato] (t2) at (4.3,-1) {2 \ Beto};
  \node[dato] (t3) at (4.3,-2) {3 \ Zoe};
  \node[dato] (t4) at (4.3,-3) {4 \ Ana};
  \node[dato] (t5) at (4.3,-4) {5 \ Luis};
  \draw[-{Stealth[length=1.8mm]},utpGris,thick] (i1.east)--(t4.west);
  \draw[-{Stealth[length=1.8mm]},utpGris,thick] (i2.east)--(t2.west);
  \draw[-{Stealth[length=1.8mm]},utpRojo,very thick] (i3.east)--(t5.west);
  \draw[-{Stealth[length=1.8mm]},utpGris,thick] (i4.east)--(t1.west);
  \draw[-{Stealth[length=1.8mm]},utpGris,thick] (i5.east)--(t3.west);
  \node[font=\scriptsize\bfseries,above=1mm of i1] {índice};
  \node[font=\scriptsize,above=1mm of t1] {\tbl{clientes}};
  \node[font=\tiny,text=utpGris,below=1mm of i5] {ordenado};
  \node[font=\tiny,text=utpGris,below=1mm of t5] {en el orden en que llegaron};
\end{tikzpicture}
\column{.46\textwidth}
\begin{itemize}\setlength\itemsep{8pt}
  \item Es una \textbf{lista aparte}, \textbf{ordenada} por una columna.
  \item Cada entrada \textbf{apunta} a su fila en la tabla.
  \item La tabla \textbf{no cambia}: sigue igual, con todos sus datos.
\end{itemize}
\end{columns}
\vspace{0.3cm}
\small Para encontrar a Luis, el gestor busca en la lista ordenada y \textbf{salta directo} a la fila 5.
\vspace{0.2cm}
\analogia{Es el \textbf{índice al final de un libro}: las palabras en orden alfabético y, al lado, la página donde están.}
\end{frame}

\begin{frame}{\texttt{CREATE INDEX}: crear un índice}
\begin{center}
\begin{tikzpicture}[y=0.80cm]
  \renewcommand{\piezax}{6.0}
  \pieza{1}{\nuevo{\kw{CREATE INDEX}} idx\_clientes\_nombre}{el \textbf{nombre} que le pones al índice}
  \pieza{0}{\nuevo{\kw{ON}} clientes (nombre);}{la \textbf{tabla} y la \textbf{columna} que ordena}
\end{tikzpicture}
\end{center}
\vspace{0.15cm}
\begin{columns}[c,onlytextwidth]
\column{.50\textwidth}
\small
\begin{itemize}\setlength\itemsep{6pt}
  \item Se crea \textbf{una sola vez}.
  \item Tus consultas \textbf{no cambian}: el gestor decide solo cuándo usarlo.
  \item Se borra con \texttt{DROP INDEX}. Los datos \textbf{no se tocan}.
\end{itemize}
\column{.46\textwidth}
\centering\small
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{>{\raggedright\arraybackslash}p{2.5cm}>{\raggedright\arraybackslash}p{2.9cm}}
\toprule
\multicolumn{2}{c}{\bfseries Buscar entre 1\,000\,000 de filas} \\
\midrule
Sin índice & \mal{hasta 1\,000\,000} de pasos \\
Con índice & unos \textbf{20} pasos \\
\bottomrule
\end{tabular}
\end{columns}
\vspace{0.3cm}
\analogia{Con la lista ordenada se busca como en el diccionario: abres \textbf{por la mitad}, descartas una mitad, y repites. En 20 mitades ya llegaste.}
\end{frame}

\begin{frame}{¿La clave primaria es un índice?}
\concepto{Sí}{\small Al declarar una \texttt{PRIMARY KEY}, el gestor crea \textbf{un índice por su cuenta}. Ya lo venías usando sin saberlo.}
\vspace{0.1cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{>{\ttfamily\bfseries\raggedright\arraybackslash}p{2.9cm}>{\raggedright\arraybackslash}p{3.2cm}>{\raggedright\arraybackslash}p{6.0cm}}
\toprule
\normalfont\bfseries Columna & \bfseries ¿Tiene índice? & \bfseries ¿Por qué? \\
\midrule
PRIMARY KEY & \textbf{Sí}, automático & para encontrar una fila por su \texttt{id} al instante \\
UNIQUE & \textbf{Sí}, automático & para revisar rápido que el valor no se repita \\
FOREIGN KEY & depende del gestor & MySQL lo crea solo; SQLite no \\
Las demás & \textcolor{utpRojo}{\textbf{No}} & lo creas tú, con \texttt{CREATE INDEX} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.2cm}
\small \textbf{Y esto se junta con lo de hoy:} un \texttt{JOIN} empareja con \texttt{ON p.cliente\_id = c.id}. Como \texttt{c.id} es clave primaria, encontrar a cada cliente es \textbf{rápido}.
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{¿Entonces le pongo índice a todo?}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{Lo que ganas}\\[6pt]
\raggedright\small
\begin{itemize}\setlength\itemsep{5pt}
  \item Búsquedas con \texttt{WHERE} más rápidas.
  \item \texttt{JOIN} más rápidos.
  \item \texttt{ORDER BY} más rápidos: ya está ordenado.
\end{itemize}
\column{.48\textwidth}
\centering\etiqueta{Lo que pagas}\\[6pt]
\raggedright\small
\begin{itemize}\setlength\itemsep{5pt}
  \item El índice \textbf{ocupa espacio}.
  \item Cada \texttt{INSERT}, \texttt{UPDATE} y \texttt{DELETE} es \textbf{más lento}: también hay que actualizar el índice.
\end{itemize}
\end{columns}
\vspace{0.35cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{>{\raggedright\arraybackslash}p{6.0cm}>{\raggedright\arraybackslash}p{6.0cm}}
\toprule
\bfseries Conviene un índice & \bfseries No hace falta \\
\midrule
Columnas por las que \textbf{buscas seguido} & Columnas que casi nunca consultas \\
Tablas \textbf{grandes} & Tablas de pocas filas \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.15cm}
\analogia{Si a un libro le agregas páginas, hay que \textbf{rehacer su índice}. Con diez índices, lo rehaces diez veces.}
\end{frame}

\begin{frame}{Cierre: lo de hoy en una tabla}
\begin{center}
\renewcommand{\arraystretch}{1.45}
\begin{tabular}{>{\raggedright\bfseries}p{3.4cm}>{\raggedright\arraybackslash}p{9.0cm}}
\toprule
& \bfseries En una frase \\
\midrule
