\begin{frame}{Nombrar tablas: minúscula, singular y sin tildes}
\begin{center}
\small\setlength\tabcolsep{7pt}
\begin{tabular}{>{\raggedright\arraybackslash}p{5.6cm}p{3.0cm}p{3.0cm}}
\toprule
\bfseries Regla & \bfseries Mal & \bfseries Bien \\
\midrule
Todo en minúscula                    & \texttt{Cliente}        & \tbl{cliente} \\[3pt]
En singular: la tabla describe \textbf{una} fila & \texttt{clientes}       & \tbl{cliente} \\[3pt]
Sin tildes ni ñ                      & \texttt{categoría}      & \tbl{categoria} \\[3pt]
Palabras unidas con guion bajo       & \texttt{MetodoPago}     & \tbl{metodo\_pago} \\[3pt]
Sin espacios ni abreviaturas raras   & \texttt{det ped}        & \tbl{detalle\_pedido} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.3cm}
\begin{block}{La prueba del nombre}
\small Si el nombre de la tabla completa bien la frase ``\textit{cada fila es un\ldots}'', está bien elegido. \textit{Cada fila es un} \tbl{cliente}. \textit{Cada fila es un} \tbl{detalle\_pedido}.
\end{block}
\end{frame}

% 5 ---------------------------------------------------------------------
\begin{frame}{Nombrar atributos: el nombre dice qué guarda}
\begin{itemize}\setlength\itemsep{5pt}\small
  \item La clave primaria siempre se llama \texttt{id}.
  \item La clave foránea es el nombre de la tabla a la que apunta más \texttt{\_id}: \texttt{cliente\_id} apunta a \tbl{cliente}.
  \item No repitas el nombre de la tabla dentro de la columna: en \tbl{cliente} va \texttt{nombre}, no \texttt{nombre\_cliente}.
  \item Las fechas empiezan con \texttt{fecha\_}: \texttt{fecha\_pedido}, \texttt{fecha\_nacimiento}.
\end{itemize}
\vspace{0.2cm}
\begin{block}{Columnas que solo guardan sí o no}
\small Cuando la columna únicamente puede valer \textbf{sí} o \textbf{no}, el nombre se escribe como una \textbf{afirmación que se pueda responder con sí o no}, empezando por \texttt{es\_} o \texttt{tiene\_}:
\vspace{0.2cm}
\centerline{\small
\begin{tabular}{@{}l@{\hspace{1.4em}}l@{\hspace{1.4em}}l@{}}
\texttt{es\_activo}       & \textit{¿el cliente está activo?}    & sí / no \\
\texttt{es\_pagado}       & \textit{¿el pedido está pagado?}     & sí / no \\
\texttt{tiene\_descuento} & \textit{¿el pedido tiene descuento?} & sí / no \\
\end{tabular}}
\end{block}
\vspace{0.1cm}
\begin{center}
\small \tblb{pedido}\texttt{(\underline{id}, fecha\_pedido, total, es\_pagado, cliente\_id)}
\end{center}
\end{frame}

  \draw[relacion] (cli)--(ped);
  \pic[rotate=180] at (cli.east) {cfuno};
  \pic at (ped.west) {cfomuchos};
  \node[anchor=south,font=\tiny,inner sep=2pt]
    at ($(cli.east)!0.5!(ped.west)+(0,1mm)$) {hace};
\end{tikzpicture}
\end{center}
\vspace{0.2cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{exampleblock}{Mirando el símbolo de la derecha}
\small Un cliente hace \textbf{cero o muchos} pedidos. La pata de gallo está pegada a \tbl{pedido}, así que el lado ``muchos'' es el de los pedidos.
\end{exampleblock}
\column{.48\textwidth}
\begin{exampleblock}{Mirando el símbolo de la izquierda}
\small Un pedido pertenece a \textbf{uno y solo un} cliente. Por eso \tbl{pedido} guarda \texttt{cliente\_id}: un solo valor le basta.
\end{exampleblock}
\end{columns}
\end{frame}

  \draw[-{Stealth[length=1.8mm]},semithick,utpRojo] (o1)--(cli.south);
  \node[font=\scriptsize\bfseries,text=utpRojo,below=12mm of ped] (o2) {aquí vive la clave foránea};
  \draw[-{Stealth[length=1.8mm]},semithick,utpRojo] (o2)--(ped.south);
\end{tikzpicture}
\end{center}
\vspace{0.2cm}
\begin{columns}[T,onlytextwidth]
\column{.50\textwidth}
\begin{block}{La regla, en una línea}
\small La clave foránea vive siempre en la tabla del lado de la \textbf{pata de gallo}, y apunta a la clave primaria del otro lado.
\end{block}
\column{.46\textwidth}
\begin{exampleblock}{Qué garantiza}
\small Que no exista un \tbl{pedido} con un \texttt{cliente\_id} que no corresponde a ningún cliente. La base de datos lo rechaza.
\end{exampleblock}
\end{columns}
\vspace{0.2cm}
\begin{center}
\small\texttt{FOREIGN KEY (cliente\_id) REFERENCES cliente(id)}
\end{center}
\end{frame}

\begin{frame}{La pregunta que separa los tres casos}
\begin{block}{¿Dónde cae la clave foránea?}
\large ¿La clave foránea forma parte de la \textbf{clave primaria} de la tabla hija, o es solo una columna más?
\end{block}
\vspace{0.35cm}
\begin{center}
\small
\begin{tabular}{>{\bfseries}p{3.4cm}>{\raggedright\arraybackslash}p{8.6cm}}
\toprule
Identificadora & La \fk\ \textbf{es parte} de la clave primaria. La hija no existe sin la madre. Línea \textbf{continua}. \\[4pt]
No identificadora & La \fk\ es una columna común. La hija tiene su propio \texttt{id} y vida propia. Línea \textbf{punteada}. \\[4pt]
Generalización & Una entidad general y sus casos particulares. Los hijos comparten el \texttt{id} del padre. \\
\bottomrule
\end{tabular}
\end{center}
\end{frame}
