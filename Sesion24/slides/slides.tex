\begin{frame}{Dos listas, tres preguntas}
\small Con esas dos listas puedes responder tres preguntas distintas, y cada una lleva a una decisión.
\vspace{0.2cm}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}
\centering
\vennchico{on}{on}{on}\\[6pt]
\etiqueta{UNION}\\[6pt]
\small ¿Quién está en \textbf{alguna} de las dos listas?\\[6pt]
\footnotesize\gris{Enviarles el saludo de fin de año.}
\column{.31\textwidth}
\centering
\vennchico{off}{on}{off}\\[6pt]
\etiqueta{INTERSECT}\\[6pt]
\small ¿Quién está en \textbf{las dos}?\\[6pt]
\footnotesize\gris{Darles el descuento de personal.}
\column{.31\textwidth}
\centering
\vennchico{on}{off}{off}\\[6pt]
\etiqueta{EXCEPT}\\[6pt]
\small ¿Quién es cliente y \textbf{no} empleado?\\[6pt]
\footnotesize\gris{Enviarles la promoción de noviembre.}
\end{columns}
\vspace{0.3cm}
\concepto{Combinar SELECT}{\small Se escriben dos \texttt{SELECT} y, entre ellos, un operador. El resultado es \textbf{una sola lista}. \texttt{JOIN} agrega \textbf{columnas}; estos operadores trabajan con \textbf{filas}.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{UNION}: todo lo de las dos listas}
\begin{columns}[c,onlytextwidth]
\column{.42\textwidth}
\centering
\venn{on}{on}{on}
\column{.55\textwidth}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT} dni, nombre \kw{FROM} cliente\\
\dentro{\kw{UNION}}\\
\kw{SELECT} dni, nombre \kw{FROM} empleado;
\end{sqlcode}
\vspace{-0.1cm}
\res{ll}{\cab{dni} & \cab{nombre}}{45123678 & Lucía Ramos \\ 46234789 & Pedro Salas \\ 47345890 & Rosa Medina \\ 48456901 & Bruno Díaz \\ 43567012 & Ana Torres \\ 44678123 & Carla Núñez \\}
\end{columns}
\vspace{0.3cm}
\analogia{Bruno Díaz está en las dos tablas con el mismo DNI y sale \textbf{una sola vez}: \texttt{UNION} quita los repetidos. Con \texttt{UNION ALL}, que los deja, saldría dos veces.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{INTERSECT}: solo lo que está en las dos}
\begin{columns}[c,onlytextwidth]
\column{.42\textwidth}
\centering
\venn{off}{on}{off}
\column{.55\textwidth}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT} dni, nombre \kw{FROM} cliente\\
\dentro{\kw{INTERSECT}}\\
\kw{SELECT} dni, nombre \kw{FROM} empleado;
\end{sqlcode}
\vspace{-0.1cm}
\res{ll}{\cab{dni} & \cab{nombre}}{48456901 & Bruno Díaz \\}
\end{columns}
\vspace{0.3cm}
\analogia{Bruno Díaz está en \textbf{las dos listas} con el mismo DNI: le corresponde el descuento de personal. El DNI evita confundir a dos personas que se llaman igual.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{EXCEPT}: lo que está en la primera y no en la segunda}
\begin{columns}[c,onlytextwidth]
\column{.42\textwidth}
\centering
\venn{on}{off}{off}
\column{.55\textwidth}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT} dni, nombre \kw{FROM} cliente\\
\dentro{\kw{EXCEPT}}\\
\kw{SELECT} dni, nombre \kw{FROM} empleado;
\end{sqlcode}
\vspace{-0.1cm}
\res{ll}{\cab{dni} & \cab{nombre}}{45123678 & Lucía Ramos \\ 46234789 & Pedro Salas \\ 47345890 & Rosa Medina \\}
\end{columns}
\vspace{0.3cm}
\analogia{Lucía, Pedro y Rosa son clientes y \textbf{no trabajan aquí}: reciben la promoción. \textbf{El orden importa}: al revés, empleado \texttt{EXCEPT} cliente da \textbf{Ana} y \textbf{Carla}.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Reglas para combinar dos \texttt{SELECT}}
\begin{itemize}\setlength\itemsep{5pt}
  \item[\npieza{1}] La misma \textbf{cantidad de columnas} en los dos \texttt{SELECT}.
  \item[\npieza{2}] Las columnas en el mismo \textbf{orden} y con tipos compatibles.
  \item[\npieza{3}] Los \textbf{nombres} de las columnas salen del primer \texttt{SELECT}.
  \item[\npieza{4}] Un solo \texttt{ORDER BY}, al \textbf{final} de todo.
\end{itemize}
\begin{sqlcode}[.8\textwidth]
\kw{SELECT} nombre, telefono \kw{FROM} cliente \kw{UNION SELECT} nombre \kw{FROM} empleado;\\[4pt]
\color{utpRojo}ERROR 1222 (21000): The used SELECT statements have a different number of columns
\end{sqlcode}
\vspace{-0.2cm}
\ojo{\texttt{INTERSECT} y \texttt{EXCEPT} funcionan desde \textbf{MySQL 8.0.31}. Revisa tu versión con \texttt{SELECT VERSION();}. En versiones anteriores dan error de sintaxis y se reemplazan con \texttt{IN} y \texttt{NOT IN}.}
\end{frame}

\begin{frame}{\ins{Práctica 2} \texttt{EXCEPT} en los dos gestores}
\small Abre \texttt{ejemplos/practicas/practica\_2\_combinar.sql} y ejecútalo en los dos gestores.
\vspace{0.1cm}
\begin{enumerate}\setlength\itemsep{3pt}
  \item \textbf{MySQL:} en Workbench.
  \item \textbf{SQL Server:} en SSMS.
\end{enumerate}
\vspace{0.1cm}
\small ¿Qué servicios cuestan menos de S/ 60, sin contar los que duran una hora o más?
\begin{columns}[c,onlytextwidth]
\column{.57\textwidth}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{USE} peluqueria;\\
\kw{SELECT} nombre, precio \kw{FROM} servicio\\
\kw{WHERE} precio < 60\\
\dentro{\kw{EXCEPT}}\\
\kw{SELECT} nombre, precio \kw{FROM} servicio\\
\kw{WHERE} duracion >= 60\\
\kw{ORDER BY} precio;
\end{sqlcode}
\column{.39\textwidth}
\res{lr}{\cab{nombre} & \cab{precio}}{Corte de uñas & 15.00 \\ Baño e higiene & 35.00 \\}
\end{columns}
\vspace{0.1cm}
\concepto{Idea clave}{\small La lista sale igual en los dos gestores: \texttt{EXCEPT} se escribe de la misma forma.}
\end{frame}

\seccion{Subconsultas}
\begin{frame}{Una pregunta que necesita otra pregunta}
\begin{center}
\begin{tikzpicture}
  \node[caja,cajanaranja,font=\large,inner sep=9pt] {¿Qué servicios cuestan \textbf{más que el promedio}?};
\end{tikzpicture}
\end{center}
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\npieza{1}\ \small Primero, ¿cuál es el promedio?
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT AVG}(precio) \kw{FROM} servicio;
\end{sqlcode}
\vspace{-0.1cm}
\res{r}{\cab{AVG(precio)}}{46.250000 \\}
\column{.49\textwidth}
\npieza{2}\ \small Luego, ¿qué servicios cuestan más?
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT} nombre, precio \kw{FROM} servicio\\
\kw{WHERE} precio > \mal{46.25};
\end{sqlcode}
\vspace{-0.1cm}
\res{lr}{\cab{nombre} & \cab{precio}}{Corte de pelo & 55.00 \\ Baño completo con corte & 80.00 \\}
\end{columns}
\vspace{0.2cm}
\ojo{Funciona, pero copiaste \texttt{46.25} \textbf{a mano}. Si mañana cambia un precio, ese número queda viejo.}
\end{frame}

\begin{frame}{En una sola: la subconsulta}
\small En lugar de escribir el número, escribe \textbf{la consulta que lo calcula}:
\vspace{0.2cm}
\begin{center}
\begin{tikzpicture}
  \node[codigo,anchor=west] (a) at (0,0) {\kw{WHERE} precio >};
  \node[codigo,anchor=west,draw=utpRojo,thick,fill=utpRojo!18] (n) at ([xshift=2pt]a.east) {46.25};
  \node[codigo,anchor=west] (b) at (0,-1.5) {\kw{WHERE} precio >};
  \node[sub,anchor=west] (s) at ([xshift=2pt]b.east) {(\kw{SELECT AVG}(precio) \kw{FROM} servicio)};
  \draw[flecha] (n.south)--node[right,font=\scriptsize]{se reemplaza por}(n.south|-s.north);
\end{tikzpicture}
\end{center}
\vspace{0.1cm}
\begin{sqlcode}[.62\textwidth]
\kw{SELECT} nombre, precio\\
\kw{FROM} servicio\\
\kw{WHERE} precio > \dentro{(\kw{SELECT AVG}(precio) \kw{FROM} servicio)};
\end{sqlcode}
\vspace{-0.1cm}
\concepto{Subconsulta}{\small Una consulta escrita \textbf{dentro de otra}, entre \textbf{paréntesis}. El gestor resuelve \textbf{primero} la de adentro y usa su resultado en la de afuera.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{¿Qué devuelve la de adentro?}
\small Lo primero es mirar la \textbf{forma} de su resultado. Eso decide cómo se usa.
\vspace{0.3cm}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}
\centering
{\large\bfseries Un valor}\\[8pt]
\forma{|c|}{\cellcolor{yellow!45}46.250000 \\ \hline}\\[8pt]
\footnotesize una fila, una columna\\[10pt]
\etiqueta{= \ > \ < }\\[6pt]
\scriptsize se compara
\column{.31\textwidth}
\centering
{\large\bfseries Una lista}\\[8pt]
\forma{|c|}{\cellcolor{yellow!45}1 \\ \hline \cellcolor{yellow!45}2 \\ \hline}\\[8pt]
\footnotesize una columna, varias filas\\[10pt]
\etiqueta{IN \ NOT IN}\\[6pt]
\scriptsize se busca dentro
\column{.31\textwidth}
\centering
{\large\bfseries Una tabla}\\[8pt]
\forma{|c|c|}{\cellcolor{yellow!45}1 & \cellcolor{yellow!45}4 \\ \hline \cellcolor{yellow!45}2 & \cellcolor{yellow!45}3 \\ \hline}\\[8pt]
\footnotesize varias columnas y filas\\[10pt]
\etiqueta{FROM}\\[6pt]
\scriptsize se usa como tabla
\end{columns}
\vspace{0.2cm}
\ojo{Si la de adentro devuelve una lista y usas \texttt{=}, sale \texttt{ERROR 1242: Subquery returns more than 1 row}. El signo \texttt{=} espera un solo valor; para una lista, usa \texttt{IN}.}
\end{frame}

\begin{frame}{Con \texttt{IN} y \texttt{NOT IN}: ¿qué empleados tienen citas?}
\small \texttt{IN} ya lo conoces: \texttt{WHERE id IN (1, 2)}. Ahora la lista la escribe \textbf{una consulta}.
\begin{center}
\begin{tikzpicture}
  \node (ci) at (0,0) {\tablacita};

\begin{frame}{En \texttt{FROM}: ¿cuántas citas tiene el que más tiene?}
\small \texttt{MAX(COUNT(*))} \textbf{no se puede} escribir. Primero cuentas; luego buscas el máximo.
\begin{center}
\begin{tikzpicture}
  \node (ci) at (0,0) {\tablacita};
  \node[draw=utpRojo,very thick,dashed,rounded corners=3pt,fill=yellow!20,inner sep=6pt] (co) at (4.6,0)
    {\mini{conteo}{rr}{\cab{empleado\_id} & \cab{total}}{1 & 4 \\ 2 & 3 \\}};
  \node[font=\scriptsize,text=utpRojo,below=2pt of co] {tabla temporal};
  \node[caja,cajaverde,font=\large\bfseries,minimum width=1.2cm] (r) at (8.6,0) {4};
  \draw[flecha] (ci.east)--node[above,font=\scriptsize]{\npieza{1} contar}(co.west);
  \draw[flecha] (co.east)--node[above,font=\scriptsize]{\npieza{2} \texttt{MAX}}(r.west);
\end{tikzpicture}
\end{center}
\vspace{0.1cm}
\begin{sqlcode}[.62\textwidth]
\kw{SELECT MAX}(total)\\
\kw{FROM} \dentro{(\kw{SELECT} empleado\_id, \kw{COUNT}(*) \kw{AS} total}\\
\ind\ind\dentro{\kw{FROM} cita \kw{GROUP BY} empleado\_id)} \kw{AS} conteo;
\end{sqlcode}
\vspace{-0.5cm}
\ojo{En \texttt{FROM}, la subconsulta \textbf{necesita un nombre} (\texttt{AS conteo}). Sin él: \texttt{ERROR 1248: Every derived table must have its own alias}.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{En \texttt{SELECT}: una columna más}
\small Si la de adentro devuelve \textbf{un valor}, puede ir como una columna. ¿Cuántas citas tiene \textbf{cada} empleado? Aquí la de adentro usa \texttt{e.id}, que viene de la consulta de afuera.
\begin{sqlcode}[.74\textwidth]
\kw{SELECT} e.nombre,\\
\ind\dentro{(\kw{SELECT COUNT}(*) \kw{FROM} cita c \kw{WHERE} c.empleado\_id = \mal{e.id})} \kw{AS} citas\\
\kw{FROM} empleado e;
\end{sqlcode}
\vspace{-0.1cm}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{@{}lll@{}}
\toprule
\bfseries Fila de afuera & \bfseries La de adentro cuenta\ldots & \bfseries citas \\
\midrule
Ana Torres, \texttt{id = 1} & las citas con \texttt{empleado\_id = 1} & \textbf{4} \\
Bruno Díaz, \texttt{id = 2} & las citas con \texttt{empleado\_id = 2} & \textbf{3} \\
Carla Núñez, \texttt{id = 3} & las citas con \texttt{empleado\_id = 3} & \textbf{0} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.1cm}
\concepto{Subconsulta correlacionada}{\small Usa una columna de afuera. Por eso \textbf{no} se resuelve una sola vez: se ejecuta \textbf{una vez por cada fila} de afuera.}
\end{frame}

\begin{frame}{\texttt{EXISTS}: ¿hay al menos una fila?}
\small No necesitas un dato. Solo quieres saber si \textbf{existe}: ¿este empleado tiene alguna cita?
\begin{sqlcode}[.76\textwidth]
\kw{SELECT} e.nombre\\
\kw{FROM} empleado e\\
\kw{WHERE EXISTS} \dentro{(\kw{SELECT} 1 \kw{FROM} cita c \kw{WHERE} c.empleado\_id = e.id)};
\end{sqlcode}
\vspace{-0.1cm}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{@{}llc@{}}
\toprule
\bfseries Fila de afuera & \bfseries ¿Tiene alguna cita? & \bfseries ¿Se queda? \\
\midrule
Ana Torres, \texttt{id = 1} & sí: encontró la cita 1 y dejó de buscar & \si \\
Bruno Díaz, \texttt{id = 2} & sí: encontró la cita 2 y dejó de buscar & \si \\
Carla Núñez, \texttt{id = 3} & no: revisó todas y no hay & \no \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.1cm}
\concepto{\texttt{EXISTS}}{\small Responde \textbf{sí o no}: sí, cuando la de adentro encuentra \textbf{al menos una fila}. No mira los valores, por eso se suele escribir \texttt{SELECT 1}.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{NOT EXISTS}: que no haya ninguna}
\small Es la pregunta contraria. Sirve para encontrar lo que \textbf{quedó sin relación}.
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{Empleados sin citas}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT} e.nombre\\
\kw{FROM} empleado e\\
\kw{WHERE NOT EXISTS} \dentro{(\kw{SELECT} 1}\\
\ind\dentro{\kw{FROM} cita c}\\
\ind\dentro{\kw{WHERE} c.empleado\_id = e.id)};
\end{sqlcode}
\vspace{-0.1cm}
\res{l}{\cab{nombre}}{Carla Núñez \\}
\column{.48\textwidth}
\centering\etiqueta{Citas atendidas sin cobrar}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT} c.id, c.fecha\_hora\\
\kw{FROM} cita c\\
\kw{WHERE} c.estado = 'completada'\\
\ind\kw{AND NOT EXISTS} \dentro{(\kw{SELECT} 1}\\
\ind\ind\dentro{\kw{FROM} pago p}\\
\ind\ind\dentro{\kw{WHERE} p.cita\_id = c.id)};
\end{sqlcode}
\vspace{-0.1cm}
\res{rl}{\cab{id} & \cab{fecha\_hora}}{6 & 2026-10-20 09:00:00 \\}
\end{columns}
\vspace{0.1cm}
\analogia{La segunda también sale con \texttt{EXCEPT}: citas completadas menos citas en \texttt{pago}. Con \texttt{NOT EXISTS} puedes mostrar más columnas, como la fecha.}
\end{frame}

\begin{frame}{\texttt{EXISTS} vs \texttt{IN}}
\small Las dos consultas dan el mismo resultado: \textbf{Ana Torres} y \textbf{Bruno Díaz}.
\vspace{-0.1cm}
\begin{columns}[T,onlytextwidth]
\column{.45\textwidth}
\centering\etiqueta{IN}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT} nombre \kw{FROM} empleado\\
\kw{WHERE} id \kw{IN}\\
\ind\dentro{(\kw{SELECT} empleado\_id}\\
\ind\dentro{\kw{FROM} cita)};
\end{sqlcode}
\column{.52\textwidth}
\centering\etiqueta{EXISTS}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT} e.nombre \kw{FROM} empleado e\\
\kw{WHERE EXISTS}\\
\ind\dentro{(\kw{SELECT} 1 \kw{FROM} cita c}\\
\ind\dentro{\kw{WHERE} c.empleado\_id = e.id)};
\end{sqlcode}
\end{columns}
\vspace{-0.1cm}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{@{}>{\bfseries}l>{\raggedright}p{4.7cm}>{\raggedright\arraybackslash}p{5.1cm}@{}}
\toprule
 & \texttt{IN} & \texttt{EXISTS} \\
\midrule
Pregunta & ¿este valor está en la lista? & ¿hay al menos una fila? \\
La de adentro & se puede ejecutar sola & usa una columna de afuera \\
Cómo trabaja & arma la lista y busca en ella & prueba fila por fila \\
Conviene si & comparas una columna con una lista & necesitas varias condiciones \\
\bottomrule
\end{tabular}
\end{center}
\vspace{-0.1cm}
\analogia{Hoy los gestores suelen ejecutar las dos igual de rápido: elige la que se lea mejor. La diferencia real aparece con \texttt{NOT IN}.}
\end{frame}

\begin{frame}{Ojo: \texttt{NOT IN} y los \texttt{NULL}}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT} nombre \kw{FROM} empleado\\
\kw{WHERE} id \kw{NOT IN} (1, 2);
\end{sqlcode}
\vspace{-0.1cm}
\res{l}{\cab{nombre}}{Carla Núñez \\}
\column{.48\textwidth}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{SELECT} nombre \kw{FROM} empleado\\
\kw{WHERE} id \kw{NOT IN} (1, 2, \mal{NULL});
\end{sqlcode}
\vspace{-0.1cm}
\centering{\scriptsize\color{utpGris}resultado}\\[6pt]
\small\textbf{ninguna fila}
\end{columns}
\vspace{0.3cm}
\small ¿Por qué? Para Carla, el gestor compara su id con cada valor de la lista:
\vspace{0.1cm}
\begin{center}
\small
\texttt{3 <> 1} \si \qquad \texttt{3 <> 2} \si \qquad \texttt{3 <> NULL} \textbf{no se sabe}
\end{center}
\vspace{0.1cm}
\ojo{\texttt{NULL} es un valor desconocido. Si la lista tiene un \texttt{NULL}, el gestor no puede asegurar que el valor no esté, y descarta la fila. Pasa con columnas que admiten \texttt{NULL}, como \texttt{cliente.telefono}. \texttt{NOT EXISTS} no tiene este problema.}
\end{frame}

\seccion{Expresiones de tabla comunes}
\begin{frame}{Cuando la consulta crece}
\small ¿Qué empleado tiene \textbf{más citas}? Con lo que sabemos, se puede escribir así:
\begin{sqlcode}[.96\textwidth]
\kw{SELECT} e.nombre, conteo.total\\
\kw{FROM} empleado e\\
\kw{JOIN} \dentro{(\kw{SELECT} empleado\_id, \kw{COUNT}(*) \kw{AS} total \kw{FROM} cita \kw{GROUP BY} empleado\_id)} \kw{AS} conteo\\
\ind\kw{ON} conteo.empleado\_id = e.id\\
\kw{WHERE} conteo.total = (\kw{SELECT MAX}(total)\\
\ind\kw{FROM} \dentro{(\kw{SELECT} empleado\_id, \kw{COUNT}(*) \kw{AS} total \kw{FROM} cita \kw{GROUP BY} empleado\_id)} \kw{AS} otro);
\end{sqlcode}
\vspace{0.1cm}
\small Funciona y devuelve \textbf{Ana Torres, 4}. Pero tiene dos problemas:
\vspace{0.1cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item El mismo conteo está escrito \textbf{dos veces}.
  \item Se lee de \textbf{adentro hacia afuera}: hay que saltar por todo el código para entenderla.
\end{itemize}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{CTE: una subconsulta con nombre}
\concepto{CTE (\textit{Common Table Expression})}{\small Una subconsulta a la que le pones \textbf{un nombre}, escrita \textbf{antes} de la consulta principal. En español: expresión de tabla común.}
\vspace{0.1cm}
\begin{center}
\renewcommand{\piezax}{7.2}
\begin{tikzpicture}
  \pieza{0}{\kw{WITH} conteo \kw{AS} (}{le pones un nombre}
  \pieza{-0.68}{\ind\kw{SELECT} empleado\_id, \kw{COUNT}(*) \kw{AS} total}{cuenta las citas\ldots}
  \pieza{-1.36}{\ind\kw{FROM} cita \kw{GROUP BY} empleado\_id}{\ldots{} de cada empleado}
  \pieza{-2.04}{)}{se cierra, sin punto y coma}
  \pieza{-2.72}{\kw{SELECT MAX}(total) \kw{FROM} conteo;}{la consulta la usa como tabla}
\end{tikzpicture}
\end{center}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Se escribe una vez y se usa dos}
\small Volvamos a la pregunta larga: ¿qué empleado tiene más citas?
\vspace{0.1cm}
\begin{sqlcode}[.72\textwidth]
\kw{WITH} conteo \kw{AS} \dentro{(\kw{SELECT} empleado\_id, \kw{COUNT}(*) \kw{AS} total}\\
\ind\ind\ind\dentro{\kw{FROM} cita \kw{GROUP BY} empleado\_id)}\\
\kw{SELECT} e.nombre, conteo.total\\
\kw{FROM} empleado e\\
\kw{JOIN} \mal{conteo} \kw{ON} conteo.empleado\_id = e.id\\
\kw{WHERE} conteo.total = (\kw{SELECT MAX}(total) \kw{FROM} \mal{conteo});
\end{sqlcode}
\vspace{-0.1cm}
\resg{lr}{\cab{nombre} & \cab{total}}{Ana Torres & 4 \\}
\vspace{0.3cm}
\concepto{Idea clave}{\small Si cambias cómo se cuenta, lo cambias en \textbf{un solo lugar}. Un CTE solo existe durante su consulta: \textbf{no} queda guardado en la base.}
\end{frame}

\begin{frame}{Todo junto: la lista de cobros}
\small \texttt{WITH} se escribe \textbf{una vez} y los CTE se separan con \textbf{comas}. Cada uno puede usar los anteriores.
\begin{columns}[c,onlytextwidth]
\column{.6\textwidth}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\kw{WITH} atendidas \kw{AS} (\kw{SELECT} id \kw{FROM} cita\\
\ind\ind\ind\ind\kw{WHERE} estado = 'completada'),\\
cobradas \kw{AS} (\kw{SELECT} cita\_id \kw{FROM} pago),\\
sin\_cobrar \kw{AS} (\kw{SELECT} id \kw{FROM} atendidas\\
\ind\ind\ind\ind\kw{EXCEPT}\\
\ind\ind\ind\ind\kw{SELECT} cita\_id \kw{FROM} cobradas)\\
\kw{SELECT} c.id, c.fecha\_hora,\\
\ind s.nombre \kw{AS} servicio, s.precio\\
\kw{FROM} sin\_cobrar sc\\
\kw{JOIN} cita c \kw{ON} c.id = sc.id\\
\kw{JOIN} servicio s \kw{ON} s.id = c.servicio\_id;
\end{sqlcode}

\texttt{UNION} & Junta las filas de dos \texttt{SELECT} sin repetir. \texttt{UNION ALL} deja los repetidos. \\
\texttt{INTERSECT} & Solo las filas que están en las dos listas. \\
\texttt{EXCEPT} & Las filas de la primera que no están en la segunda. El orden importa. \\
Subconsulta & Una consulta dentro de otra, entre paréntesis. Puede ir en \texttt{WHERE}, \texttt{FROM} o \texttt{SELECT}. \\
\texttt{IN} & Busca un valor en una lista. Cuidado con \texttt{NOT IN} y \texttt{NULL}. \\
\texttt{EXISTS} & Responde sí o no: ¿hay al menos una fila? \\
CTE & Una subconsulta con nombre, escrita antes con \texttt{WITH}. Solo vive durante su consulta. \\
En una app & Cada pantalla que muestra datos ejecuta un \texttt{SELECT}. \\
