\begin{frame}{Programa, proceso y servicio}
\begin{description}\setlength\itemsep{10pt}
  \item[Programa] Un archivo con instrucciones. Ejemplos: \texttt{mysqld.exe} (MySQL) y \texttt{sqlservr.exe} (SQL Server).
  \item[Proceso] Un programa \textbf{mientras está funcionando}.
  \item[Servicio] Un proceso que funciona \textbf{sin ventana} y arranca solo al encender Windows.
\end{description}
\vspace{0.3cm}
\small Los servicios se ven en Windows con \textbf{Windows + R} $\rightarrow$ \texttt{services.msc}.
\end{frame}

\begin{center}

\concepto{Por qué importa}{\small Un gestor trabaja en \textbf{memoria} para ser rápido y guarda en \textbf{disco} para no perder nada.}

\begin{frame}{\npieza{1} Gestor de conexiones}
\concepto{Conexión}{\small Un canal abierto entre quien pide y el gestor. Por él viajan las consultas y las respuestas.}
\vspace{0.2cm}
\begin{itemize}\setlength\itemsep{8pt}
  \item \textbf{Autenticación}: comprueba \textbf{quién eres} (usuario y contraseña).
  \item \textbf{Autorización}: revisa \textbf{qué puedes hacer} (ver, agregar, borrar\ldots).
\end{itemize}
\vspace{0.2cm}
\ojo{El usuario del gestor \textbf{no es} tu usuario de Windows. En MySQL, el administrador se llama \texttt{root}.}
\end{frame}
% ---------------------------------------------------------------------

\begin{center}
\resizebox{\textwidth}{!}{%

  \node[caja,cajanaranja,text width=2.8cm,font=\scriptsize\ttfamily] (q) at (0,0) {SELECT nombre FROM servicio WHERE id = 2;};
  \node[paso] (a) at (3.8,0) {\textbf{Analizador}\\{\scriptsize ¿está bien escrita?\\¿existe la tabla?}};
  \node[paso] (o) at (7.6,0) {\textbf{Optimizador}\\{\scriptsize ¿cuál es el camino más rápido?}};
  \node[paso] (e) at (11.4,0) {\textbf{Ejecutor}\\{\scriptsize sigue el camino y arma el resultado}};
  \draw[flecha] (q)--(a); \draw[flecha] (a)--(o); \draw[flecha] (o)--(e);

\vspace{0.2cm}
\small Si el analizador encuentra un error, se detiene ahí:
\begin{sqlcode}[.8\textwidth]
\scriptsize
\kw{SELECT} nombre \kw{FROM} \mal{servicios};\\
\color{utpRojo}ERROR 1146 (42S02): Table 'peluqueria.servicios' doesn't exist
\end{sqlcode}
\end{frame}

\begin{frame}{\npieza{2} El optimizador y los índices}
\concepto{Índice}{\small Una lista ordenada para encontrar filas rápido, como el índice de un libro. La PK y los UNIQUE ya tienen uno.}
\vspace{0.1cm}
\small \texttt{EXPLAIN} muestra el camino que eligió el optimizador:
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\scriptsize
\kw{EXPLAIN SELECT} * \kw{FROM} servicio \kw{WHERE} id = 2;
\end{sqlcode}
\vspace{-0.15cm}
\res{ll}{\cab{type} & \cab{key}}{const & PRIMARY \\}
\\[4pt]\centering\scriptsize usa el índice: va \textbf{directo}
\column{.49\textwidth}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\scriptsize
\kw{EXPLAIN SELECT} * \kw{FROM} servicio \kw{WHERE} precio > 40;
\end{sqlcode}
\vspace{-0.15cm}
\res{ll}{\cab{type} & \cab{key}}{\mal{ALL} & NULL \\}
\\[4pt]\centering\scriptsize sin índice: revisa \textbf{todas} las filas
\end{columns}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\npieza{3}\,\npieza{4} Motor de almacenamiento y memoria intermedia}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\concepto{Motor de almacenamiento}{\small Guarda las filas en el disco y las vuelve a leer. En MySQL es \textbf{InnoDB}: el \texttt{ENGINE = InnoDB} de la sesión anterior.}
\column{.48\textwidth}
\concepto{Memoria intermedia}{\small Una parte de la RAM donde el gestor deja lo que leyó del disco. Si lo vuelve a necesitar, no va al disco.}
\end{columns}
\vspace{0.3cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item Por eso la segunda vez una consulta suele ser \textbf{más rápida}.
  \item Por eso un gestor usa \textbf{mucha RAM}: es a propósito.
\end{itemize}
\end{frame}

\begin{frame}{\npieza{5} Gestor de transacciones}
\concepto{Transacción}{\small Un grupo de cambios que ocurren \textbf{todos o ninguno}.}
\vspace{0.05cm}
\begin{sqlcode}[.7\textwidth]
\scriptsize
\kw{START TRANSACTION};\\
\kw{INSERT INTO} pago (monto, metodo, cita\_id) \kw{VALUES} (55.00, 'efectivo', 1);\\
\kw{UPDATE} cita \kw{SET} estado = 'completada' \kw{WHERE} id = 1;\\
\kw{COMMIT}; \ \com{confirma. ROLLBACK deshace}
\end{sqlcode}
\vspace{-0.1cm}
\concepto{Bloqueo}{\small Si dos personas cambian la misma fila a la vez, el gestor hace \textbf{esperar} a la segunda hasta que la primera termine.}
\end{frame}

\begin{frame}{\npieza{6} Registro de recuperación}
\concepto{Registro (\textit{log})}{\small Un archivo donde el gestor \textbf{anota cada cambio} antes de darlo por hecho.}
\vspace{0.2cm}
\begin{center}
\resizebox{.9\textwidth}{!}{%
\begin{tikzpicture}[font=\scriptsize]
  \tikzset{p/.style={caja,text width=2.45cm,minimum height=1.05cm,font=\scriptsize}}
  \node[p] (a) at (0,0) {\textbf{1.} cambia en \textbf{memoria}};
  \node[p,cajanaranja] (b) at (3.3,0) {\textbf{2.} lo anota en el \textbf{registro}};
  \node[p] (c) at (6.6,0) {\textbf{3.} responde «listo»};
  \node[p,cajaverde] (d) at (9.9,0) {\textbf{4.} más tarde, lo guarda en su \textbf{archivo}};
  \foreach \u/\v in {a/b,b/c,c/d} \draw[flecha] (\u)--(\v);
\end{tikzpicture}}
\end{center}
\vspace{0.3cm}
\small Si se va la luz antes del paso 4, al encender el gestor lee el registro y \textbf{recupera} los cambios.
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\npieza{7}\,\npieza{8} Catálogo y archivos de datos}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\concepto{Catálogo}{\small La lista de todo lo que hay: bases, tablas, columnas y usuarios. Son \textbf{datos sobre los datos}.}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\scriptsize
\kw{SHOW TABLES};\\
\kw{DESCRIBE} servicio;
\end{sqlcode}
\column{.48\textwidth}
\concepto{Archivos de datos}{\small Donde terminan guardadas las tablas. Cada tabla es un archivo \texttt{.ibd}.}
\begin{sqlcode}[\dimexpr\textwidth-2\fboxsep\relax]
\scriptsize
\kw{SELECT} @@datadir;\\
\com{C:\bs ProgramData\bs MySQL\bs \ldots\bs Data\bs}
\end{sqlcode}
\end{columns}
\vspace{0.1cm}
\ojo{Nunca copies ni borres esos archivos a mano: solo el gestor sabe leerlos.}
\end{frame}

\vspace{0.3cm}
\concepto{Workbench y SSMS no son la base de datos}{\small Son \textbf{clientes}. Si cierras Workbench, los datos siguen en el servidor. Si apagas el servicio, ningún cliente puede entrar.}
\end{frame}

\begin{frame}{SQLite: un gestor sin servidor}
\begin{center}
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{@{}>{\raggedright\bfseries}p{3.4cm}>{\raggedright}p{4.0cm}>{\raggedright\arraybackslash}p{3.6cm}@{}}
\toprule
& \etiqueta{MySQL, SQL Server} & \etiqueta{SQLite} \\
\midrule
¿Hay servidor? & sí & no \\
¿Dónde está la base? & en el servidor & en un archivo \texttt{.db} \\
¿Cómo se entra? & conectándose & abriendo el archivo \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.3cm}
\small Por eso DB Browser for SQLite no pide usuario ni contraseña. La próxima sesión veremos cómo conectarse a un gestor \textbf{con servidor}.
\end{frame}
