\begin{frame}{Revisión: una venta con transacción}
\renewcommand{\arraystretch}{0.85}
\small Vender una entrada son \textbf{dos cambios}: descontar una entrada disponible y guardar la venta.
\vspace{0.15cm}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\centering\etiqueta{La venta a Luis}
\begin{sqlmini}[\col]
\dentro{\kw{START TRANSACTION}};\\
\kw{UPDATE} entrada\\
\kw{SET} disponibles = disponibles - 1 \kw{WHERE} id = 1;\\
\kw{INSERT INTO} venta (dni, nombre, correo,\\
\ind precio, referencia\_pago)\\
\kw{VALUES} ('70123456', 'Luis Rojas',\\
\ind 'luis.rojas@example.com', 45.00, 'PAGO-0001');\\
\dentro{\kw{COMMIT}};
\end{sqlmini}
\small 500 $\rightarrow$ \textbf{499}: se guardan \textbf{los dos} cambios.
\column{.49\textwidth}
\centering\etiqueta{Luis otra vez}
\begin{sqlmini}[\col]
\dentro{\kw{START TRANSACTION}};\\
\kw{UPDATE} entrada\\
\kw{SET} disponibles = disponibles - 1 \kw{WHERE} id = 1;\\
\kw{INSERT INTO} venta (dni, nombre, correo,\\
\ind precio, referencia\_pago)\\
\kw{VALUES} (\mal{'70123456'}, 'Luis Rojas',\\
\ind 'luis.rojas@example.com', 45.00, 'PAGO-0002');\\
\err{-- ERROR 1062: Duplicate entry '70123456'}\\
\dentro{\kw{ROLLBACK}};
\end{sqlmini}
\small 498 (pendiente) $\rightarrow$ \textbf{499}: se deshacen \textbf{los dos}.
\end{columns}
\vspace{0.3cm}
\concepto{Transacción}{\small Un grupo de cambios que se guardan \textbf{todos juntos} (\texttt{COMMIT}) o \textbf{ninguno} (\texttt{ROLLBACK}). En SQL Server empieza con \texttt{BEGIN TRANSACTION}.}
\end{frame}

\begin{frame}{¿Qué es autocommit?}
\concepto{Autocommit}{\small El gestor hace el \texttt{COMMIT} \textbf{por ti} después de cada instrucción: cada instrucción es una transacción \textbf{por sí sola} y se guarda apenas termina.}
\vspace{0.4cm}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\centering\etiqueta{Con autocommit}\\[2pt]
{\scriptsize\gris{así trabajan MySQL y SQL Server al conectarte}}\\[8pt]
\begin{tikzpicture}
  \node[inst] (a) at (0,0) {\kw{UPDATE} \ldots};
  \node[inst,fill=utpRojo!7,draw=utpRojo!75] (ac) at (2.9,0) {\texttt{COMMIT}\\\normalfont\tiny automático};
  \node[inst] (b) at (0,-1.2) {\kw{INSERT} \ldots};
  \node[inst,fill=utpRojo!7,draw=utpRojo!75] (bc) at (2.9,-1.2) {\texttt{COMMIT}\\\normalfont\tiny automático};
  \draw[flecha] (a)--(ac); \draw[flecha] (b)--(bc);
\end{tikzpicture}\\[6pt]
\small Dos instrucciones = \textbf{dos} transacciones.\\Si el \texttt{INSERT} falla, el \texttt{UPDATE} \textbf{ya se guardó}.
\column{.49\textwidth}
\centering\etiqueta{Con una transacción}\\[2pt]
{\scriptsize\gris{tú decides cuándo se guarda}}\\[8pt]
\begin{tikzpicture}
  \node[inst] (s) at (0,0) {\kw{START}\\\kw{TRANSACTION};};
  \node[inst] (a) at (2.6,0) {\kw{UPDATE} \ldots};
  \node[inst] (b) at (2.6,-1.2) {\kw{INSERT} \ldots};
  \node[inst,fill=utpRojo!7,draw=utpRojo!75] (c) at (5.2,-1.2) {\texttt{COMMIT};\\\normalfont\tiny lo escribes tú};
  \draw[flecha] (s)--(a); \draw[flecha] (a)--(b); \draw[flecha] (b)--(c);
\end{tikzpicture}\\[6pt]
\small Dos instrucciones = \textbf{una} transacción.\\Si el \texttt{INSERT} falla, puedes hacer \texttt{ROLLBACK}.
\end{columns}
\end{frame}

\begin{center}
\begin{tikzpicture}

\begin{frame}{Antes de la práctica: instalar Node.js}
\small La webapp de la práctica está hecha en JavaScript. Para ejecutarla en tu computadora necesitas \textbf{Node.js}. Se instala desde \textbf{PowerShell}:
\vspace{0.2cm}
\begin{columns}[T,onlytextwidth]
\column{.42\textwidth}
\footnotesize
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{@{}c>{\raggedright\arraybackslash}p{4.4cm}@{}}
\npieza{1} & Abre \textbf{PowerShell} \gris{(búscalo en el menú Inicio)}. \\
\npieza{2} & Instala Node.js con \texttt{winget}. Si pregunta, responde \texttt{Y}. \\
\npieza{3} & \textbf{Cierra} PowerShell y ábrelo de nuevo. \\
\npieza{4} & Comprueba con \texttt{node -v} y \texttt{npm -v}. \\
\end{tabular}
\column{.55\textwidth}

\small
\renewcommand{\arraystretch}{1.12}
\begin{tabular}{>{\raggedright\bfseries}p{3.6cm}>{\raggedright\arraybackslash}p{9.4cm}}
\toprule
Autocommit & Cada instrucción se guarda sola, apenas termina. Viene activado en MySQL y SQL Server. \\
Transacción explícita & Tú marcas el inicio (\texttt{START} / \texttt{BEGIN TRANSACTION}) y el fin (\texttt{COMMIT} o \texttt{ROLLBACK}). \\
Transacción implícita & Se abre sola y queda abierta hasta el \texttt{COMMIT} o el \texttt{ROLLBACK}. \\
\texttt{@@TRANCOUNT} & En SQL Server, cuántas transacciones hay abiertas. Se guarda al llegar a 0. \\
SQLite & Una base de datos en un archivo. No necesita un servidor. \\
\bottomrule
\end{tabular}
