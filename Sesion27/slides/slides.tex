\begin{frame}{Revisión: una venta con transacción}
\renewcommand{\arraystretch}{0.85}
\small Vender una entrada son \textbf{dos cambios}: descontar una entrada disponible y guardar la venta.
\vspace{0.15cm}
\begin{columns}[T,onlytextwidth]
\column{.49\textwidth}
\centering\etiqueta{La venta a Luis}
\begin{sqlmini}[\col]
\dentro{\kw{START TRANSACTION}};\\
\kw{UPDATE} entrada\\
\kw{SET} disponibles = disponibles - 1 \kw{WHERE} id = 1;\\
\kw{INSERT INTO} venta (dni, nombre, correo,\\
\ind precio, referencia\_pago)\\
\kw{VALUES} ('70123456', 'Luis Rojas',\\
\ind 'luis.rojas@example.com', 45.00, 'PAGO-0001');\\
\dentro{\kw{COMMIT}};
\end{sqlmini}
\small 500 $\rightarrow$ \textbf{499}: se guardan \textbf{los dos} cambios.
\column{.49\textwidth}
\centering\etiqueta{Luis otra vez}
\begin{sqlmini}[\col]
\dentro{\kw{START TRANSACTION}};\\
\kw{UPDATE} entrada\\
\kw{SET} disponibles = disponibles - 1 \kw{WHERE} id = 1;\\
\kw{INSERT INTO} venta (dni, nombre, correo,\\
\ind precio, referencia\_pago)\\
\kw{VALUES} (\mal{'70123456'}, 'Luis Rojas',\\
\ind 'luis.rojas@example.com', 45.00, 'PAGO-0002');\\
\err{-- ERROR 1062: Duplicate entry '70123456'}\\
\dentro{\kw{ROLLBACK}};
\end{sqlmini}
\small 498 (pendiente) $\rightarrow$ \textbf{499}: se deshacen \textbf{los dos}.
\end{columns}
\vspace{0.3cm}
\concepto{Transacción}{\small Un grupo de cambios que se guardan \textbf{todos juntos} (\texttt{COMMIT}) o \textbf{ninguno} (\texttt{ROLLBACK}). En SQL Server empieza con \texttt{BEGIN TRANSACTION}.}
\end{frame}

\begin{frame}{¿Qué es autocommit?}
\concepto{Autocommit}{\small El gestor hace el \texttt{COMMIT} \textbf{por ti} después de cada instrucción: cada instrucción es una transacción \textbf{por sí sola} y se guarda apenas termina.}
\vspace{0.4cm}
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\centering\etiqueta{Con autocommit}\\[2pt]
{\scriptsize\gris{así trabajan MySQL y SQL Server al conectarte}}\\[8pt]
\begin{tikzpicture}
  \node[inst] (a) at (0,0) {\kw{UPDATE} \ldots};
  \node[inst,fill=utpRojo!7,draw=utpRojo!75] (ac) at (2.9,0) {\texttt{COMMIT}\\\normalfont\tiny automático};
  \node[inst] (b) at (0,-1.2) {\kw{INSERT} \ldots};
  \node[inst,fill=utpRojo!7,draw=utpRojo!75] (bc) at (2.9,-1.2) {\texttt{COMMIT}\\\normalfont\tiny automático};
  \draw[flecha] (a)--(ac); \draw[flecha] (b)--(bc);
\end{tikzpicture}\\[6pt]
\small Dos instrucciones = \textbf{dos} transacciones.\\Si el \texttt{INSERT} falla, el \texttt{UPDATE} \textbf{ya se guardó}.
\column{.49\textwidth}
\centering\etiqueta{Con una transacción}\\[2pt]
{\scriptsize\gris{tú decides cuándo se guarda}}\\[8pt]
\begin{tikzpicture}
  \node[inst] (s) at (0,0) {\kw{START}\\\kw{TRANSACTION};};
  \node[inst] (a) at (2.6,0) {\kw{UPDATE} \ldots};
  \node[inst] (b) at (2.6,-1.2) {\kw{INSERT} \ldots};
  \node[inst,fill=utpRojo!7,draw=utpRojo!75] (c) at (5.2,-1.2) {\texttt{COMMIT};\\\normalfont\tiny lo escribes tú};
  \draw[flecha] (s)--(a); \draw[flecha] (a)--(b); \draw[flecha] (b)--(c);
\end{tikzpicture}\\[6pt]
\small Dos instrucciones = \textbf{una} transacción.\\Si el \texttt{INSERT} falla, puedes hacer \texttt{ROLLBACK}.
\end{columns}
\end{frame}

\seccion{Transacciones en MySQL}
\begin{frame}{MySQL: al conectarte, todo se guarda al instante}
\small Al abrir una conexión, MySQL trabaja en \textbf{autocommit}: cada instrucción se guarda \textbf{sola}, apenas termina.
\vspace{0.5cm}
\begin{center}
\begin{tikzpicture}
  \node[valor] (s) at (-3.6,0) {quedan\\\textbf{\large 499}};
  \node[inst] (a) at (0,0) {\kw{UPDATE} entrada \ldots\\disponibles - 1};
  \node[inst] (b) at (4.2,0) {\kw{ROLLBACK};};
  \draw[flecha] (s)--(a); \draw[flecha] (a)--(b);
  \node[valor] at (0,-1.1) {quedan \textbf{498}\\\gris{ya está guardado}};
  \node[valor] at (4.2,-1.1) {siguen \mal{498}\\\gris{no hay nada que deshacer}};
\end{tikzpicture}
\end{center}
\vspace{0.5cm}
\concepto{Autocommit}{\small Viene activado: \texttt{SELECT @@autocommit;} devuelve \texttt{1}. Para poder deshacer, primero hay que escribir \texttt{START TRANSACTION}.}
\end{frame}

\seccion{Transacciones en SQL Server}
\begin{frame}{SQL Server: al conectarte, todo se guarda al instante}
\small Igual que MySQL, SQL Server trabaja en \textbf{autocommit}: cada instrucción se guarda \textbf{sola}, apenas termina.
\vspace{0.5cm}
\begin{center}
\begin{tikzpicture}
  \node[valor] (s) at (-3.6,0) {quedan\\\textbf{\large 499}};
  \node[inst] (a) at (0,0) {\kw{UPDATE} entrada \ldots\\disponibles - 1};
  \node[inst] (b) at (4.2,0) {\kw{ROLLBACK};};
  \draw[flecha] (s)--(a); \draw[flecha] (a)--(b);
  \node[valor] at (0,-1.1) {quedan \textbf{498}\\\gris{ya está guardado}};
  \node[valor,text width=4.6cm] at (4.2,-1.25) {siguen \mal{498}\\\err{\scriptsize Msg 3903: The ROLLBACK TRANSACTION request has no corresponding BEGIN TRANSACTION.}};
\end{tikzpicture}
\end{center}
\vspace{0.4cm}
\concepto{La diferencia}{\small MySQL no dice nada. SQL Server \textbf{avisa con un error} que no había ninguna transacción que deshacer.}
\end{frame}

\begin{frame}{Variables: guardar un valor para usarlo después}
\concepto{Variable}{\small Un \textbf{nombre} que guarda \textbf{un valor} mientras se ejecuta tu código. Se le da un valor con \texttt{SET}.}
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlchico}[\col]
\kw{SET} @precio = 120;\\
\kw{SET} @descuento = 20;\\
\kw{SELECT} @precio - @descuento \kw{AS} a\_pagar;\\
\com{-- 100}
\end{sqlchico}
\footnotesize Guardar lo que devuelve un \texttt{SELECT}:
\begin{sqlchico}[\col]
\kw{SELECT} valor\_descuento \dentro{\kw{INTO} @descuento}\\
\kw{FROM} cupon \kw{WHERE} codigo = 'NOVIEMBRE20';
\end{sqlchico}
\column{.48\textwidth}
\centering\etiqueta{SQL Server}
\begin{sqlchico}[\col]
\kw{DECLARE} @precio \kw{DECIMAL}(10,2) = 120;\\
\kw{DECLARE} @descuento \kw{DECIMAL}(10,2) = 20;\\
\kw{SELECT} @precio - @descuento \kw{AS} a\_pagar;\\
\com{-- 100.00}
\end{sqlchico}
\footnotesize Guardar lo que devuelve un \texttt{SELECT}:
\begin{sqlchico}[\col]
\kw{SELECT} \dentro{@descuento = valor\_descuento}\\
\kw{FROM} cupon \kw{WHERE} codigo = 'NOVIEMBRE20';
\end{sqlchico}
\end{columns}
\ojo{Dentro de una función de MySQL, las variables se crean con \texttt{DECLARE} y van \textbf{sin} \texttt{@}: \texttt{DECLARE v\_tipo VARCHAR(10);}}
\end{frame}

\begin{frame}{\texttt{IF}: elegir un camino}
\begin{columns}[c,onlytextwidth]
\column{.42\textwidth}
\centering
\begin{tikzpicture}
  \node[decision] (d) at (0,0) {¿el tipo es\\'porcentaje'?};
  \node[paso,text width=2.4cm] (si) at (-1.65,-1.9) {descuento =\\precio $\times$ valor / 100};
  \node[paso,text width=2.1cm] (no) at (1.65,-1.9) {descuento =\\valor};
  \draw[flecha] (d.west) -| node[above,pos=.25,font=\scriptsize\bfseries]{sí} (si.north);
  \draw[flecha] (d.east) -| node[above,pos=.25,font=\scriptsize\bfseries]{no} (no.north);
\end{tikzpicture}\\[8pt]
\small Se ejecuta \textbf{solo uno} de los dos caminos.
\column{.55\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlcode}[\col]
\kw{IF} v\_tipo = 'porcentaje' \mal{\kw{THEN}}\\
\ind\kw{SET} v\_descuento = p\_precio * v\_valor / 100;\\
\kw{ELSE}\\
\ind\kw{SET} v\_descuento = v\_valor;\\
\mal{\kw{END IF}};
\end{sqlcode}
\centering\etiqueta{SQL Server}
\begin{sqlcode}[\col]
\kw{IF} @tipo = 'porcentaje'\\
\ind\kw{SET} @descuento = @precio * @valor / 100;\\
\kw{ELSE}\\
\ind\kw{SET} @descuento = @valor;
\end{sqlcode}
\end{columns}
\analogia{En SQL Server no hay \texttt{THEN} ni \texttt{END IF}. Si un camino tiene \textbf{varias} instrucciones, van entre \texttt{BEGIN} y \texttt{END}.}
\end{frame}

\begin{columns}[c,onlytextwidth]
\column{.58\textwidth}
\begin{sqlmicro}[\col]
\dentro{\kw{DELIMITER} \$\$}\\
\kw{CREATE FUNCTION} precio\_final(p\_precio \kw{DECIMAL}(10,2),\\
\ind\ind\ind\ind\ind\ind\ind\ind\ind\ind p\_cupon\_id \kw{INT})\\
\kw{RETURNS DECIMAL}(10,2)\\
\kw{READS SQL DATA}\\
\kw{BEGIN}\\
\ind\kw{DECLARE} v\_tipo \kw{VARCHAR}(10);\\
\ind\kw{DECLARE} v\_valor \kw{DECIMAL}(10,2);\\
\ind\kw{DECLARE} v\_descuento \kw{DECIMAL}(10,2);\\[1.5pt]
\ind\kw{SELECT} tipo\_descuento, valor\_descuento \kw{INTO} v\_tipo, v\_valor\\
\ind\kw{FROM} cupon \kw{WHERE} id = p\_cupon\_id;\hfill\npieza{1}\\[1.5pt]
\ind\kw{IF} v\_tipo = 'porcentaje' \kw{THEN}\\
\ind\ind\kw{SET} v\_descuento = p\_precio * v\_valor / 100;\\
\ind\kw{ELSE}\\
\ind\ind\kw{SET} v\_descuento = v\_valor;\\
\ind\kw{END IF};\hfill\npieza{2}\\[1.5pt]
\ind\kw{IF} v\_descuento > p\_precio \kw{THEN}\\
\ind\ind\kw{SET} v\_descuento = p\_precio;\\
\ind\kw{END IF};\\[1.5pt]
\ind\kw{RETURN} p\_precio - v\_descuento;\hfill\npieza{3}\\
\kw{END}\dentro{\$\$}\\
\dentro{\kw{DELIMITER} ;}
\end{sqlmicro}
\column{.39\textwidth}
\raggedright\footnotesize
\textbf{\texttt{BEGIN} \ldots{} \texttt{END}}: agrupa varias instrucciones, cada una con su \texttt{;}.\\[5pt]
\textbf{\texttt{DELIMITER \$\$}}: le dice a MySQL que la función termina en \texttt{\$\$}, no en el primer \texttt{;}.\\[5pt]
\textbf{\texttt{READS SQL DATA}}: la función lee una tabla.\\[5pt]
El segundo \texttt{IF} evita que se pague menos de 0.
\vspace{0.2cm}
\begin{sqlmini}[\col]
\kw{SELECT} precio\_final(120, 1),\ \com{-- 108.00}\\
\ind\ind\ind precio\_final(120, 2),\ \com{-- 100.00}\\
\ind\ind\ind precio\_final(30, 5);\ \ \ \com{-- \ \ 0.00}
\end{sqlmini}
{\scriptsize\gris{Cupones 1, 2 y 5. Con \texttt{PRUEBA50} el descuento (50) es mayor que el precio (30): paga 0.}}
\end{columns}
\end{frame}

\begin{frame}{\texttt{IF} con varios caminos: \texttt{ELSEIF}}
\renewcommand{\arraystretch}{0.85}
\small ¿En qué situación está cada cupón? Se revisa \textbf{en orden}: gana la primera condición que se cumple.
\begin{columns}[T,onlytextwidth]
\column{.5\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlmini}[\col]
\kw{IF} v\_estado = 'inactivo' \kw{THEN}\\
\ind\kw{SET} v\_situacion = 'inactivo';\\
\dentro{\kw{ELSEIF}} p\_fecha < v\_desde \kw{THEN}\\
\ind\kw{SET} v\_situacion = 'por empezar';\\
\dentro{\kw{ELSEIF}} v\_hasta \kw{IS NOT NULL}\\
\ind\ind\kw{AND} p\_fecha >= v\_hasta \kw{THEN}\\
\ind\kw{SET} v\_situacion = 'vencido';\\
\kw{ELSE}\\
\ind\kw{SET} v\_situacion = 'vigente';\\
\kw{END IF};
\end{sqlmini}
\centering\etiqueta{SQL Server}
\begin{sqlmini}[\col]
\kw{IF} @estado = 'inactivo' \ldots\\
\dentro{\kw{ELSE IF}} @fecha < @desde \ldots\\
\dentro{\kw{ELSE IF}} @hasta \kw{IS NOT NULL AND} \ldots\\
\kw{ELSE} \ldots
\end{sqlmini}
\column{.47\textwidth}
\begin{sqlmini}[\col]
\kw{SELECT} codigo,\\
\ind situacion\_cupon(id, '2026-11-10') \kw{AS} situacion\\
\kw{FROM} cupon \kw{ORDER BY} id;
\end{sqlmini}
\res{ll}{\cab{codigo} & \cab{situacion}}{BIENVENIDA10 & vigente \\ NOVIEMBRE20 & vigente \\ VERANO15 & vencido \\ NAVIDAD25 & por empezar \\ PRUEBA50 & inactivo \\}
\par\vspace{4pt}
{\centering\normalfont\scriptsize\color{utpGris}La fecha es fija para que todos veamos lo mismo.\\En la aplicación se usa la fecha de hoy.\par}
\end{columns}
\end{frame}
