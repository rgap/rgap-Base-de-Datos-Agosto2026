% ---------------------------------------------------------------------
\begin{frame}{DDL construye, DML llena}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{DDL: el envase}\\[6pt]
\raggedright\small Crea y cambia la \textbf{forma} de las tablas: qué columnas tienen, de qué tipo, con qué reglas.
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{CREATE TABLE} \ldots\\
\kw{ALTER TABLE} \ldots\\
\kw{DROP TABLE} \ldots
\end{sqlcode}
\column{.48\textwidth}
\centering\etiqueta{DML: el contenido}\\[6pt]
\raggedright\small Trabaja con las \textbf{filas}: las agrega, las lee, las cambia y las borra. La tabla sigue igual.
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{INSERT} \ \ \ \kw{SELECT}\\
\kw{UPDATE} \ \ \ \kw{DELETE}\\
\kw{TRUNCATE}
\end{sqlcode}
\end{columns}
\vspace{0.3cm}
\analogia{Un armario es el \textbf{DDL}: sus repisas y cajones. La ropa que metes, sacas o cambias de lugar es el \textbf{DML}. Puedes ordenar la ropa mil veces sin tocar el armario.}
\vspace{0.1cm}
\footnotesize\color{utpGris} DML = \textit{Data Manipulation Language}, lenguaje de manipulación de datos.
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{Cinco palabras para cinco acciones}
\begin{center}
\resizebox{\textwidth}{!}{%
\begin{tikzpicture}
  \node[cajanaranja,text width=2.5cm,font=\scriptsize,align=center] (i)
    {\textbf{\texttt{INSERT}}\\[3pt] \textbf{agrega}\\ filas nuevas\\[3pt] \color{utpGris}llega un producto};
  \node[cajaverde,text width=2.5cm,font=\scriptsize,align=center,right=5mm of i] (s)

  \node[cajanaranja,text width=2.5cm,font=\scriptsize,align=center,right=5mm of s] (u)
    {\textbf{\texttt{UPDATE}}\\[3pt] \textbf{cambia}\\ datos que ya están\\[3pt] \color{utpGris}sube un precio};
  \node[cajanaranja,text width=2.5cm,font=\scriptsize,align=center,right=5mm of u] (d)
    {\textbf{\texttt{DELETE}}\\[3pt] \textbf{borra}\\ las filas que elijas\\[3pt] \color{utpGris}se descontinúa uno};
  \node[cajanaranja,text width=2.5cm,font=\scriptsize,align=center,right=5mm of d] (t)

\end{center}
\vspace{0.5cm}
\small Cuatro de ellas \textbf{cambian} los datos. Solo \texttt{SELECT} se limita a \textbf{mirar}: por eso es la más segura para practicar.
\vspace{0.3cm}

\enlarealidad{Cada vez que compras en línea pasan estas cosas: tu pedido se \textbf{inserta}, el catálogo se \textbf{consulta}, el stock se \textbf{actualiza} y tu carrito, al pagar, se \textbf{borra}.}
\end{frame}

\begin{frame}{Cada valor va a su columna, en orden}
\begin{center}
\begin{tikzpicture}[
  col/.style={draw=utpNegro,thick,rounded corners=1.5pt,fill=utpGrisClaro,font=\ttfamily\small,
    minimum width=2.5cm,minimum height=0.65cm},
  val/.style={draw=utpRojo!75,thick,rounded corners=1.5pt,fill=utpRojo!7,font=\ttfamily\small,
    minimum width=2.5cm,minimum height=0.65cm}]
  \foreach \x/\c/\v [count=\i] in {0/nombre/'Leche', 3/categoria/'bebidas', 6/precio/3.80, 9/stock/20} {
    \node[col] (c\i) at (\x,0) {\c};
    \node[val] (v\i) at (\x,-1.6) {\v};
    \draw[flecha,utpGris] (v\i.north) -- (c\i.south);
  }
  \node[font=\scriptsize,text=utpGris,anchor=east] at (-1.5,0) {columnas};
  \node[font=\scriptsize,text=utpGris,anchor=east] at (-1.5,-1.6) {valores};
  \node[font=\scriptsize,text=utpGris] at (1.5,-2.35) {texto: \textbf{con comillas}};
  \node[font=\scriptsize,text=utpGris] at (7.5,-2.35) {números: \textbf{sin comillas}, con punto decimal};
\end{tikzpicture}
\end{center}
\vspace{0.3cm}
\begin{itemize}\setlength\itemsep{5pt}
  \item El \textbf{primer} valor va a la \textbf{primera} columna, el segundo a la segunda, y así.
  \item Debe haber \textbf{tantos valores como columnas}: ni uno más, ni uno menos.
  \item El texto va entre \textbf{comillas simples} \texttt{'así'}. Las fechas también: \texttt{'2026-09-26'}.
\end{itemize}
\vspace{0.2cm}
\analogia{Es como llenar un formulario en papel: cada dato va en \textbf{su casillero}. Si escribes tu edad donde iba tu nombre, el formulario queda mal aunque «tenga todos los datos».}
\end{frame}

\begin{frame}{\texttt{SELECT}: ¿qué columnas y de qué tabla?}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{sqlcode}[\textwidth]
\com{todas las columnas}\\
\kw{SELECT} \nuevo{*} \kw{FROM} producto;
\end{sqlcode}
\vspace{-0.2cm}
\centering\scriptsize\ttfamily
\begin{tabular}{lllrr}
\toprule
\cabproducto
\midrule
1 & Arroz & abarrotes & 4.50 & 30 \\
2 & Gaseosa & bebidas & 3.00 & 12 \\
3 & Agua & bebidas & 1.50 & 0 \\
4 & Galletas & snacks & 1.20 & 50 \\
\bottomrule
\end{tabular}
\column{.48\textwidth}
\begin{sqlcode}[\textwidth]
\com{solo las que me interesan}\\
\kw{SELECT} \nuevo{nombre, precio} \kw{FROM} producto;
\end{sqlcode}
\vspace{-0.2cm}
\centering\scriptsize\ttfamily
\begin{tabular}{lr}
\toprule
\normalfont\bfseries nombre & \normalfont\bfseries precio \\
\midrule
Arroz & 4.50 \\
Gaseosa & 3.00 \\
Agua & 1.50 \\
Galletas & 1.20 \\
\bottomrule
\end{tabular}
\end{columns}
\vspace{0.4cm}
\small
\begin{itemize}\setlength\itemsep{4pt}
  \item \texttt{*} se lee «\textbf{todas las columnas}». Útil para echar un vistazo rápido.
  \item Con nombres, el resultado trae \textbf{solo esas} columnas y \textbf{en ese orden}.
  \item \texttt{SELECT} \textbf{nunca} cambia la tabla: el resultado es una copia para mirar.
\end{itemize}
\end{frame}

\begin{frame}{\texttt{ORDER BY}: ordenar el resultado}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} nombre, precio \kw{FROM} producto\\
\nuevo{\kw{ORDER BY} precio};
\end{sqlcode}
\vspace{-0.25cm}
\centering\scriptsize\ttfamily
\begin{tabular}{lr}
\toprule
\normalfont\bfseries nombre & \normalfont\bfseries precio \\
\midrule
Galletas & 1.20 \\
Agua & 1.50 \\
Gaseosa & 3.00 \\
Arroz & 4.50 \\
\bottomrule
\end{tabular}\\[6pt]
\normalfont\small De menor a mayor (\texttt{ASC}, el normal).
\column{.48\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} nombre, precio \kw{FROM} producto\\
\nuevo{\kw{ORDER BY} precio \kw{DESC}};
\end{sqlcode}
\vspace{-0.25cm}
\centering\scriptsize\ttfamily
\begin{tabular}{lr}
\toprule
\normalfont\bfseries nombre & \normalfont\bfseries precio \\
\midrule
Arroz & 4.50 \\
Gaseosa & 3.00 \\
Agua & 1.50 \\
Galletas & 1.20 \\
\bottomrule
\end{tabular}\\[6pt]
\normalfont\small De mayor a menor (\texttt{DESC}, descendente).
\end{columns}
\vspace{0.4cm}
\small \textbf{Sin \texttt{ORDER BY}, el orden no está garantizado.} Suele salir en el orden en que se guardó, pero el gestor no lo promete. Si el orden importa, \textbf{pídelo}.
\vspace{0.15cm}

\small Con texto ordena alfabéticamente: \texttt{ORDER BY nombre} da Agua, Arroz, Galletas, Gaseosa.
\end{frame}

\begin{frame}{El orden de las piezas de un \texttt{SELECT}}
\begin{center}
\begin{tikzpicture}
  \pieza{0}{\kw{SELECT} \ \ nombre, precio}{¿Qué \textbf{columnas} quiero ver?}
  \pieza{-0.85}{\kw{FROM} \ \ \ \ producto}{¿De qué \textbf{tabla}?}
  \pieza{-1.7}{\kw{WHERE} \ \ \ stock > 0}{¿Qué \textbf{filas} me quedo?}
  \pieza{-2.55}{\kw{ORDER BY} precio \kw{DESC};}{¿En qué \textbf{orden}?}
\end{tikzpicture}
\end{center}
\vspace{0.35cm}
\small Solo \texttt{SELECT} y \texttt{FROM} son obligatorias. Las demás son \textbf{opcionales}, pero si aparecen, van \textbf{en este orden}. Escribir \texttt{ORDER BY} antes de \texttt{WHERE} da error.
\end{frame}

\small La gaseosa subió de precio. Solo cambia \textbf{esa celda}: el resto de la fila y las demás filas quedan igual.
\column{.52\textwidth}
\centering\scriptsize\ttfamily
\begin{tabular}{lllrr}
\toprule
\cabproducto
\midrule
1 & Arroz & abarrotes & 4.50 & 30 \\
2 & Gaseosa & bebidas & \nuevo{3.50} & 12 \\
3 & Agua & bebidas & 1.50 & 0 \\
4 & Galletas & snacks & 1.20 & 50 \\
\bottomrule
\end{tabular}
\end{columns}
\vspace{0.3cm}
\analogia{Es como corregir con \textbf{corrector líquido}: no arrancas la hoja, solo tapas el dato viejo y escribes el nuevo encima.}
\end{frame}

\begin{frame}{El olvido más caro: \texttt{UPDATE} sin \texttt{WHERE}}
\begin{columns}[c,onlytextwidth]
\column{.46\textwidth}
\begin{sqlcode}[\textwidth]
\kw{UPDATE} producto \kw{SET} precio = 3.50;\\
\com{¡se olvidó el WHERE!}
\end{sqlcode}
\column{.50\textwidth}
\centering\scriptsize\ttfamily
\begin{tabular}{lllrr}
\toprule
\cabproducto
\midrule
1 & Arroz & abarrotes & \mal{3.50} & 30 \\
2 & Gaseosa & bebidas & \mal{3.50} & 12 \\
3 & Agua & bebidas & \mal{3.50} & 0 \\
4 & Galletas & snacks & \mal{3.50} & 50 \\
\bottomrule
\end{tabular}
\end{columns}
\vspace{0.2cm}
\ojo{Sin \texttt{WHERE}, \textbf{todas las filas} reciben el cambio. Toda la bodega pasa a costar S/ 3.50 y los precios anteriores \textbf{se perdieron}: no hay «deshacer».}
\vspace{0.15cm}
\begin{block}{Buena costumbre: primero mira, luego cambia}
\small Prueba el \texttt{WHERE} con un \texttt{SELECT}. Si salen justo las filas que querías, recién haz el \texttt{UPDATE}.
\begin{sqlcode}[.92\textwidth]
\scriptsize
\kw{SELECT} * \kw{FROM} producto \nuevo{\kw{WHERE} id = 2}; \ \ \ \ \com{1. ¿son estas?}\\
\kw{UPDATE} producto \kw{SET} precio = 3.50 \nuevo{\kw{WHERE} id = 2}; \ \com{2. ahora sí}
\end{sqlcode}
\end{block}
\end{frame}

\begin{frame}{\texttt{DELETE}: borrar filas}
\begin{center}
\begin{tikzpicture}
  \pieza{0}{\kw{DELETE FROM} producto}{¿De \textbf{qué tabla}?}
  \pieza{-0.85}{\kw{WHERE} \ \ \ \ \ \ stock = 0;}{¿\textbf{Cuáles} filas se van?}
\end{tikzpicture}
\end{center}
\vspace{0.3cm}
\begin{columns}[c,onlytextwidth]
\column{.44\textwidth}
\small
\begin{itemize}\setlength\itemsep{4pt}
  \item Se borra la \textbf{fila entera}, con todas sus columnas.
  \item No se escriben columnas: \texttt{DELETE nombre FROM\ldots} es un error.
\end{itemize}
\column{.52\textwidth}
\centering\scriptsize\ttfamily
\begin{tabular}{lllrr}
\toprule
\cabproducto
\midrule
1 & Arroz & abarrotes & 4.50 & 30 \\
2 & Gaseosa & bebidas & 3.00 & 12 \\
\fuera{3} & \fuera{Agua} & \fuera{bebidas} & \fuera{1.50} & \fuera{0} \\
4 & Galletas & snacks & 1.20 & 50 \\
\bottomrule
\end{tabular}
\end{columns}
\vspace{0.35cm}
\analogia{¿Y si solo quiero \textbf{vaciar un dato}, no la fila? Eso no es \texttt{DELETE}, es un \texttt{UPDATE}: \texttt{UPDATE producto SET categoria = NULL WHERE id = 3;}}
\end{frame}

% ---------------------------------------------------------------------
\begin{frame}{Antes de borrar, tres cuidados}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}
\centering\etiqueta{1. No olvides el WHERE}\\[6pt]
\raggedright\small \texttt{DELETE FROM producto;} borra \textbf{todas} las filas. La tabla sigue existiendo, pero vacía.
\column{.31\textwidth}
\centering\etiqueta{2. No hay papelera}\\[6pt]
\raggedright\small Lo borrado \textbf{no se recupera} con un clic. Mira antes con un \texttt{SELECT} y el mismo \texttt{WHERE}.
\column{.31\textwidth}
\centering\etiqueta{3. Las claves foráneas}\\[6pt]
\raggedright\small Si una \tbl{venta} apunta al producto, el gestor \textbf{no deja borrarlo} (salvo \texttt{ON DELETE CASCADE}).
\end{columns}
\vspace{0.5cm}
\begin{sqlcode}[.80\textwidth]
\kw{SELECT} * \kw{FROM} producto \nuevo{\kw{WHERE} stock = 0}; \ \ \ \com{1. ¿cuáles se irían?}\\
\kw{DELETE FROM} producto \nuevo{\kw{WHERE} stock = 0}; \ \ \ \ \ \com{2. ahora sí}
\end{sqlcode}
\end{frame}

\seccion{TRUNCATE}

% ---------------------------------------------------------------------
\begin{frame}{\texttt{TRUNCATE}: vaciar la tabla de golpe}
\begin{sqlcode}[.60\textwidth]
\kw{TRUNCATE TABLE} producto;
\end{sqlcode}
\vspace{0.1cm}
\begin{columns}[c,onlytextwidth]
\column{.52\textwidth}
\small
\begin{itemize}\setlength\itemsep{5pt}
  \item Borra \textbf{todas} las filas, siempre. No acepta \texttt{WHERE}.
  \item La tabla \textbf{sigue existiendo}, con sus columnas y reglas.
  \item El autoincremento \textbf{vuelve a empezar} en 1.
  \item Es \textbf{más rápido} que \texttt{DELETE} en tablas grandes.
\end{itemize}
\column{.44\textwidth}
\centering\tbl{producto}\\[4pt]
\scriptsize\ttfamily
\begin{tabular}{lllrr}
\toprule
\cabproducto
\midrule
\multicolumn{5}{c}{\normalfont\color{utpGris}\textit{(vacía)}} \\
\bottomrule
\end{tabular}
\end{columns}
\vspace{0.2cm}
\analogia{\texttt{DELETE} es borrar renglones de un cuaderno \textbf{con borrador}, uno por uno, eligiendo cuáles. \texttt{TRUNCATE} es \textbf{arrancar todas las hojas escritas}: el cuaderno queda como nuevo y vuelves a la página 1.}
\vspace{0.1cm}
\footnotesize\color{utpGris} Funciona igual en MySQL y SQL Server. SQLite no lo tiene: ahí se usa \texttt{DELETE FROM producto;}.
\end{frame}

\begin{frame}{Resumen: las cinco instrucciones}
\begin{center}
\small
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{>{\leavevmode\color{utpRojo}\ttfamily\bfseries}p{2.0cm}>{\raggedright\arraybackslash}p{2.6cm}>{\ttfamily\scriptsize}p{5.4cm}>{\centering\arraybackslash}p{1.8cm}}
\toprule
\normalfont\bfseries Instrucción & \bfseries Qué hace & \normalfont\small\bfseries Forma básica & \bfseries ¿Cambia datos? \\
\midrule
INSERT & agrega filas & INSERT INTO t (a, b) VALUES (1, 2); & sí \\
SELECT & lee filas & SELECT a, b FROM t WHERE \ldots; & \textbf{no} \\
UPDATE & cambia valores & UPDATE t SET a = 1 WHERE \ldots; & sí \\
DELETE & borra filas elegidas & DELETE FROM t WHERE \ldots; & sí \\
TRUNCATE & vacía la tabla & TRUNCATE TABLE t; & sí \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.4cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}\small \textbf{\texttt{WHERE} es el mismo} en \texttt{SELECT}, \texttt{UPDATE} y \texttt{DELETE}: aprendes a filtrar una vez y sirve para las tres.
\column{.48\textwidth}\small \textbf{La regla de oro:} antes de un \texttt{UPDATE} o \texttt{DELETE}, prueba su \texttt{WHERE} con un \texttt{SELECT}.
\end{columns}
\end{frame}
