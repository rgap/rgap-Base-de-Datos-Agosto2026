\begin{frame}{La misma consulta, una y otra vez}
\begin{columns}[c,onlytextwidth]
\column{.46\textwidth}
\tablaproducto
\column{.50\textwidth}
\small El encargado de bebidas pide esto \textbf{todos los días}:
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{SELECT} nombre, precio\\
\kw{FROM} producto\\
\kw{WHERE} categoria = 'bebidas';
\end{sqlcode}
\vspace{-0.2cm}
\small Escribirla cada vez \textbf{cansa}, y es fácil \textbf{equivocarse}.
\end{columns}
\vspace{0.3cm}
\concepto{La idea}{\small Le ponemos \textbf{un nombre} a esa consulta y el gestor \textbf{la guarda}. Eso es una \textbf{vista}.}
\vspace{0.1cm}
\analogia{Es como \textbf{guardar un contacto}: en vez de marcar el número completo cada vez, buscas «Mamá».}
\end{frame}

\midrule

\bottomrule
\end{tabular}
\end{columns}
\vspace{0.35cm}
\small El \texttt{CREATE VIEW} se escribe \textbf{una sola vez}. Es igual en MySQL, SQL Server y SQLite.
\end{frame}

\begin{frame}{\texttt{DROP VIEW}: borrar la vista, no los datos}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{sqlcode}[\textwidth]
\nuevo{\kw{DROP VIEW}} bebidas;
\end{sqlcode}
\vspace{-0.2cm}
Se borra \textbf{la consulta guardada}.\\[3pt]
La tabla \tbl{producto} sigue con \textbf{todas sus filas}.
\column{.48\textwidth}
\begin{sqlcode}[\textwidth]
\kw{DROP TABLE} producto;
\end{sqlcode}
\vspace{-0.2cm}
Se borra \textbf{la tabla con sus datos}.\\[3pt]
Esto sí es grave.
\end{columns}
\vspace{0.5cm}
\analogia{Borrar un \textbf{contacto} del celular no desaparece a la persona. Borrar una vista no desaparece los datos.}
\vspace{0.2cm}
\small ¿Y si quiero \textbf{cambiar} una vista? Lo más simple: la \textbf{borras} y la \textbf{creas de nuevo}.
\end{frame}

\begin{frame}{¿Cuándo es útil una vista?}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}
\centering\etiqueta{Para no repetir}\\[6pt]
\raggedright\small Una consulta que se usa seguido se escribe \textbf{una vez}.
\column{.31\textwidth}
\centering\etiqueta{Para simplificar}\\[6pt]
\raggedright\small Detrás puede haber algo largo; quien la usa ve algo \textbf{sencillo}.
\column{.31\textwidth}
\centering\etiqueta{Para ocultar}\\[6pt]
\raggedright\small Muestra solo \textbf{algunas columnas}: el precio sí, el costo no.
\end{columns}
\vspace{0.6cm}
\begin{alertblock}{¿Y cuándo no?}
\small Si la consulta la vas a usar \textbf{una sola vez}, no hace falta una vista.
\end{alertblock}
\vspace{0.2cm}
\enlarealidad{En un colegio, la secretaría usa una vista con el \textbf{nombre y el teléfono} de cada alumno. Las notas y los pagos no aparecen ahí.}
\end{frame}

\begin{frame}{DBML: describir una base de datos con texto}
\begin{columns}[c,onlytextwidth]
\column{.50\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{Table} cliente \{\\
\ind id int [pk]\\
\ind nombre varchar(80)\\
\ind correo varchar(120)\\
\}\\[5pt]
\kw{Table} pedido \{\\
\ind id int [pk]\\
\ind fecha date\\
\ind cliente\_id int\\
\}\\[5pt]
\kw{Ref}: pedido.cliente\_id > cliente.id
\end{sqlcode}
\column{.46\textwidth}
\centering\diagtiendav\\[4pt]
{\scriptsize\color{utpGris}el diagrama que sale de ese texto}
\end{columns}
\vspace{0.15cm}
\small \textbf{DBML} es un lenguaje pequeño y gratuito, hecho solo para \textbf{describir tablas y sus relaciones}. Se lee casi sin haberlo estudiado.
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{¿Por qué DBML, si ya sabemos SQL?}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{SQL}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{CREATE TABLE} pedido (\\
\ind id INT \kw{PRIMARY KEY},\\
\ind fecha DATE \kw{NOT NULL},\\
\ind cliente\_id INT,\\
\ind \kw{FOREIGN KEY} (cliente\_id)\\
\ind\ind \kw{REFERENCES} cliente(id)\\
);
\end{sqlcode}
\column{.48\textwidth}
\centering\etiqueta{DBML}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{Table} pedido \{\\
\ind id int [pk]\\
\ind fecha date [not null]\\
\ind cliente\_id int\\
\}\\[5pt]
\kw{Ref}: pedido.cliente\_id > cliente.id
\end{sqlcode}
\end{columns}
\vspace{0.1cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item \textbf{Es más corto:} menos palabras y sin comas que se olvidan.
  \item \textbf{Es uno solo:} el SQL cambia de un gestor a otro; el DBML no.
  \item \textbf{Se vuelve diagrama al instante:} sirve para conversar el diseño \textbf{antes} de crear nada.
\end{itemize}
\end{frame}

  \draw[flecha] (d.east)--(g.west);
  \draw[flecha] (d.east)--(s.west);
  \draw[flecha] (s)--(b) node[midway,above,font=\tiny] {se ejecuta};
\end{tikzpicture}
\end{center}
\vspace{0.15cm}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\etiqueta{DBML sí}\\[5pt]
\raggedright\small
\begin{itemize}\setlength\itemsep{4pt}
  \item Describe tablas y relaciones.
  \item Genera el diagrama.
  \item Genera el \texttt{CREATE TABLE}.
\end{itemize}
\column{.48\textwidth}
\etiqueta{DBML no}\\[5pt]
\raggedright\small
\begin{itemize}\setlength\itemsep{4pt}
  \item \textbf{No crea} la base de datos.
  \item \textbf{No guarda} datos.
  \item \textbf{No consulta}: no hay \texttt{SELECT}.
\end{itemize}
\end{columns}
\vspace{0.3cm}
\analogia{DBML es el \textbf{plano} de la casa; SQL es \textbf{la obra}. El plano ayuda a pensar, pero nadie vive en un plano.}
\end{frame}

\begin{frame}{Las reglas entre corchetes}

increment & AUTO\_INCREMENT & el gestor numera solo \\
not null & NOT NULL & no se puede dejar vacío \\
unique & UNIQUE & no se puede repetir \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.2cm}
\begin{sqlcode}[.60\textwidth]
id int [\nuevo{pk}, \nuevo{increment}]
\end{sqlcode}
\vspace{-0.1cm}
Son las \textbf{mismas reglas} que ya conoces, más cortas. Si hay varias, se separan con \textbf{comas}.
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{Las relaciones: \texttt{Ref}}
\begin{sqlcode}[.70\textwidth]
\nuevo{\kw{Ref}:} pedido.cliente\_id \nuevo{>} cliente.id
\end{sqlcode}
\vspace{-0.1cm}
\begin{columns}[c,onlytextwidth]
\column{.50\textwidth}
\small
\begin{itemize}\setlength\itemsep{6pt}
  \item Se lee: «\texttt{cliente\_id} de \tbl{pedido} \textbf{apunta} al \texttt{id} de \tbl{cliente}».
  \item Es la \textbf{clave foránea}, en una sola línea.
\end{itemize}
\column{.47\textwidth}
\centering\diagtienda\\[5pt]
{\scriptsize\color{utpGris}un cliente, muchos pedidos}
\end{columns}
\vspace{0.25cm}
\begin{center}
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{>{\leavevmode\color{utpRojo}\ttfamily\bfseries}c>{\raggedright\arraybackslash}p{3.0cm}>{\ttfamily\small}p{6.6cm}}
\toprule
\normalfont\color{utpNegro}\bfseries Símbolo & \bfseries Relación & \normalfont\bfseries Ejemplo \\
\midrule
> & muchos a uno & pedido.cliente\_id > cliente.id \\
- & uno a uno & usuario.id - perfil.usuario\_id \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.1cm}
\small \textbf{Truco:} el \texttt{>} se abre hacia el lado «\textbf{muchos}», igual que la pata de gallo.
\end{frame}

\seccion{DDL en SQLite}
% ---------------------------------------------------------------------
\begin{frame}{¿Por qué SQLite?}
\begin{center}
\begin{tikzpicture}
  \node[caja,text width=2.3cm] (p1) {Tu programa};
  \node[cajaverde,text width=2.7cm,right=9mm of p1] (s1) {\textbf{Servidor}\\[-1pt]{\scriptsize MySQL, SQL Server}};
  \node[caja,text width=1.9cm,right=9mm of s1] (d1) {Datos};
  \draw[flechavuelta] (p1)--(s1); \draw[flechavuelta] (s1)--(d1);
  \node[caja,text width=2.3cm,below=6mm of p1] (p2) {Tu programa};
  \node[cajanaranja,text width=2.7cm,right=9mm of p2] (d2) {\textbf{Un archivo}\\[-1pt]{\scriptsize \texttt{tienda.db}}};
  \draw[flechavuelta] (p2)--(d2);
  \node[font=\scriptsize,text=utpGris,anchor=west] at ([xshift=4mm]d1.east) {hay que instalar y encender};
  \node[font=\scriptsize,text=utpGris,anchor=west] at ([xshift=4mm]d2.east) {\textbf{SQLite}: sin servidor};
\end{tikzpicture}
\end{center}
\vspace{0.2cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item Toda la base de datos es \textbf{un solo archivo}: se copia y se envía como una foto.
  \item \textbf{No se instala} nada ni se crean usuarios.
  \item Está \textbf{en todas partes}: en tu celular, en tu navegador y dentro de Python.
\end{itemize}
\vspace{0.25cm}
\ojo{Por ser tan pequeño, SQLite \textbf{simplifica}. Las instrucciones ya las conoces: solo veremos \textbf{qué cambia}.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{CREATE TABLE} en SQLite}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{MySQL}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{CREATE TABLE} cliente (\\
\ind id INT \kw{PRIMARY KEY} \kw{AUTO\_INCREMENT},\\
\ind nombre VARCHAR(80) \kw{NOT NULL},\\
\ind registro DATE\\
);
\end{sqlcode}
\column{.48\textwidth}
\centering\etiqueta{SQLite}
\begin{sqlcode}[\textwidth]
\scriptsize
\kw{CREATE TABLE} cliente (\\
\ind id \nuevo{INTEGER} \kw{PRIMARY KEY} \nuevo{\kw{AUTOINCREMENT}},\\
\ind nombre \nuevo{TEXT} \kw{NOT NULL},\\
\ind registro \nuevo{TEXT}\\
);
\end{sqlcode}
\end{columns}
\vspace{0.1cm}
\begin{itemize}\setlength\itemsep{6pt}
  \item \textbf{Pocos tipos:} \texttt{INTEGER} para números enteros, \texttt{REAL} para decimales y \texttt{TEXT} para todo lo demás, incluso las fechas.
  \item \textbf{No hay} \texttt{CREATE DATABASE}: la base de datos \textbf{es el archivo}.
  \item Lo demás (\texttt{PRIMARY KEY}, \texttt{NOT NULL}, \texttt{UNIQUE}) se escribe \textbf{igual}.
\end{itemize}
\vspace{0.15cm}
\small\color{utpGris} \textbf{¿Por qué?} SQLite quiere ser pequeño y simple: con tres tipos le alcanza.
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{FOREIGN KEY} en SQLite: hay que encenderla}
\begin{columns}[c,onlytextwidth]
\column{.52\textwidth}
\begin{sqlcode}[\textwidth]
\scriptsize
\nuevo{\kw{PRAGMA} foreign\_keys = ON;}\\[5pt]
\kw{CREATE TABLE} pedido (\\
\ind id INTEGER \kw{PRIMARY KEY},\\
\ind fecha TEXT,\\
\ind cliente\_id INTEGER,\\
\ind \kw{FOREIGN KEY} (cliente\_id)\\
\ind\ind \kw{REFERENCES} cliente(id)\\
);
\end{sqlcode}
\column{.44\textwidth}
\begin{itemize}\setlength\itemsep{8pt}
  \item Se escribe \textbf{igual} que en MySQL.
  \item Pero viene \textbf{apagada}: hay que encenderla con el \texttt{PRAGMA}.
\end{itemize}
\end{columns}
\vspace{0.15cm}
\begin{center}
\small
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{>{\raggedright\arraybackslash}p{6.0cm}>{\raggedright\arraybackslash}p{6.0cm}}
\toprule
\bfseries Un pedido del cliente 99, que no existe & \bfseries Qué pasa \\
\midrule
Sin el \texttt{PRAGMA} & \mal{Se guarda}, y apunta a nadie \\
Con el \texttt{PRAGMA} & SQLite lo \textbf{rechaza} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.15cm}
\analogia{Es como una \textbf{alarma} que viene instalada pero apagada: si no la enciendes, no avisa.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{ALTER TABLE} en SQLite: solo lo básico}
\begin{columns}[T,onlytextwidth]
\column{.52\textwidth}
\centering\etiqueta{Sí se puede}
\begin{sqlcode}[\textwidth]
\scriptsize
\com{agregar una columna}\\
\kw{ALTER TABLE} cliente\\
\ind \nuevo{\kw{ADD COLUMN}} telefono TEXT;\\[5pt]
\com{cambiarle el nombre a una columna}\\
\kw{ALTER TABLE} cliente\\
\ind \nuevo{\kw{RENAME COLUMN}} correo \kw{TO} email;
\end{sqlcode}
\column{.44\textwidth}
\centering\etiqueta{No se puede}\\[10pt]
\raggedright
\begin{itemize}\setlength\itemsep{8pt}
  \item \textbf{Cambiar el tipo} de una columna.
  \item \textbf{Agregar una clave foránea} a una tabla que ya existe.
\end{itemize}
\end{columns}
\vspace{0.3cm}
\ojo{En SQLite, las claves foráneas se escriben \textbf{al crear la tabla}. Después ya no se pueden agregar.}
\vspace{0.1cm}
\small\color{utpGris} \textbf{¿Por qué?} Para seguir siendo pequeño, SQLite solo trae los cambios más sencillos.
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{¿Y si necesito justo lo que no se puede?}
Se \textbf{rehace la tabla}, como quien pasa un cuaderno en limpio:
\vspace{0.4cm}
\begin{center}
\begin{tikzpicture}
  \tikzset{paso/.style={cajaverde,font=\small,text width=2.45cm,minimum height=1.35cm}}

\vspace{0.5cm}
\concepto{Lo más fácil mientras aprendes}{\small Corrige el \texttt{CREATE TABLE} en tu archivo \texttt{.sql} y \textbf{crea la base de nuevo}. Para eso sirve el archivo que veremos al final.}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{\texttt{DROP TABLE} en SQLite}
\begin{sqlcode}[.60\textwidth]
\kw{DROP TABLE} pedido;
\end{sqlcode}
\vspace{-0.1cm}
Se escribe \textbf{igual} que en MySQL: borra la tabla \textbf{con todos sus datos}.
\vspace{0.3cm}
\begin{center}
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{>{\raggedright\arraybackslash}p{3.8cm}>{\ttfamily\small}p{4.1cm}>{\raggedright\arraybackslash}p{4.3cm}}
\toprule
\bfseries Quiero\ldots & \normalfont\etiqueta{MySQL} & \etiqueta{SQLite} \\
\midrule
Borrar una tabla & DROP TABLE pedido; & \texttt{\small DROP TABLE pedido;} \\
Vaciar una tabla & TRUNCATE TABLE pedido; & \texttt{\small DELETE FROM pedido;} \\
Borrar toda la base & DROP DATABASE tienda; & \textbf{borrar el archivo} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.25cm}
\ojo{Borra primero la tabla que \textbf{apunta} (\tbl{pedido}) y después la \textbf{apuntada} (\tbl{cliente}).}
\end{frame}

\begin{frame}{Un archivo \texttt{.sql}: instrucciones guardadas}
\begin{columns}[c,onlytextwidth]
\column{.54\textwidth}
\etiqueta{tienda.sql}
\vspace{-0.15cm}
\begin{sqlcode}[\textwidth]
\scriptsize
\com{Tienda: tablas y datos de prueba}\\[4pt]
\kw{CREATE TABLE} cliente (\\
\ind id INTEGER \kw{PRIMARY KEY},\\
\ind nombre TEXT \kw{NOT NULL}\\
);\\[4pt]
\kw{INSERT INTO} cliente (nombre) \kw{VALUES} ('Ana');\\
\kw{INSERT INTO} cliente (nombre) \kw{VALUES} ('Luis');
\end{sqlcode}
\column{.42\textwidth}
\begin{itemize}\setlength\itemsep{8pt}
  \item Es un archivo de \textbf{texto} con instrucciones SQL.
  \item El gestor las ejecuta \textbf{de arriba hacia abajo}.
  \item Se abre con \textbf{cualquier editor}.
\end{itemize}
\end{columns}
\vspace{0.3cm}
Es lo mismo que has escrito hasta hoy, pero \textbf{guardado} para no perderlo.
\end{frame}

\begin{frame}{¿Para qué sirve un archivo \texttt{.sql}?}
\begin{columns}[T,onlytextwidth]
\column{.31\textwidth}
\centering\etiqueta{Repetir}\\[6pt]
\raggedright\small Crea la base completa \textbf{desde cero}, las veces que haga falta.
\column{.31\textwidth}
\centering\etiqueta{Compartir}\\[6pt]
\raggedright\small Lo envías y la otra persona obtiene \textbf{la misma} base.
\column{.31\textwidth}
\centering\etiqueta{Respaldar}\\[6pt]
\raggedright\small Queda una \textbf{copia} de las tablas y de los datos.
\end{columns}
\vspace{0.5cm}
\begin{center}

\end{frame}
