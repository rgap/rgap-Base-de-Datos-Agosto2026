\begin{frame}{¿Qué guarda un archivo \texttt{.bak}?}
\small En SQL Server, cada base son \textbf{dos archivos} en el disco.
\vspace{0.2cm}
\begin{center}
\begin{tikzpicture}
  \node[arch,text width=3cm] (m) at (0,.7) {agenda.mdf\\\normalfont\scriptsize los \textbf{datos}: las filas};

  \node[arch,cajanaranja,minimum height=2.35cm] (b) at (7,0) {agenda.bak};
  \draw[flecha] (m.east) -- (b.west |- m.east);
  \draw[flecha] (l.east) -- (b.west |- l.east);
  \node[font=\ttfamily\scriptsize,fill=white] at (3.6,0) {BACKUP DATABASE};
\end{tikzpicture}
\end{center}
\vspace{0.1cm}
\small \texttt{BACKUP DATABASE} copia los dos, por páginas (bloques de 8 KB). En SSMS, pestaña \textbf{Messages}:
\begin{sqlmicro}[.92\textwidth]
Processed 384 pages for database 'agenda', file \dentro{'agenda'} on file 1.\\
Processed 2 pages for database 'agenda', file \dentro{'agenda\_log'} on file 1.\\
BACKUP DATABASE successfully processed 386 pages in 0.025 seconds (120.468 MB/sec).
\end{sqlmicro}
\vspace{-0.05cm}
\concepto{El archivo .bak}{\small La copia de los archivos de la base, \textbf{tal como están en el disco}. Solo SQL Server lo abre.}
\end{frame}

\begin{frame}{Copia lógica o copia física}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\centering\etiqueta{Lógica: agenda.sql}\par\vspace{2pt}
\begin{sqlmicro}[\col]
\kw{CREATE TABLE} \bq contacto\bq{} ( \ldots{} );\\
\kw{INSERT INTO} \bq contacto\bq{} \kw{VALUES} (1,'Ana Torres',\ldots);
\end{sqlmicro}
\vspace{-0.15cm}
\small Guarda las \textbf{instrucciones} para crear la base otra vez.
\column{.48\textwidth}
\centering\etiqueta{Física: agenda.bak}\par\vspace{2pt}
\begin{sqlmicro}[\col]
\gris{(binario: las páginas de agenda.mdf}\\
\gris{y de agenda\_log.ldf)}
\end{sqlmicro}
\vspace{-0.15cm}
\small Guarda los \textbf{archivos} de la base, tal como están.
\end{columns}
\vspace{0.1cm}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.12}
\begin{tabular}{@{}>{\bfseries\raggedright}p{3.3cm}>{\raggedright}p{4.6cm}>{\raggedright\arraybackslash}p{4.6cm}@{}}
\toprule
 & Lógica (\texttt{.sql}) & Física (\texttt{.bak}) \\
\midrule
¿Se puede leer? & Sí, con el Bloc de notas & No \\
Volver a la copia & Se ejecuta cada instrucción & Se copian los archivos de vuelta \\
Una base muy grande & Lento & Rápido \\
¿Sirve en otro gestor? & Sí, con algunos cambios & No: solo SQL Server \\
\bottomrule
\end{tabular}
\end{center}
\vspace{-0.05cm}
\concepto{¿Cuál conviene?}{\small Para mover datos o una base pequeña: la \textbf{lógica}. Para una base grande que debe volver rápido: la \textbf{física}.}

\begin{frame}{SQL Server: sacar las tres copias}
\small La agenda de la semana: el domingo, la completa; el lunes Marco registra a Rosa; el martes, a Pedro.
\vspace{0.05cm}
\begin{sqlmini}[.95\textwidth]
\kw{ALTER DATABASE} agenda \kw{SET RECOVERY} \dentro{\kw{FULL}};\hfill\npieza{1}\\[4pt]
\kw{BACKUP DATABASE} agenda \kw{TO DISK} = 'agenda\_completa.bak' \kw{WITH INIT};\hfill\npieza{2}\\[4pt]
\kw{INSERT INTO} agenda.dbo.contacto (nombre, correo) \kw{VALUES} (N'Rosa Paz', 'rosa@correo.pe');\\
\kw{BACKUP DATABASE} agenda \kw{TO DISK} = 'agenda\_diferencial.bak' \kw{WITH} \dentro{\kw{DIFFERENTIAL}}, \kw{INIT};\hfill\npieza{3}\\[4pt]
\kw{INSERT INTO} agenda.dbo.contacto (nombre, correo) \kw{VALUES} (N'Pedro Huamán', 'pedro@correo.pe');\\
\kw{BACKUP} \dentro{\kw{LOG}} agenda \kw{TO DISK} = 'agenda\_log.trn' \kw{WITH INIT};\hfill\npieza{4}
\end{sqlmini}
\vspace{-0.1cm}
\begin{columns}[T,onlytextwidth]
\column{.5\textwidth}
\footnotesize\raggedright
\npieza{1} \textbf{Modelo de recuperación}: con \texttt{SIMPLE}, el log se vacía solo; con \texttt{FULL}, guarda cada cambio hasta que se copia.\\[4pt]
\npieza{2} Completa.\quad \npieza{3} Diferencial.\quad \npieza{4} Del log.
\column{.46\textwidth}
\footnotesize Sin el paso 1, en una base \texttt{SIMPLE}:\\[2pt]
\errchico{Msg 4208, Level 16, State 1, Line 1\\
The statement BACKUP LOG is not allowed while the recovery model is SIMPLE. Use BACKUP DATABASE or change the recovery model using ALTER DATABASE.}
\end{columns}

{\centering\scriptsize\gris{Todo está en \texttt{ejemplos/agenda/copias\_sqlserver.sql}, parte 4.}\par}
\end{frame}

\begin{frame}{MySQL: crear un usuario}
\begin{sqlchico}[.82\textwidth]
\kw{CREATE USER} 'app\_agenda'@'localhost' \kw{IDENTIFIED BY} 'Agenda2026';
\end{sqlchico}
\vspace{-0.05cm}
{\centering\footnotesize\npieza{1} \texttt{'app\_agenda'}: el nombre\qquad
\npieza{2} \texttt{'localhost'}: solo desde esta computadora\qquad
\npieza{3} la contraseña\par}
\vspace{0.3cm}
\small Un usuario nuevo puede conectarse, pero \textbf{no puede hacer nada}. Conectado como \texttt{app\_agenda}:
\vspace{0.1cm}
\begin{columns}[T,onlytextwidth]
\column{.36\textwidth}
\begin{sqlchico}[\col]
\kw{SHOW DATABASES};
\end{sqlchico}
\resg{l}{\cab{Database}}{information\_schema \\ performance\_schema \\}
\vspace{3pt}
{\centering\footnotesize No ve la base \texttt{agenda}.\par}
\column{.6\textwidth}
\begin{sqlchico}[\col]
\kw{SELECT} nombre \kw{FROM} agenda.contacto;
\end{sqlchico}
\vspace{-6pt}
{\raggedright\errlinea{ERROR 1142 (42000): SELECT command denied to user 'app\_agenda'@'localhost' for table 'contacto'}\par}
\end{columns}
\vspace{0.15cm}
{\centering\scriptsize\gris{Para conectarte como \texttt{app\_agenda}: \texttt{.\textbackslash mysql -u app\_agenda -p}, o en Workbench una conexión nueva con ese \textbf{Username}.}\\[2pt]
\gris{Los pasos de esta sección están en \texttt{ejemplos/agenda/usuarios\_mysql.sql}, en el mismo orden.}\par}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{MySQL: dar permisos con \texttt{GRANT}}
\small \texttt{GRANT} \textit{permisos} \texttt{ON} \textit{dónde} \texttt{TO} \textit{quién}. La aplicación necesita cuatro:
\vspace{0.05cm}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.2}
\begin{tabular}{@{}>{\ttfamily}l>{\raggedright}p{2.6cm}>{\raggedright\arraybackslash}p{6cm}@{}}
\toprule
\normalfont\bfseries Permiso & \bfseries Deja\ldots & \bfseries En la agenda \\
\midrule
SELECT & leer filas & ver los contactos y los grupos \\
INSERT & agregar filas & registrar un contacto o un grupo \\
UPDATE & cambiar filas & editar un contacto, marcarlo como favorito \\
DELETE & borrar filas & eliminar un contacto o un grupo \\
\bottomrule
\end{tabular}
\end{center}
\vspace{-0.1cm}
\begin{sqlchico}[.97\textwidth]
\kw{GRANT} \dentro{\kw{SELECT}, \kw{INSERT}, \kw{UPDATE}, \kw{DELETE}} \kw{ON} \dentro{agenda.*} \kw{TO} 'app\_agenda'@'localhost';\\[3pt]
\kw{SHOW GRANTS FOR} 'app\_agenda'@'localhost';
\end{sqlchico}
\vspace{-0.05cm}
\begin{center}
{\scriptsize\color{utpGris}resultado}\\[2pt]
{\fontsize{6.5}{7.8}\selectfont\ttfamily
\begin{tabular}{@{}l@{}}
\toprule
\cab{Grants for app\_agenda@localhost} \\
\midrule
GRANT USAGE ON *.* TO \bq app\_agenda\bq @\bq localhost\bq \\
GRANT SELECT, INSERT, UPDATE, DELETE ON \bq agenda\bq .* TO \bq app\_agenda\bq @\bq localhost\bq \\
\bottomrule
\end{tabular}\par}
\vspace{2pt}
{\scriptsize\gris{\texttt{agenda.*}: todas las tablas de la base. \texttt{USAGE}: puede conectarse, nada más.}\par}
\end{center}
\end{frame}
% ---------------------------------------------------------------------
\begin{frame}{¿Dónde se da el permiso?}
\vspace{-0.1cm}
\begin{center}
\footnotesize
\scriptsize
\renewcommand{\arraystretch}{1.1}
\begin{tabular}{@{}>{\bfseries}l>{\ttfamily}l>{\raggedright\arraybackslash}p{5.2cm}@{}}
\toprule
Nivel & \normalfont\bfseries Se escribe & \bfseries Ejemplo \\
\midrule
Todo el servidor & *.* & Casi nunca: es como ser \texttt{root}. \\
Una base & agenda.* & La aplicación: \texttt{app\_agenda}. \\
Una tabla & agenda.contacto & Solo los contactos, no los grupos. \\
Unas columnas & SELECT (nombre, telefono) & Recepción: solo nombre y teléfono. \\
\bottomrule
\end{tabular}
\end{center}
\vspace{-0.05cm}
\begin{sqlmini}[.92\textwidth]
\kw{CREATE USER} 'recepcion'@'localhost' \kw{IDENTIFIED BY} 'Recepcion2026';\\
\kw{GRANT SELECT} \dentro{(nombre, telefono)} \kw{ON} agenda.contacto \kw{TO} 'recepcion'@'localhost';
\end{sqlmini}
\vspace{-0.1cm}
\small Conectado como \texttt{recepcion}:
\begin{columns}[T,onlytextwidth]
\column{.45\textwidth}
\begin{sqlmicro}[\col]
\kw{SELECT} nombre, telefono \kw{FROM} contacto;
\end{sqlmicro}
\vspace{-6pt}
\centering{\scriptsize\ttfamily\begin{tabular}{ll}\toprule\cab{nombre} & \cab{telefono}\\\midrule Ana Torres & 987654321 \\ Luis Ramos & 912345678 \\ Carla Díaz & NULL \\\bottomrule\end{tabular}}\par
\column{.52\textwidth}
\begin{sqlmicro}[\col]
\kw{SELECT} nombre, \mal{correo} \kw{FROM} contacto;
\end{sqlmicro}
\vspace{-6pt}
{\raggedright\errlinea{ERROR 1143 (42000): SELECT command denied to user 'recepcion'@'localhost' for column 'correo' in table 'contacto'}\par}
\end{columns}
\end{frame}

\begin{frame}{Roles: permisos con un nombre}
\small Si varias personas necesitan los mismos permisos, se crean \textbf{una vez}, en un \textbf{rol}.
\vspace{0.05cm}
\begin{columns}[T,onlytextwidth]
\column{.56\textwidth}
\begin{sqlmicro}[\col]
\kw{CREATE ROLE} 'solo\_lectura';\hfill\npieza{1}\\
\kw{GRANT SELECT ON} agenda.* \kw{TO} 'solo\_lectura';\\[4pt]
\kw{CREATE USER} 'reporte'@'localhost'\hfill\npieza{2}\\
\ind\kw{IDENTIFIED BY} 'Reporte2026';\\
\kw{GRANT} 'solo\_lectura' \kw{TO} 'reporte'@'localhost';\\[4pt]
\kw{SET DEFAULT ROLE} 'solo\_lectura'\hfill\npieza{3}\\
\ind\kw{TO} 'reporte'@'localhost';
\end{sqlmicro}
\vspace{-0.1cm}
\footnotesize\raggedright
\npieza{1} El rol y sus permisos.\quad \npieza{2} El usuario recibe el rol.\\
\npieza{3} El rol se \textbf{activa} al conectarse.
\column{.4\textwidth}
\footnotesize\textbf{Sin} el paso 3, conectado como \texttt{reporte}:
\begin{sqlmicro}[\col]
\kw{SELECT} \kw{CURRENT\_ROLE}();
\end{sqlmicro}
\vspace{-6pt}
\resg{l}{\cab{CURRENT\_ROLE()}}{NONE \\}
\vspace{2pt}
{\raggedright\errchico{ERROR 1142 (42000): SELECT command denied to user 'reporte'@'localhost' for table 'contacto'}\par}
\vspace{4pt}
\footnotesize\textbf{Con} el paso 3:
\resg{l}{\cab{CURRENT\_ROLE()}}{\bq solo\_lectura\bq @\bq \%\bq \\}
\end{columns}
\end{frame}

\begin{frame}{MySQL y SQL Server: usuarios y permisos}
\vspace{-0.15cm}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}>{\bfseries\raggedright}p{2.4cm}>{\raggedright}p{5.0cm}>{\raggedright\arraybackslash}p{5.3cm}@{}}
\toprule
 & MySQL & SQL Server \\
\midrule
Crear & \texttt{CREATE USER 'app\_agenda'@'localhost' IDENTIFIED BY '\ldots';} & Dos pasos: \texttt{CREATE LOGIN app\_agenda WITH PASSWORD = '\ldots';} y \texttt{CREATE USER app\_agenda FOR LOGIN app\_agenda;} \\
Dar & \texttt{GRANT SELECT ON agenda.* TO \ldots;} & \texttt{GRANT SELECT ON SCHEMA::dbo TO app\_agenda;} \\
Quitar & \texttt{REVOKE DELETE ON agenda.* FROM \ldots;} & \texttt{REVOKE DELETE ON SCHEMA::dbo FROM app\_agenda;} \\
Un rol & \texttt{GRANT 'solo\_lectura' TO 'reporte'@'localhost';} & \texttt{ALTER ROLE solo\_lectura ADD MEMBER reporte;} \\
Roles que ya existen & \gris{---} & \texttt{db\_datareader} (leer), \texttt{db\_datawriter} (escribir) \\
Sin permiso & \texttt{ERROR 1142 (42000)} & \texttt{Msg 229, Level 14} \\
\bottomrule
\end{tabular}
\end{center}
\vspace{0.05cm}
\concepto{En SQL Server, dos niveles}{\small El \textbf{login} entra al servidor; el \textbf{usuario} entra a una base. En MySQL, el usuario hace las dos cosas.}
\end{frame}
